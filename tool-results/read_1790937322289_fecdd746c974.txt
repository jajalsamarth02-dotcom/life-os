   101→function ErpConnectionCard() {

   113→
   114→  const invalidateAll = () => qc.invalidateQueries();

   116→  const connect = useMutation({
   117→    mutationFn: () => connectErp(username.trim(), password),

   119→      setConnectError(null);
   120→      setPassword("");
   121→      invalidateAll();

   132→    },

   136→    mutationFn: disconnectErp,
   137→    onSuccess: () => {
   138→      invalidateAll();

   141→    onError: (e) => toast.error("Disconnect failed", { description: e.message }),
   142→  });

   145→    return (

   147→        <CardHeader className="pb-3">
   148→          <Skeleton className="h-5 w-40" />
   149→          <Skeleton className="h-3 w-64" />
   150→        </CardHeader>
   151→        <CardContent className="space-y-3">
   152→          <Skeleton className="h-10 w-full" />

   156→      </Card>

   158→  }
   159→
   160→  return (
   161→    <Card className="rounded-xl border-border/60">
   162→      <CardHeader className="pb-3">
   163→        <CardTitle className="flex items-center gap-2 text-base tracking-tight">
   164→          <KeyRound className="size-4 text-primary" aria-hidden="true" />
   165→          ERP connection
   166→        </CardTitle>
   167→        <CardDescription>
   168→          Connect with your own student credentials from erp.gmiu.edu.in.
   169→        </CardDescription>
   170→      </CardHeader>
   171→      <CardContent className="space-y-4">
   172→        {erp.connected ? (
   173→          <div className="space-y-3">
   174→            <div className="flex flex-wrap items-center gap-3 rounded-lg border border-emerald-500/30 bg-emerald-500/10 p-4">
   175→              <CheckCircle2 className="size-6 shrink-0 text-emerald-600 dark:text-emerald-400" aria-hidden="true" />
   176→              <div className="min-w-0 flex-1">
   177→                <p className="text-sm font-semibold text-emerald-700 dark:text-emerald-400">
   178→                  Connected as {erp.username}
   179→                </p>
   180→                <p className="text-xs text-muted-foreground">
   181→                  Status: {erp.connectionStatus ?? "ok"} · Last verified: {fmtDT(erp.lastVerifiedAt)}
   182→                </p>
   183→              </div>
   184→              {erp.liveAvailable !== "unknown" && (
   185→                <Badge variant="outline" className="shrink-0">
   186→                  live sync: {erp.liveAvailable}
   187→                </Badge>
   188→              )}
   189→            </div>
   190→
   191→            <AlertDialog open={confirmDisconnect} onOpenChange={setConfirmDisconnect}>
   192→              <AlertDialogTrigger asChild>

   306→function SyncSettingsForm({ initial }: { initial: AppSettings }) {
   307→  const qc = useQueryClient();
   308→  const [target, setTarget] = useState<number>(initial.targetPercentage);
   309→  const [interval, setIntervalMin] = useState<number>(initial.syncIntervalMinutes);
   310→  const [autoSync, setAutoSync] = useState<boolean>(initial.autoSyncEnabled);

   313→    mutationFn: () =>
   314→      putSettings({ targetPercentage: target, syncIntervalMinutes: interval, autoSyncEnabled: autoSync }),
   315→    onSuccess: () => {
   316→      qc.invalidateQueries();
   317→      toast.success("Settings saved", {
   318→        description: `Target ${target}% · sync every ${interval} min${autoSync ? " · auto-sync on" : " · auto-sync off"}`,

   321→    onError: (e) => toast.error("Could not save settings", { description: e.message }),

   454→function NotificationPrefsCard() {
   455→  const qc = useQueryClient();
   456→  const { data: settings } = useQuery({ queryKey: qk.settings, queryFn: fetchSettings });
   457→  const [draft, setDraft] = useState<Record<string, boolean> | null>(null);
   458→
   459→  const prefs = draft ?? settings?.notifyPrefs ?? {};
   460→  const enabledCount = NOTIFY_META.filter((m) => prefs[m.key] !== false).length;
   461→
   462→  const save = useMutation({
   463→    mutationFn: (next: Record<string, boolean>) => putSettings({ notifyPrefs: next }),
   464→    onSuccess: (_s, next) => {
   465→      setDraft(null);
   466→      qc.invalidateQueries({ queryKey: qk.settings });
   467→      const off = Object.entries(next).filter(([, v]) => !v).length;
   468→      toast.success(
   469→        off > 0 ? `Notifications updated — ${off} type${off === 1 ? "" : "s"} muted` : "All notifications enabled"
   470→      );
   471→    },
   472→    onError: (e) => toast.error("Could not save notification preferences", { description: e.message }),
   473→  });
   474→
   475→  const setPref = (key: string, value: boolean) => {
   476→    setDraft({ ...prefs, [key]: value });
   477→  };
   478→
   479→  const dirty =
   480→    draft !== null &&
   481→    NOTIFY_META.some((m) => (draft[m.key] ?? true) !== (settings?.notifyPrefs?.[m.key] ?? true));
   482→
   483→  return (
   484→    <Card className="rounded-xl border-border/60">
   485→      <CardHeader className="flex flex-row items-start justify-between space-y-0 pb-3">
   486→        <div>
   487→          <CardTitle className="flex items-center gap-2 text-base tracking-tight">
   488→            <Bell className="size-4 text-primary" aria-hidden="true" />
   489→            Notifications
   490→          </CardTitle>
   491→          <CardDescription>Choose which events create a notification</CardDescription>
   492→        </div>
   493→        <Badge variant="secondary" className="text-[11px] tabular-nums">
   494→          {enabledCount}/{NOTIFY_META.length} on
   495→        </Badge>
   496→      </CardHeader>
   497→      <CardContent className="space-y-3">
   498→        {!settings ? (
   499→          <div className="space-y-2">
   500→            {Array.from({ length: 3 }).map((_, i) => (
   501→              <Skeleton key={i} className="h-11 w-full rounded-lg" />
   502→            ))}
   503→          </div>
   504→        ) : (
   505→          <div className="grid gap-2 sm:grid-cols-2">
   506→            {NOTIFY_META.map((m) => {
   507→              const on = prefs[m.key] !== false;
   508→              const Icon = on ? m.icon : BellOff;
   509→              return (
   510→                <div
   511→                  key={m.key}
   512→                  className={cn(
   513→                    "flex items-center gap-3 rounded-lg border p-3 transition-colors",
   514→                    on ? "border-border/50 bg-card" : "border-border/40 bg-muted/40"
   515→                  )}
   516→                >
   517→                  <span
   518→                    className={cn(
   519→                      "flex size-8 shrink-0 items-center justify-center rounded-lg",
   520→                      on ? "bg-muted" : "bg-muted/60 opacity-50"
   521→                    )}
   522→                    aria-hidden="true"
   523→                  >
   524→                    <Icon className={cn("size-4", on ? m.className : "text-muted-foreground")} />
   525→                  </span>
   526→                  <div className="min-w-0 flex-1">
   527→                    <Label htmlFor={`notify-${m.key}`} className={cn("text-sm", !on && "text-muted-foreground")}>
   528→                      {m.label}
   529→                    </Label>
   530→                    <p className="truncate text-[11px] text-muted-foreground">{m.description}</p>
   531→                  </div>
   532→                  <Switch
   533→                    id={`notify-${m.key}`}
   534→                    checked={on}
   535→                    onCheckedChange={(v) => setPref(m.key, v)}
   536→                    disabled={save.isPending}
   537→                    aria-label={`${on ? "Disable" : "Enable"} ${m.label} notifications`}
   538→                  />
   539→                </div>
   540→              );
   541→            })}
   542→          </div>
   543→        )}
   544→        <div className="flex items-center justify-between gap-3">
   545→          <p className="text-[11px] text-muted-foreground">
   546→            Applies to future events — existing notifications are kept.
   547→          </p>
   548→          <Button
   549→            size="sm"
   550→            className="h-8 gap-1.5"
   551→            disabled={!dirty || save.isPending}
   552→            onClick={() => save.mutate(draft ?? {})}
   553→            aria-label="Save notification preferences"
   554→          >
   555→            {save.isPending ? <Loader2 className="size-3.5 animate-spin" aria-hidden="true" /> : <Save className="size-3.5" aria-hidden="true" />}
   556→            Save
   557→          </Button>
   558→        </div>
   559→      </CardContent>
   560→    </Card>
   561→  );
   562→}
