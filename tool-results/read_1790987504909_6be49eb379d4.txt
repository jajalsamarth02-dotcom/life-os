    73→/** Tiny CGPA sparkline from the REGULAR exams' printed career CGPA line. */
    74→function CgpaSparkline({ points }: { points: CgpaPoint[] }) {
    75→  if (points.length < 2) return null;
    76→  const W = 96;
    77→  const H = 26;
    78→  const PAD = 3;
    79→  const vals = points.map((p) => p.cgpa);
    80→  const min = Math.min(...vals);
    81→  const max = Math.max(...vals);
    82→  const span = Math.max(max - min, 0.8); // never amplify a near-flat line
    83→  const xs = points.map((_, i) => PAD + (i * (W - 2 * PAD)) / (points.length - 1));
    84→  const ys = vals.map((v) => H - PAD - ((v - min) / span) * (H - 2 * PAD));
    85→  const line = xs.map((x, i) => `${i ? "L" : "M"}${x.toFixed(1)},${ys[i].toFixed(1)}`).join(" ");
    86→  const area = `${line} L${xs[xs.length - 1].toFixed(1)},${H - PAD} L${xs[0].toFixed(1)},${H - PAD} Z`;
    87→  const last = points.length - 1;
    88→  return (
    89→    <svg
    90→      width={W}
    91→      height={H}
    92→      viewBox={`0 0 ${W} ${H}`}
    93→      className="mt-1.5"
    94→      role="img"
    95→      aria-label={`CGPA progression: ${vals.map((v) => v.toFixed(2)).join(" → ")}`}
    96→    >
    97→      <defs>
    98→        <linearGradient id="cgpa-spark-fill" x1="0" y1="0" x2="0" y2="1">
    99→          <stop offset="0%" stopColor="rgb(139 92 246)" stopOpacity="0.28" />
   100→          <stop offset="100%" stopColor="rgb(139 92 246)" stopOpacity="0" />
   101→        </linearGradient>
   102→      </defs>
   103→      <path d={area} fill="url(#cgpa-spark-fill)" />
   104→      <path d={line} fill="none" stroke="rgb(139 92 246)" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   105→      {xs.map((x, i) => (
   106→        <circle
   107→          key={i}
   108→          cx={x}
   109→          cy={ys[i]}
   110→          r={i === last ? 2.2 : 1.5}
   111→          fill="rgb(139 92 246)"
   112→          opacity={i === last ? 1 : 0.65}
   113→        />
   114→      ))}
   115→      <text x={xs[last]} y={ys[last] - 4} textAnchor="end" fontSize="6.5" fill="currentColor" opacity="0.6">
   116→        S{points[last].semester ?? "?"}
   117→      </text>
   118→    </svg>
   119→  );
   120→}

   122→function DeltaChip({ delta }: { delta: number }) {
   123→  const up = delta > 0.0005;
   124→  const down = delta < -0.0005;
   125→  return (
   126→    <span
   127→      className={cn(
   128→        "inline-flex items-center gap-0.5 rounded-full border px-1.5 py-0.5 font-mono text-[10px] tabular-nums",
   129→        up
   130→          ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300"
   131→          : down
   132→            ? "border-rose-500/40 bg-rose-500/10 text-rose-700 dark:text-rose-300"
   133→            : "border-border bg-muted text-muted-foreground"
   134→      )}
   135→    >
   136→      {up && <TrendingUp className="size-3" aria-hidden="true" />}
   137→      {down && <TrendingDown className="size-3" aria-hidden="true" />}
   138→      {up ? "+" : ""}
   139→      {delta.toFixed(2)}
   140→    </span>
   141→  );
   142→}

   144→function VerdictChip({ result }: { result: string | null }) {
   145→  const passed = /pass/i.test(result ?? "");
   146→  const known = result != null;
   147→  return (
   148→    <span
   149→      className={cn(
   150→        "inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[11px] font-medium",
   151→        !known
   152→          ? "border-border bg-muted text-muted-foreground"
   153→          : passed
   154→            ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300"
   155→            : "border-rose-500/40 bg-rose-500/10 text-rose-700 dark:text-rose-300"
   156→      )}
   157→    >
   158→      {known && (passed ? <CheckCircle2 className="size-3" aria-hidden="true" /> : <AlertTriangle className="size-3" aria-hidden="true" />)}
   159→      {result ?? "—"}
   160→    </span>
   161→  );
   162→}

   325→export function ResultsCard({ className }: { className?: string }) {
   326→  const queryClient = useQueryClient();
   327→  const [refreshing, setRefreshing] = useState(false);
   328→  const [whatIfSgpa, setWhatIfSgpa] = useState("");
   329→  const [whatIfCredits, setWhatIfCredits] = useState("");
   330→  const query = useQuery({ queryKey: qk.results, queryFn: () => fetchResults() });
   331→  const data = query.data;
   332→  const summary = data ? deriveResultsSummary(data.exams) : null;
   333→
   334→  const onRefresh = async () => {
   335→    setRefreshing(true);
   336→    try {
   337→      const fresh = await fetchResults(true);
   338→      queryClient.setQueryData(qk.results, fresh);
   339→      if (fresh.exams.length > 0) toast.success("Exam results refreshed from the ERP");
   340→      else toast.message("No published results on the ERP yet");
   341→    } catch (e) {
   342→      toast.error(`Refresh failed: ${compactErrorText(e instanceof Error ? e.message : String(e), 120)}`);
   343→    } finally {
   344→      setRefreshing(false);
   345→    }
   346→  };
   347→
   348→  const creditsPct =
   349→    summary && summary.creditsEarned != null && summary.creditsOffered
   350→      ? Math.min(100, Math.round((summary.creditsEarned / summary.creditsOffered) * 100))
   351→      : null;
   352→  const maxSgpa = Math.max(10, ...(summary?.regularSeries.map((p) => p.sgpa) ?? [0]));
   353→  const progression = data ? cgpaProgression(data.exams) : [];
   354→  const defaultNextCredits = summary?.latestRegular?.offeredCredit ?? null;
   355→  const parsedSgpa = parseFloat(whatIfSgpa);
   356→  const parsedCredits = parseFloat(whatIfCredits) || defaultNextCredits;
   357→  const sgpaValid = whatIfSgpa.trim() !== "" && !isNaN(parsedSgpa) && parsedSgpa >= 0 && parsedSgpa <= 10;
   358→  const creditsValid = parsedCredits != null && parsedCredits > 0;
   359→  const projected =
   360→    summary && sgpaValid && creditsValid
   361→      ? projectCgpa(summary.cgpa, summary.creditsOffered, parsedSgpa, parsedCredits)
   362→      : null;
   363→
   364→  return (
   365→    <Card className={cn("overflow-hidden", className)} data-testid="results-card">
   366→      <CardHeader className="pb-3">
   367→        <div className="flex items-start gap-3">
   368→          <span
   369→            className="flex size-8 shrink-0 items-center justify-center rounded-md bg-violet-500/10 text-violet-600 dark:text-violet-400"
   370→            aria-hidden="true"
   371→          >
   372→            <GraduationCap className="size-4" />
   373→          </span>
   374→          <div className="min-w-0 flex-1">
   375→            <CardTitle className="text-base tracking-tight">Academic record</CardTitle>
   376→            <CardDescription className="text-xs">
   377→              Published exam results from the ERP&apos;s Final Results page — grades, SGPA/CGPA and pending subjects.
   378→            </CardDescription>
   379→          </div>
   380→          <div className="flex shrink-0 items-center gap-1.5 print:hidden">
   381→            {data?.fetchedAt && (
   382→              <span className="hidden text-[10px] text-muted-foreground sm:inline" title="Last fetched from the ERP">
   383→                {ago(data.fetchedAt)}
   384→              </span>
   385→            )}
   386→            {data?.stale && (
   387→              <Badge variant="outline" className="border-amber-500/40 text-[10px] text-amber-700 dark:text-amber-300">
   388→                cached
   389→              </Badge>
   390→            )}
   391→            <Button
   392→              variant="ghost"
   393→              size="sm"
   394→              className="h-7 gap-1 px-2 text-xs"
   395→              onClick={onRefresh}
   396→              disabled={refreshing || query.isLoading}
   397→              aria-label="Refresh exam results from the ERP"
   398→            >
   399→              <RefreshCw className={cn("size-3.5", (refreshing || query.isLoading) && "animate-spin")} aria-hidden="true" />
   400→              Refresh
   401→            </Button>
   402→          </div>
   403→        </div>
   404→      </CardHeader>
   405→      <CardContent className="space-y-3">
   406→        {query.isLoading ? (
   407→          <div className="space-y-2" aria-label="Loading exam results">
   408→            <div className="h-14 animate-pulse rounded-lg bg-muted" />
   409→            <div className="h-16 animate-pulse rounded-lg bg-muted" />
   410→            <div className="h-10 animate-pulse rounded-lg bg-muted" />
   411→          </div>
   412→        ) : !data ? (
   413→          <p className="py-4 text-center text-sm text-muted-foreground">Could not load exam results.</p>
   414→        ) : !data.available ? (
   415→          <p className="py-4 text-center text-sm text-muted-foreground">{data.note ?? "Connect your ERP account in Settings to load exam results."}</p>
   416→        ) : data.exams.length === 0 ? (
   417→          <div className="rounded-lg border border-dashed py-5 text-center">
   418→            <p className="text-sm font-medium">No published results yet</p>
   419→            <p className="mx-auto mt-1 max-w-sm text-xs text-muted-foreground">
   420→              {data.note ?? "The ERP lists no Final Results for your enrollment — the card fills in the moment one is published."}
   421→            </p>
   422→          </div>
   423→        ) : summary ? (
   424→          <>
   425→            {/* hero — latest REGULAR exam is the career line */}
   426→            <div className="flex flex-wrap items-center justify-between gap-4 rounded-lg border border-violet-500/20 bg-gradient-to-br from-violet-500/10 via-indigo-500/5 to-transparent px-4 py-3">
   427→              <div>
   428→                <div className="flex items-baseline gap-2">
   429→                  <Award className="size-5 self-center text-violet-500/70" aria-hidden="true" />
   430→                  <span className="text-3xl font-bold tabular-nums tracking-tight text-violet-700 dark:text-violet-300">
   431→                    {summary.cgpa?.toFixed(2) ?? "—"}
   432→                  </span>
   433→                  <div className="leading-tight">
   434→                    <p className="text-sm font-medium">CGPA</p>
   435→                    <p className="text-[11px] text-muted-foreground">
   436→                      {summary.latestRegular?.examDateLabel
   437→                        ? `career line after ${summary.latestRegular.examDateLabel}`
   438→                        : "career cumulative"}
   439→                    </p>
   440→                  </div>
   441→                </div>
   442→                <CgpaSparkline points={progression} />
   443→              </div>
   444→              <div className="min-w-40 flex-1 sm:max-w-56">
   445→                <div className="mb-1 flex items-baseline justify-between text-[11px]">
   446→                  <span className="font-medium tabular-nums">
   447→                    {summary.creditsEarned ?? "—"} / {summary.creditsOffered ?? "—"} credits
   448→                  </span>
   449→                  <span className="tabular-nums text-muted-foreground">{creditsPct != null ? `${creditsPct}%` : ""}</span>
   450→                </div>
   451→                <div className="h-1.5 overflow-hidden rounded-full bg-muted" role="presentation">
   452→                  <div
   453→                    className="h-full rounded-full bg-gradient-to-r from-violet-500 to-indigo-500 transition-[width] duration-700"
   454→                    style={{ width: `${creditsPct ?? 0}%` }}
   455→                  />
   456→                </div>
   457→                <p className="mt-1 text-[10px] text-muted-foreground">
   458→                  earned / offered · as printed on the latest regular result
   459→                </p>
   460→              </div>
   461→            </div>
   462→
   463→            {/* pending backlogs — the honest current state */}
   464→            {summary.pending.length > 0 ? (
   465→              <div className="rounded-lg border border-amber-500/30 bg-amber-500/5 px-3 py-2.5" data-testid="pending-backlogs">
   466→                <p className="flex items-center gap-1.5 text-xs font-medium text-amber-700 dark:text-amber-300">
   467→                  <AlertTriangle className="size-3.5" aria-hidden="true" />
   468→                  {summary.pending.length} subject{summary.pending.length === 1 ? "" : "s"} not cleared yet
   469→                </p>
   470→                <ul className="mt-1.5 space-y-0.5">
   471→                  {summary.pending.map((p) => (
   472→                    <PendingItem key={p.code} pending={p} exams={data.exams} />
   473→                  ))}
   474→                </ul>
   475→                <p className="mt-1.5 text-[10px] leading-relaxed text-muted-foreground">
   476→                  Latest ERP mention per subject decides — earlier failures that a later exam cleared are not listed.
   477→                  The next remedial window is where these can be retried.
   478→                </p>
   479→              </div>
   480→            ) : (
   481→              <p className="flex items-center gap-1.5 rounded-lg border border-emerald-500/30 bg-emerald-500/5 px-3 py-2 text-xs font-medium text-emerald-700 dark:text-emerald-300">
   482→                <CheckCircle2 className="size-3.5" aria-hidden="true" />
   483→                No pending backlogs — every published subject result is cleared.
   484→              </p>
   485→            )}
   486→
   487→            {/* what-if — honest CGPA projection for the next semester */}
   488→            {summary.hasRecord && summary.cgpa != null && summary.creditsOffered != null && (
   489→              <div
   490→                className="rounded-lg border border-dashed border-violet-500/30 bg-violet-500/5 px-3 py-2.5"
   491→                data-testid="cgpa-whatif"
   492→              >
   493→                <div className="flex flex-wrap items-center gap-2 text-xs">
   494→                  <TrendingUp className="size-3.5 shrink-0 text-violet-500/70" aria-hidden="true" />
   495→                  <span className="font-medium">What if</span>
   496→                  <input
   497→                    type="number"
   498→                    inputMode="decimal"
   499→                    min={0}
   500→                    max={10}
   501→                    step={0.1}
   502→                    placeholder="SGPA"
   503→                    value={whatIfSgpa}
   504→                    onChange={(e) => setWhatIfSgpa(e.target.value)}
   505→                    aria-label="Projected SGPA for your next semester (0 to 10)"
   506→                    className="h-7 w-16 rounded border border-border bg-background px-1.5 text-center font-mono text-xs tabular-nums outline-none focus-visible:ring-2 focus-visible:ring-violet-500/40"
   507→                  />
   508→                  <span className="text-muted-foreground">next semester with</span>
   509→                  <input
   510→                    type="number"
   511→                    inputMode="numeric"
   512→                    min={1}
   513→                    max={45}
   514→                    step={1}
   515→                    value={whatIfCredits}
   516→                    onChange={(e) => setWhatIfCredits(e.target.value)}
   517→                    placeholder={defaultNextCredits != null ? String(defaultNextCredits) : "cr"}
   518→                    aria-label="Credits you will take next semester"
   519→                    className="h-7 w-14 rounded border border-border bg-background px-1.5 text-center font-mono text-xs tabular-nums outline-none placeholder:text-muted-foreground/60 focus-visible:ring-2 focus-visible:ring-violet-500/40"
   520→                  />
   521→                  <span className="text-muted-foreground">credits —</span>
   522→                  {projected != null && summary.cgpa != null ? (
   523→                    <span className="ml-auto inline-flex items-baseline gap-1.5">
   524→                      <span className="text-[11px] text-muted-foreground">CGPA →</span>
   525→                      <span className="text-lg font-bold tabular-nums tracking-tight text-violet-700 dark:text-violet-300">
   526→                        {projected.toFixed(3)}
   527→                      </span>
   528→                      <DeltaChip delta={projected - summary.cgpa} />
   529→                    </span>
   530→                  ) : (
   531→                    <span className="ml-auto text-[11px] text-muted-foreground">
   532→                      {sgpaValid && !creditsValid ? "credits must be 1–45" : "enter a target SGPA (0–10)"}
   533→                    </span>
   534→                  )}
   535→                </div>
   536→                <p className="mt-1 text-[10px] leading-relaxed text-muted-foreground">
   537→                  Estimate only — treats the ERP&apos;s career CGPA as cumulative points ÷ offered credits
   538→                  ({summary.cgpa.toFixed(3)} × {summary.creditsOffered}). Your real result follows the
   539→                  university&apos;s exact rules; clearing a pending subject next semester lifts it further.
   540→                </p>
   541→              </div>
   542→            )}
   543→
   544→            {/* SGPA trend — REGULAR exams as bars, remedials as markers */}
   545→            <div>
   546→              <p className="mb-1.5 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   547→                SGPA by semester (regular exams)
   548→              </p>
   549→              {summary.regularSeries.length > 0 ? (
   550→                <div className="flex items-end gap-2" data-testid="sgpa-trend">
   551→                  {summary.regularSeries.map((p) => (
   552→                    <div key={p.typeId} className="flex min-w-0 flex-1 flex-col items-center gap-1">
   553→                      <span className="font-mono text-[10px] tabular-nums text-muted-foreground">{p.sgpa.toFixed(2)}</span>
   554→                      <div
   555→                        className="flex w-full max-w-14 items-end justify-center rounded-t-md border-b border-border/60"
   556→                        style={{ height: 64 }}
   557→                        role="presentation"
   558→                      >
   559→                        <div
   560→                          className={cn(
   561→                            "w-full rounded-t-md transition-[height] duration-700",
   562→                            p.passed
   563→                              ? "bg-gradient-to-t from-emerald-500/70 to-emerald-400/80"
   564→                              : "bg-gradient-to-t from-rose-500/70 to-rose-400/80"
   565→                          )}
   566→                          style={{ height: `${Math.max(6, (p.sgpa / maxSgpa) * 100)}%` }}
   567→                          title={`Semester ${p.semester}: SGPA ${p.sgpa.toFixed(2)} (${p.passed ? "Pass" : "Fail"})`}
   568→                        />
   569→                      </div>
   570→                      <span className="text-[10px] font-medium text-muted-foreground">S{p.semester}</span>
   571→                    </div>
   572→                  ))}
   573→                </div>
   574→              ) : (
   575→                <p className="text-xs text-muted-foreground">No regular exam results parsed yet.</p>
   576→              )}
   577→              {summary.remedialSeries.length > 0 && (
   578→                <div className="mt-1.5 flex flex-wrap items-center gap-1.5" data-testid="remedial-markers">
   579→                  <span className="text-[10px] text-muted-foreground">Remedial attempts:</span>
   580→                  {summary.remedialSeries.map((p) => (
   581→                    <span
   582→                      key={p.typeId}
   583→                      className="inline-flex items-center gap-1 rounded-full border border-amber-500/40 bg-amber-500/10 px-2 py-0.5 font-mono text-[10px] tabular-nums text-amber-700 dark:text-amber-300"
   584→                      title={`Remedial attempt for semester ${p.semester}: SGPA ${p.sgpa.toFixed(2)} (${p.passed ? "Pass" : "Fail"})`}
   585→                    >
   586→                      ↺ S{p.semester} · {p.sgpa.toFixed(2)}
   587→                    </span>
   588→                  ))}
   589→                </div>
   590→              )}
   591→            </div>
   592→
   593→            {/* per-exam breakdown */}
   594→            <div className="space-y-1.5">
   595→              <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">All published results</p>
   596→              {data.exams.map((exam, i) => (
   597→                <ExamRow key={exam.typeId} exam={exam} defaultOpen={i === 0} />
   598→              ))}
   599→            </div>
   600→
   601→            {data.note && <p className="text-[10px] text-muted-foreground">{data.note}</p>}
   602→          </>
   603→        ) : null}
   604→      </CardContent>
   605→    </Card>
   606→  );
