    35→/** GSM 03.38 basic character set (single chars). */
    36→const GSM7_BASIC =
    37→  "@£$¥èéùìòÇ\nØø\rÅåΔ_ΦΓΛΩΠΨΣΘΞÆæßÉ !\"#¤%&'()*+,-./0123456789:;<=>?¡ABCDEFGHIJKLMNOPQRSTUVWXYZÄÖÑÜ§¿abcdefghijklmnopqrstuvwxyzäöñüà";
    38→/** GSM 03.38 extension table (each costs 2 chars). */
    39→const GSM7_EXT = "^{}\\[~]|€";

    41→export function smsSegments(text: string): {
    42→  segments: number;
    43→  chars: number;
    44→  encoding: "GSM-7" | "UCS-2";
    45→} {
    46→  let effective = 0;
    47→  let nonGsm = 0;
    48→  const chars = [...text].length;
    49→  for (const ch of text) {
    50→    if (GSM7_BASIC.includes(ch)) effective += 1;
    51→    else if (GSM7_EXT.includes(ch)) effective += 2;
    52→    else nonGsm += 1;
    53→  }
    54→  if (nonGsm === 0) {
    55→    return { segments: effective <= 160 ? 1 : Math.ceil(effective / 153), chars, encoding: "GSM-7" };
    56→  }
    57→  return { segments: chars <= 70 ? 1 : Math.ceil(chars / 67), chars, encoding: "UCS-2" };
    58→}

    64→const IST_TZ = "Asia/Kolkata";

    66→/** Calendar parts of an instant, rendered in IST. Some ICU builds render
    67→ *  midnight as "24" — folded back to "00" here too (shared.tsx's istNow
    68→ *  already does this for the schedule math; the DISPLAY side must agree,
    69→ *  or the bubble header would print "generated 24:00 IST"). */
    70→function istParts(now: number) {
    71→  const p = new Intl.DateTimeFormat("en-GB", {
    72→    timeZone: IST_TZ,
    73→    weekday: "short",
    74→    day: "2-digit",
    75→    month: "short",
    76→    year: "numeric",
    77→    hour: "2-digit",
    78→    minute: "2-digit",
    79→    hour12: false,
    80→  }).formatToParts(new Date(now));
    81→  const get = (t: string) => p.find((x) => x.type === t)?.value ?? "";
    82→  const h = Number(get("hour")) % 24;
    83→  return {
    84→    weekday: get("weekday").toUpperCase(),
    85→    day: get("day"),
    86→    month: get("month").toUpperCase(),
    87→    year: get("year"),
    88→    hhmm: `${String(h).padStart(2, "0")}:${get("minute")}`,
    89→  };
    90→}

    92→/** Force dynamic strings into the SMS-safe ASCII plane (typographic chars out). */
    93→function ascii(s: string): string {
    94→  return s
    95→    .replace(/[\u2018\u2019\u201B]/g, "'")
    96→    .replace(/[\u201C\u201D]/g, '"')
    97→    .replace(/[\u2013\u2014]/g, "-")
    98→    .replace(/\u2026/g, "...")
    99→    .replace(/\u00b7/g, "|")
   100→    .replace(/[^\x20-\x7e\n]/g, "");
   101→}

   142→/** Minutes-since-midnight → "HH:MM" (24h, zero-padded — GSM-7 friendly). */

   147→/** The today-card's live class state — ONE pure path for both the CLASSES
   148→ *  SMS line and the card's live chip, so the two can never disagree.
   149→ *  HONEST AT ANY DISTANCE: a class 8 hours out is still "next", never "done"
   150→ *  (the live smoke caught exactly that leak in Round 62 — the 90-min calm
   151→ *  window is the CHIP's business, see todayChipClass). Ongoing wins over
   152→ *  upcoming; null only when nothing is left today (or no data). */
   153→export function todayLiveClass(
   154→  todayClasses: TodayClassBrief[] | null | undefined,
   155→  now: number
   156→): TodayLiveClass | null {
   157→  if (!todayClasses || todayClasses.length === 0) return null;
   158→  /* Round 72: the local hh:mm parse is GONE — this is shared.tsx's istNow,
   159→   * the ONE IST clock every schedule surface already uses. Beyond the dedup
   160→   * it also folds the ICU "24:00" midnight quirk (the old parse read
   161→   * nowMin=1440 at IST midnight, silently flipping "upcoming" to "all done"). */
   162→  const nowMin = istNow(now).min;
   163→  const left = todayClasses
   164→    .filter((c) => c.endMin > nowMin)
   165→    .sort((a, b) => a.startMin - b.startMin);
   166→  const first = left[0];
   167→  if (!first) return null;
   168→  if (nowMin >= first.startMin) {
   169→    return { kind: "ongoing", cls: first, mins: first.endMin - nowMin, leftCount: left.length };
   170→  }
   171→  return { kind: "upcoming", cls: first, mins: first.startMin - nowMin, leftCount: left.length };
   172→}

   174→/** Chip variant of the state: an upcoming class more than 90 minutes out
   175→ *  keeps the card calm (null). Ongoing classes always show. The SMS line
   176→ *  deliberately does NOT use this — the bubble must stay honest at any
   177→ *  distance ("next 18:00" at 07:00 is true; "done" would be a lie). */
   178→export function todayChipClass(
   179→  todayClasses: TodayClassBrief[] | null | undefined,
   180→  now: number
   181→): TodayLiveClass | null {
   182→  const st = todayLiveClass(todayClasses, now);
   183→  if (st && st.kind === "upcoming" && st.mins > 90) return null;

   187→/** How LOUD the chip should be — the urgency ladder both class chips share
   188→ *  (today-card + header), so the surfaces can never disagree on tone:
   189→ *   - "ongoing":  class happening NOW → EMERALD, live pulse (r82: was amber
   190→ *                 — amber now means "a transition is approaching", the same
   191→ *                 language the banner/rows/week-chips speak since r78);
   192→ *   - "wrapping": ongoing with ≤ JUST_ENDED_WINDOW_MIN left → amber, slow
   193→ *                 ping (the end is near — symmetric with imminent);
   194→ *   - "imminent": starts in ≤ IMMINENT_MIN → amber, FAST ping (get moving);
   195→ *   - "soon":     inside the 90-min calm window → emerald, calm;
   196→ *   - null:       nothing to say (no class / too far out). */
   197→export type ChipUrgency = "ongoing" | "wrapping" | "imminent" | "soon";
   198→
   199→export const IMMINENT_MIN = 10;
   200→
   201→/** The end-of-class mirror of IMMINENT_MIN (r78 "wrapping up"): the last
   202→ *  JUST_ENDED_WINDOW_MIN minutes of an ongoing class flip amber across ALL
   203→ *  schedule surfaces. Home moved here in r82 so the chip-urgency ladder can
   204→ *  consume it without a today-card → timetable-card import (which would
   205→ *  cycle — timetable-card imports IMMINENT_MIN from here). One constant,
   206→ *  re-exported by timetable-card for its row/chip/bar consumers. */
   207→export const JUST_ENDED_WINDOW_MIN = 15;
   208→
   209→export function chipUrgency(st: TodayLiveClass | null): ChipUrgency | null {
   210→  if (!st) return null;
   211→  if (st.kind === "ongoing") {
   212→    /* r82: the last sync-cadence window of the class is a TRANSITION — the
   213→       chips now flip amber for it exactly like the banner (r78), row (r79)
   214→       and week-grid chip (r80) already do. One threshold, imported from
   215→       this module by every surface. */
   216→    return st.mins <= JUST_ENDED_WINDOW_MIN ? "wrapping" : "ongoing";
   217→  }
   218→  return st.mins <= IMMINENT_MIN ? "imminent" : "soon";
   219→}
   220→
   221→/** The post-class bridge, chip edition (r83): when todayLiveClass has run
   222→ *  dry — no ongoing, nothing upcoming — but the LAST class ended within
   223→ *  JUST_ENDED_WINDOW_MIN, the day isn't "done" yet: the auto-sync hasn't
   224→ *  recorded the class. Returns the just-ended class + minutes since its
   225→ *  end, or null (a live state still speaks / bridge closed / no data).
   226→ *  The today-card chip thereby keeps speaking the five-state language
   227→ *  until the sync takes over, instead of vanishing the instant the class
   228→ *  ends — the same honest gap the timetable rows and week-grid chips have
   229→ *  bridged since r75/r76. r84: also returns drainPct — the same depleting
   230→ *  amber math as the timetable's justEndedDrainPct (r81), computed inline
   231→ *  over the SAME pinned constant (today-card cannot import timetable-card
   232→ *  without cycling; the test suite pins the two against each other), so
   233→ *  the chip's bridge visibly burns down while the sync is pending. Pure —
   234→ *  pinned in tests. */
   235→export function justEndedChip(
   236→  todayClasses: TodayClassBrief[] | null | undefined,
   237→  now: number
   238→): { cls: TodayClassBrief; sinceMin: number; drainPct: number } | null {
   239→  if (!todayClasses || todayClasses.length === 0) return null;
   240→  /* Precedence: an ongoing/upcoming state is the day's voice — the chip
   241→     must not say "just ended" while a later class still speaks. */
   242→  if (todayLiveClass(todayClasses, now)) return null;
   243→  const nowMin = istNow(now).min;
   244→  const lastEnd = Math.max(...todayClasses.map((c) => c.endMin));
   245→  const sinceMin = nowMin - lastEnd;
   246→  if (sinceMin < 0 || sinceMin > JUST_ENDED_WINDOW_MIN) return null;
   247→  const cls = todayClasses.find((c) => c.endMin === lastEnd) as TodayClassBrief;
   248→  /* 100 at the end minute → 0 at the window boundary (inclusive). */
   249→  const drainPct = ((JUST_ENDED_WINDOW_MIN - sinceMin) / JUST_ENDED_WINDOW_MIN) * 100;
   250→  return { cls, sinceMin, drainPct };
   251→}

   253→/** The week-ahead "next up" pure (r87 → r90): once today is fully done
   254→ *  — every caller guards this first (header chip: !shown && !justEnded;
   255→ *  deadline strip: allDone) — the day arc should still answer "when IS the
   256→ *  next one?". Strictly LATER THIS WEEK: days-ahead = (day − today + 7)
   257→ *  % 7 ≥ 1; the +7 wrap is what keeps a Saturday honest about Monday (a
   258→ *  naive day > today comparison breaks at the week boundary). Earliest day
   259→ *  wins, then earliest start. Today is deliberately excluded (d ≥ 1): the
   260→ *  muted family's promise is "later this week", and the callers' guards
   261→ *  already mean today has nothing left to say. r90: promoted from the
   262→ *  header's inline 3-liner (live-verified r87) so the header chip and the
   263→ *  strip's done state share ONE math and can never drift. Pure — pinned
   264→ *  in tests (r90). */
   265→export function nextUpcomingThisWeek(
   266→  slots: TimetableSlotRow[] | null | undefined,
   267→  now: number
   268→): TimetableSlotRow | null {
   269→  if (!slots || slots.length === 0) return null;
   270→  const { day: today } = istNow(now);
   271→  return (
   272→    slots
   273→      .map((s) => ({ s, d: (s.dayOfWeek - today + 7) % 7 }))
   274→      .filter((x) => x.d >= 1)
   275→      .sort((a, b) => a.d - b.d || a.s.startMin - b.s.startMin)[0]?.s ?? null
   276→  );
   277→}
   278→
   279→/** Cyclic days-ahead for one weekday vs the IST clock (r91): 0 = today,
   280→ *  1 = tomorrow … 6 = the day before today next week. The SAME +7 wrap the
   281→ *  week-ahead selection uses — exported so the header chip and the strip's
   282→ *  done line qualify their "Mon 11:00" with ONE distance math instead of
   283→ *  two drifting re-derivations. Pure — pinned in tests (r91). */
   284→export function daysAheadThisWeek(dayOfWeek: number, now: number): number {
   285→  return (dayOfWeek - istNow(now).day + 7) % 7;
   286→}
   287→
   288→/** Display twin of daysAheadThisWeek (r91): 0 reads "today", 1 reads
   289→ *  "tomorrow" (the friendly, still-honest case), every other gap reads
   290→ *  "in Nd" — the humanizeMins dialect ("in 4h") one unit up. Pure —
   291→ *  pinned in tests (r91). */
   292→export function daysAheadLabel(d: number): string {
   293→  return d === 0 ? "today" : d === 1 ? "tomorrow" : `in ${d}d`;
   294→}
   295→
   296→/** r94: the qualifier's TONE is family math too — "tomorrow" (d=1) lifts one
   297→ *  rung of the r87 muted ladder (prep tonight is actionable tonight, so the
   298→ *  wait deserves full muted presence instead of /70), while "in Nd" stays at
   299→ *  the ladder's quietest rung. d=0 never reaches these surfaces (today is
   300→ *  excluded upstream) but stays quiet defensively. ONE source — the header
   301→ *  chip, the strip's done line and the today-card pill all consume it, so
   302→ *  the whole family shifts tone together. Pinned in tests (r94). */
   303→export function daysAheadQualifierClass(d: number): string {
   304→  return d === 1 ? "text-muted-foreground" : "text-muted-foreground/70";
   305→}

   307→/** Build the exact SMS text shown in the bubble (and used for copy/share). */
   308→export function buildTodaySms(input: TodaySmsInput): string {
   309→  const p = istParts(input.now);
   310→  const lines: string[] = [];
   311→
   312→  lines.push(`ERP SYNC - ${p.weekday} ${p.day} ${p.month} ${p.year}`);
   313→
   314→  const o = input.overall;
   315→  if (o.total === 0) {
   316→    lines.push("OVERALL no classes recorded yet");
   317→  } else {
   318→    lines.push(
   319→      `OVERALL ${o.present}/${o.total} (${o.percentage.toFixed(2)}%) vs ${input.target}% target - ${o.status.toUpperCase()}`
   320→    );
   321→    if (o.classesToTarget > 0) {
   322→      lines.push(
   323→        `NEED ${o.classesToTarget} more attended class${o.classesToTarget === 1 ? "" : "es"} to reach ${input.target}%`
   324→      );
   325→    } else if (o.canSkip > 0) {
   326→      lines.push(`CAN SKIP ${o.canSkip} class${o.canSkip === 1 ? "" : "es"} and stay above ${input.target}%`);
   327→    } else {
   328→      lines.push("ON TARGET - keep it up");
   329→    }
   330→    const budget = input.subjects.reduce((n, s) => n + (s.total > 0 ? s.canSkip : 0), 0);
   331→    lines.push(
   332→      budget > 0
   333→        ? `BUDGET ${budget} safe skip${budget === 1 ? "" : "s"} across subjects`
   334→        : "BUDGET 0 - attend everything"
   335→    );
   336→  }
   337→
   338→  /* Today's schedule — a time-sensitive fact, so it sits right after the
   339→     attendance block and before the worst-subjects roll-up. Honest absence
   340→     when no timetable has synced; "none scheduled" for a genuinely free day. */
   341→  if (input.todayClasses) {
   342→    const tc = input.todayClasses;
   343→    if (tc.length === 0) {
   344→      lines.push("CLASSES TODAY none scheduled");
   345→    } else {
   346→      const live = todayLiveClass(tc, input.now);
   347→      if (!live) {
   348→        lines.push(
   349→          tc.length === 1 ? "CLASSES TODAY done (1 class)" : `CLASSES TODAY all ${tc.length} done`
   350→        );
   351→      } else if (live.kind === "ongoing") {
   352→        lines.push(
   353→          `CLASSES TODAY ${live.leftCount} left - in class ${live.cls.subjectCode} until ${hhmmFromMin(live.cls.endMin)}`
   354→        );
   355→      } else {
   356→        lines.push(
   357→          `CLASSES TODAY ${live.leftCount} left - next ${hhmmFromMin(live.cls.startMin)} ${live.cls.subjectCode}`
   358→        );
   359→      }
   360→    }
   361→  }
   362→
   363→  const worst = input.subjects
   364→    .filter((s) => s.total > 0 && (s.status === "danger" || s.status === "warning"))
   365→    .sort((a, b) => a.percentage - b.percentage)
   366→    .slice(0, 2);
   367→  if (worst.length > 0) {
   368→    lines.push(
   369→      `WORST ${worst
   370→        .map((s) => `${s.code} ${s.percentage.toFixed(1)}% (${s.present}/${s.total})`)
   371→        .join(", ")}`
   372→    );
   373→  }
   374→
   375→  if (input.alaTotal > 0) {
   376→    let ala = `ALAS ${input.alaPending} pending / ${input.alaTotal} tracked`;
   377→    const bits: string[] = [];
   378→    if (input.alaOverdue > 0) bits.push(`${input.alaOverdue} overdue`);
   379→    if (input.alaDueWeek > 0) bits.push(`${input.alaDueWeek} due in 7 days`);
   380→    if (bits.length > 0) ala += ` (${bits.join(", ")})`;
   381→    lines.push(ala);
   382→  }
   383→
   384→  if (input.nextDue) {
   385→    const d = istParts(new Date(input.nextDue.dueAt).getTime());
   386→    const title = ascii(input.nextDue.title);
   387→    const clipped = title.length > 28 ? `${title.slice(0, 27)}...` : title;
   388→    lines.push(
   389→      `NEXT ${clipped} (${input.nextDue.subjectCode ?? "?"}) due ${d.weekday} ${d.day} ${d.month}`
   390→    );
   391→  }
   392→
   393→  if (input.notedCount > 0) {
   394→    lines.push(`NOTES ${input.notedCount} private note${input.notedCount === 1 ? "" : "s"} on this device`);
   395→  }
   396→
   397→  const synced = input.lastSyncAt ? istParts(new Date(input.lastSyncAt).getTime()).hhmm : null;
   398→  lines.push(`SYNCED ${synced ?? "not yet"} IST - erp.gmiu.edu.in`);
   399→
   400→  return ascii(lines.join("\n"));
   401→}

   407→/** Keywords that open the brief's lines — highlighted in the bubble's
   408→ *  RENDER only. The clipboard/print text stays byte-identical: these are
   409→ *  spans around the exact same characters, nothing added, nothing moved.
   410→ *  Exported so the tone picker's test can walk the FULL table. */
   411→export const SMS_KEYWORDS = [
   412→  "CLASSES TODAY", // longest first — before any shorter prefix could match
   413→  "OVERALL",
   414→  "NEED",
   415→  "CAN SKIP",
   416→  "ON TARGET",
   417→  "BUDGET",
   418→  "WORST",
   419→  "ALAS",
   420→  "NEXT",
   421→  "NOTES",
   422→  "SYNCED",
   423→];
   424→
   425→/** Split one brief line into { keyword, rest } for the bubble's styled
   426→ *  render — null when the line opens with no known keyword (the header,
   427→ *  "ERP SYNC - ...", stays plain). Exact character content is preserved
   428→ *  either way; rest always begins with the original whitespace. */
   429→export function smsLineParts(line: string): { keyword: string; rest: string } | null {
   430→  for (const k of SMS_KEYWORDS) {
   431→    if (line === k) return { keyword: k, rest: "" };
   432→    if (line.startsWith(k + " ")) return { keyword: k, rest: line.slice(k.length) };
   433→  }
   434→  return null;
   435→}

   441→/** Bubble keyword tones — the styled render color-codes line-openers by
   442→ *  their ROLE in the brief (clipboard text is untouched; these are paint
   443→ *  classes on the exact same characters):
   444→ *   - "WORST" → amber: the app's warning/danger language (status chips, ALA

   448→ *   - everything else → emerald: the card's own identity color.
   449→ *  Unknown keywords (future lines added without a tone decision) fall back

   451→export type SmsKeywordTone = "emerald" | "amber" | "violet";

   453→export function smsKeywordTone(keyword: string): SmsKeywordTone {
   454→  if (keyword === "WORST") return "amber";
   455→  if (keyword === "NEXT") return "violet";
   456→  return "emerald";

   467→/** Legacy clipboard fallback for insecure contexts (plain-HTTP views, older
   468→ *  WebViews where navigator.clipboard is undefined or rejects): an off-screen
   469→ *  textarea + execCommand("copy"). execCommand is deprecated but STILL the
   470→ *  only clipboard write path those contexts have — the honest ladder is
   471→ *  async-clipboard first, this second, a visible failure state last.
   472→ *  Selection is restored so an accidental user selection is never clobbered.
   473→ *  Guarded for non-DOM environments (bun tests import this module). */
   474→export function legacyCopy(text: string): boolean {
   475→  if (typeof document === "undefined") return false;
   476→  try {
   477→    const ta = document.createElement("textarea");
   478→    ta.value = text;
   479→    ta.setAttribute("readonly", "");
   480→    ta.style.position = "fixed";
   481→    ta.style.top = "-9999px";
   482→    ta.style.left = "-9999px";
   483→    ta.style.opacity = "0";
   484→    document.body.appendChild(ta);
   485→    const sel = document.getSelection();
   486→    const prev = sel && sel.rangeCount > 0 ? sel.getRangeAt(0) : null;
   487→    ta.select();
   488→    ta.setSelectionRange(0, text.length);
   489→    let ok = false;
   490→    try {
   491→      ok = document.execCommand("copy");
   492→    } catch {
   493→      ok = false;
   494→    }
   495→    document.body.removeChild(ta);
   496→    if (prev && sel) {
   497→      sel.removeAllRanges();
   498→      sel.addRange(prev);
   499→    }
   500→    return ok;
   501→  } catch {
   502→    return false;
   503→  }
   504→}
