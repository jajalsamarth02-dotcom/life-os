    93→function loadSavedSort(): SubjectSort {
    94→  if (typeof window === "undefined") return "lowest";
    95→  try {
    96→    const saved = window.localStorage.getItem(SUBJECT_SORT_KEY);
    97→    return SUBJECT_SORTS.some((o) => o.key === saved) ? (saved as SubjectSort) : "lowest";
    98→  } catch {
    99→    return "lowest";
   100→  }
   101→}

   103→/* No-data subjects (total === 0) have no attendance yet — a 0.00% is not a
   104→   real value, so they never compete with actual percentages: they sink to
   105→   the end of both directions with a name tie-break. */
   106→function noDataLast(a: SubjectView, b: SubjectView): number {
   107→  const an = a.total === 0;
   108→  const bn = b.total === 0;
   109→  if (an !== bn) return an ? 1 : -1;
   110→  return 0;
   111→}
   112→
   113→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
   114→  const copy = [...list];
   115→  switch (sort) {
   116→    case "lowest":
   117→      return copy.sort(
   118→        (a, b) => noDataLast(a, b) || a.percentage - b.percentage || a.name.localeCompare(b.name)
   119→      );
   120→    case "highest":
   121→      return copy.sort(
   122→        (a, b) => noDataLast(a, b) || b.percentage - a.percentage || a.name.localeCompare(b.name)
   123→      );
   124→    case "name":
   125→      return copy.sort((a, b) => a.name.localeCompare(b.name));
   126→    case "recent":
   127→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
   128→  }
   129→}

   131→/** Live countdown label for the next upcoming deadline. */
   132→function hhmmFromMin(min: number): string {
   133→  return `${String(Math.floor(min / 60)).padStart(2, "0")}:${String(min % 60).padStart(2, "0")}`;
   134→}
   135→
   136→function countdownLabel(dueAt: string, now: number): string {
   137→  const diff = new Date(dueAt).getTime() - now;
   138→  const abs = Math.abs(diff);
   139→  const mins = Math.round(abs / 60_000);
   140→  const hours = Math.round(abs / 3_600_000);
   141→  const days = Math.round(abs / 86_400_000);
   142→  let span: string;
   143→  if (mins < 60) span = `${mins} min`;
   144→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
   145→  else span = `${days} day${days === 1 ? "" : "s"}`;
   146→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
   147→}

   149→/** Relative urgency pill for an ALA deadline: red overdue/≤48h, amber ≤7d, muted later. */
   150→function deadlinePill(dueAt: string, now: number): { label: string; className: string } {
   151→  const due = new Date(dueAt).getTime();
   152→  const diff = due - now;
   153→  if (diff < 0)
   154→    return {
   155→      label: countdownLabel(dueAt, now),
   156→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   157→    };
   158→  if (diff <= 2 * 86_400_000)
   159→    return {
   160→      label: countdownLabel(dueAt, now),
   161→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   162→    };
   163→  if (diff <= 7 * 86_400_000)
   164→    return {
   165→      label: countdownLabel(dueAt, now),
   166→      className: "bg-amber-500/10 text-amber-700 dark:text-amber-400",
   167→    };
   168→  return { label: countdownLabel(dueAt, now), className: "bg-muted text-muted-foreground" };
   169→}

   438→function NextUpStrip({
   439→  items,
   440→  flagPriorities,
   441→}: {
   442→  items: Reminder[];
   443→  /** externalId → priority (1–3) from the local AlaNote overlay. */
   444→  flagPriorities: Map<string, number>;
   445→}) {
   446→  const now = useNow();
   447→  /* Next class TODAY from the synced routine — shares the timetable query
   448→     cache with the week plan + timetable cards, so no extra request. The
   449→     batch-pick math (same-batch collapse + parallel picks) matches the
   450→     week plan exactly, so the chip and the plan can never disagree. */
   451→  const { data: ttRows } = useQuery({ queryKey: qk.timetable, queryFn: fetchTimetable });
   452→  const nowDay = new Date(now).getDay();
   453→  const nowMin = new Date(now).getHours() * 60 + new Date(now).getMinutes();
   454→  const classInfo = useMemo(() => {
   455→    const attending = applyBatchPicks(ttRows?.slots ?? [], ttRows?.picks ?? {}).attending;
   456→    const today = attending.filter((s) => s.dayOfWeek === nowDay);
   457→    const left = today
   458→      .filter((s) => s.endMin > nowMin)
   459→      .sort((a, b) => a.startMin - b.startMin);
   460→    const next = left[0] ?? null;
   461→    return {
   462→      next,
   463→      ongoing: next !== null && next.startMin <= nowMin,
   464→      allDone: next === null && today.length > 0,
   465→    };
   466→  }, [ttRows, nowDay, nowMin]);
   467→  const upcoming = items
   468→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
   469→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
   470→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
   471→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   472→
   473→  if (upcoming.length === 0) return null;
   474→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   475→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   476→
   477→  return (
   478→    <div
   479→      className={cn(
   480→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   481→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   482→      )}
   483→    >
   484→      <span
   485→        className={cn(
   486→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   487→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   488→        )}
   489→      >
   490→        <Hourglass className="size-4" aria-hidden="true" />
   491→      </span>
   492→      <div className="min-w-0 flex-1">
   493→        <p className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   494→          <span className="relative flex size-1.5 shrink-0" aria-hidden="true">
   495→            <span

   717→/** Live "in 12 min"-style countdown for a future instant (null when absent/past). */
   718→function inCountdown(iso: string | null, now: number): string | null {
   719→  if (!iso) return null;
   720→  const diff = new Date(iso).getTime() - now;
   721→  if (diff <= 0) return null;
   722→  const mins = Math.max(1, Math.round(diff / 60_000));
   723→  if (mins < 60) return `in ${mins} min`;
   724→  const hrs = Math.floor(mins / 60);
   725→  const rem = mins % 60;
   726→  return rem > 0 ? `in ${hrs}h ${rem}m` : `in ${hrs}h`;
   727→}
