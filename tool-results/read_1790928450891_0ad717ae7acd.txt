    49→const SUBJECT_SORTS = [
    50→  { key: "lowest", label: "Lowest attendance first" },
    51→  { key: "highest", label: "Highest attendance first" },

    56→type SubjectSort = (typeof SUBJECT_SORTS)[number]["key"];

    58→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    59→  const copy = [...list];
    60→  switch (sort) {
    61→    case "lowest":
    62→      return copy.sort((a, b) => a.percentage - b.percentage);
    63→    case "highest":
    64→      return copy.sort((a, b) => b.percentage - a.percentage);
    65→    case "name":
    66→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    67→    case "recent":
    68→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    69→  }
    70→}

    72→/** Live countdown label for the next upcoming deadline. */
    73→function countdownLabel(dueAt: string, now: number): string {
    74→  const diff = new Date(dueAt).getTime() - now;
    75→  const abs = Math.abs(diff);
    76→  const mins = Math.round(abs / 60_000);
    77→  const hours = Math.round(abs / 3_600_000);
    78→  const days = Math.round(abs / 86_400_000);
    79→  let span: string;
    80→  if (mins < 60) span = `${mins} min`;
    81→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    82→  else span = `${days} day${days === 1 ? "" : "s"}`;
    83→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    84→}

    86→function NextUpStrip({ items }: { items: Reminder[] }) {
    87→  const now = useNow();
    88→  const upcoming = items
    89→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
    90→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
    91→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
    92→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
    93→
    94→  if (upcoming.length === 0) return null;
    95→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
    96→  const isOverdue = new Date(shown.dueAt).getTime() < now;
    97→
    98→  return (
    99→    <div
   100→      className={cn(
   101→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   102→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   103→      )}
   104→    >
   105→      <span
   106→        className={cn(
   107→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   108→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   109→        )}
   110→      >
   111→        <Hourglass className="size-4" aria-hidden="true" />
   112→      </span>
   113→      <div className="min-w-0 flex-1">
   114→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   115→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   116→        </p>
   117→        <p className="truncate text-sm font-medium">{shown.title}</p>
   118→      </div>
   119→      <div className="shrink-0 text-right">
   120→        <p
   121→          className={cn(
   122→            "text-sm font-bold tabular-nums",
   123→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"
   124→          )}
   125→        >
   126→          {countdownLabel(shown.dueAt, now)}
   127→        </p>
   128→        <p className="text-[11px] text-muted-foreground">{fmtDT(shown.dueAt)}</p>
   129→      </div>
   130→    </div>
   131→  );
   132→}

   136→function DueBadge({ iso, now }: { iso: string; now: number }) {
   137→  const d = new Date(iso);
   138→  if (d.getTime() < now && !isToday(d)) {
   139→    return <span className="text-xs font-medium text-red-600 dark:text-red-400">{fmtDT(iso)} · overdue</span>;
   140→  }
   141→  return <span className="text-xs text-muted-foreground">{fmtDT(iso)}</span>;
   142→}

   161→function RemindersPreview({
   162→  items,
   163→  todayCount,
   164→  weekCount,
   165→}: {
   166→  items: Reminder[];
   167→  todayCount: number;
   168→  weekCount: number;
   169→}) {
   170→  const now = useNow();
   171→  const groups: { key: string; label: string; className: string; items: Reminder[] }[] = [
   172→    { key: "overdue", label: "Overdue", className: "text-red-600 dark:text-red-400", items: [] },
   173→    { key: "today", label: "Today", className: "text-emerald-600 dark:text-emerald-400", items: [] },
   174→    { key: "tomorrow", label: "Tomorrow", className: "", items: [] },
   175→    { key: "week", label: "This week", className: "", items: [] },
   176→    { key: "later", label: "Later", className: "", items: [] },

   404→export function DashboardTab({
   405→  onOpenAttendance,
   406→  onOpenSettings,
   407→}: {
   408→  onOpenAttendance: () => void;
   409→  onOpenSettings: () => void;
   410→}) {
   411→  const [sort, setSort] = useState<SubjectSort>("lowest");
   412→  const [selectedSubject, setSelectedSubject] = useState<SubjectView | null>(null);
   413→  const { data, isPending, isError, error } = useQuery({
   414→    queryKey: qk.dashboard,
   415→    queryFn: fetchDashboard,
   416→    refetchInterval: 20_000,
   417→  });
   418→  const history = useQuery({
   419→    queryKey: qk.history,
   420→    queryFn: fetchHistory,
   421→    refetchInterval: 60_000,
   422→  });
   423→
   424→  const weekDelta = useMemo(() => {
   425→    const pts = history.data?.overall ?? [];
   426→    if (pts.length < 2) return null;
   427→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   428→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   429→    const last = pts[pts.length - 1];
   430→    const d = last.percentage - first.percentage;
   431→    return Math.abs(d) < 0.005 ? null : d;
   432→  }, [history.data]);
   433→
   434→  if (isPending) {
   435→    return (
   436→      <div className="space-y-4">
   437→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   438→          <CardSkeletonLight className="md:col-span-2 xl:col-span-2" />
   439→          <CardSkeletonLight />
   440→          <CardSkeletonLight />
   441→          <CardSkeletonLight />
   442→        </div>
   443→        <CardSkeletonLight />
   444→        <div className="grid gap-4 lg:grid-cols-2">
   445→          <CardSkeletonLight />
   446→          <CardSkeletonLight />
   447→        </div>
   448→      </div>
   449→    );
   450→  }
   451→
   452→  if (isError || !data) {
   453→    return (
   454→      <Card className="rounded-xl border-red-500/30 bg-red-500/5">
   455→        <CardContent className="flex items-center gap-3 p-6 text-sm text-red-600 dark:text-red-400">
   456→          <XCircle className="size-5 shrink-0" aria-hidden="true" />
   457→          Could not load the dashboard: {error?.message ?? "unknown error"}
   458→        </CardContent>
   459→      </Card>
   460→    );
   461→  }
   462→
   463→  const { overall, subjects, target, mode } = data;
   464→  const sorted = sortSubjects(subjects, sort);
   465→
   466→  if (mode === "none") {
   467→    return (
   468→      <EmptyHero
   469→        title="No ERP data available yet."
   470→        description="Connect your ERP account or run a first synchronization to fetch your attendance and reminders. Nothing is shown until real data arrives."
   471→        actionLabel="Connect / Sync ERP"
   472→        onAction={onOpenSettings}
   473→      />
   474→    );
   475→  }
   476→
   477→  return (
   478→    <div className="space-y-4">
   479→      {/* Subject detail drawer (also covers what-if planning) */}
   480→      <SubjectDrawer
   481→        subject={selectedSubject}
   482→        target={target}
   483→        open={selectedSubject !== null}
   484→        onOpenChange={(o) => {
   485→          if (!o) setSelectedSubject(null);
   486→        }}
   487→      />
   488→
   489→      {/* Row 0 — next deadline countdown */}
   490→      <NextUpStrip items={data.reminders.items} />
   491→
   492→      {/* Row 1 — overall + stats */}
   493→      <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   494→        <Card className="rounded-xl border-border/60 transition-shadow hover:shadow-md md:col-span-2 xl:col-span-2">
   495→          <CardContent className="flex flex-col items-center gap-4 p-6 sm:flex-row sm:gap-6">
   496→            <ProgressRing percentage={overall.percentage} target={target} size={150} animated />
   497→            <div className="min-w-0 flex-1 space-y-2.5 text-center sm:text-left">
   498→              <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   499→                <h2 className="text-lg font-semibold tracking-tight">Overall attendance</h2>
   500→                <StatusChip status={overall.status} target={target} />
   501→                {weekDelta !== null && (
   502→                  <span
   503→                    className={cn(
   504→                      "rounded-full px-2 py-0.5 text-[11px] font-semibold tabular-nums",
   505→                      weekDelta >= 0
   506→                        ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   507→                        : "bg-red-500/10 text-red-600 dark:text-red-400"
   508→                    )}
   509→                    title="Change over the last 7 days"
   510→                  >
