    66→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    67→  const copy = [...list];
    68→  switch (sort) {
    69→    case "lowest":
    70→      return copy.sort((a, b) => a.percentage - b.percentage);
    71→    case "highest":
    72→      return copy.sort((a, b) => b.percentage - a.percentage);
    73→    case "name":
    74→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    75→    case "recent":
    76→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    77→  }
    78→}

    80→/** Live countdown label for the next upcoming deadline. */
    81→function countdownLabel(dueAt: string, now: number): string {
    82→  const diff = new Date(dueAt).getTime() - now;
    83→  const abs = Math.abs(diff);
    84→  const mins = Math.round(abs / 60_000);
    85→  const hours = Math.round(abs / 3_600_000);
    86→  const days = Math.round(abs / 86_400_000);
    87→  let span: string;
    88→  if (mins < 60) span = `${mins} min`;
    89→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    90→  else span = `${days} day${days === 1 ? "" : "s"}`;
    91→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    92→}

    94→function NextUpStrip({ items }: { items: Reminder[] }) {
    95→  const now = useNow();
    96→  const upcoming = items
    97→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
    98→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
    99→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
   100→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   101→
   102→  if (upcoming.length === 0) return null;
   103→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   104→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   105→
   106→  return (
   107→    <div
   108→      className={cn(
   109→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   110→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   111→      )}
   112→    >
   113→      <span
   114→        className={cn(
   115→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   116→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   117→        )}
   118→      >
   119→        <Hourglass className="size-4" aria-hidden="true" />
   120→      </span>
   121→      <div className="min-w-0 flex-1">
   122→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   123→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   124→        </p>
   125→        <p className="truncate text-sm font-medium">{shown.title}</p>
   126→      </div>
   127→      <div className="shrink-0 text-right">
   128→        <p
   129→          className={cn(
   130→            "text-sm font-bold tabular-nums",
   131→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"
   132→          )}
   133→        >
   134→          {countdownLabel(shown.dueAt, now)}
   135→        </p>
   136→        <p className="text-[11px] text-muted-foreground">{fmtDT(shown.dueAt)}</p>
   137→      </div>
   138→    </div>
   139→  );
   140→}

   148→  }
   149→  return <span className="text-xs text-muted-foreground">{fmtDT(iso)}</span>;
   150→}

   471→export function DashboardTab({
   472→  onOpenAttendance,
   473→  onOpenSettings,
   474→  onOpenReminders,
   475→}: {
   476→  onOpenAttendance: () => void;
   477→  onOpenSettings: () => void;
   478→  onOpenReminders: () => void;
   479→}) {
   480→  const [sort, setSort] = useState<SubjectSort>("lowest");
   481→  /* Store the CODE (not the object) and derive the view from fresh query data,
   482→     so the drawer reflects updated targets/percentages after refetches. */
   483→  const [selectedCode, setSelectedCode] = useState<string | null>(null);
   484→  const [digestOpen, setDigestOpen] = useState(false);
   485→  const [showAllChanges, setShowAllChanges] = useState(false);
   486→  const [showAllNotifs, setShowAllNotifs] = useState(false);
   487→  const { data, isPending, isError, error } = useQuery({
   488→    queryKey: qk.dashboard,
   489→    queryFn: fetchDashboard,
   490→    refetchInterval: 20_000,
   491→  });
   492→  const history = useQuery({
   493→    queryKey: qk.history,
   494→    queryFn: fetchHistory,
   495→    refetchInterval: 60_000,
   496→  });
   497→
   498→  const weekDelta = useMemo(() => {
   499→    const pts = history.data?.overall ?? [];
   500→    if (pts.length < 2) return null;
   501→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   502→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   503→    const last = pts[pts.length - 1];
   504→    const d = last.percentage - first.percentage;
   505→    if (Math.abs(d) < 0.005) return null;
   506→    // Be honest about the window: when tracked history is younger than the
   507→    // 7-day span (e.g. live tracking just started), label the real span.
   508→    const spanDays = Math.max(
   509→      0,
   510→      (new Date(last.t).getTime() - new Date(first.t).getTime()) / 86_400_000
   511→    );
   512→    return { delta: d, spanDays, fullWeek: spanDays >= 6.5 };
   513→  }, [history.data]);
   514→
   515→  if (isPending) {
   516→    return (
   517→      <div className="space-y-4">
   518→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   519→          <CardSkeletonLight className="md:col-span-2 xl:col-span-2" />
   520→          <CardSkeletonLight />
   521→          <CardSkeletonLight />
   522→          <CardSkeletonLight />
   523→        </div>
   524→        <CardSkeletonLight />
   525→        <div className="grid gap-4 lg:grid-cols-2">
   526→          <CardSkeletonLight />
   527→          <CardSkeletonLight />
   528→        </div>
   529→      </div>
   530→    );
   531→  }
   532→
   533→  if (isError || !data) {
   534→    return (
   535→      <Card className="rounded-xl border-red-500/30 bg-red-500/5">
   536→        <CardContent className="flex items-center gap-3 p-6 text-sm text-red-600 dark:text-red-400">
   537→          <XCircle className="size-5 shrink-0" aria-hidden="true" />
   538→          Could not load the dashboard: {error?.message ?? "unknown error"}
   539→        </CardContent>
   540→      </Card>
   541→    );
   542→  }
   543→
   544→  const { overall, subjects, target, mode } = data;
   545→  const sorted = sortSubjects(subjects, sort);
   546→
   547→  if (mode === "none") {
   548→    return (
   549→      <EmptyHero
   550→        title="No ERP data available yet."
   551→        description="Connect your ERP account or run a first synchronization to fetch your attendance and reminders. Nothing is shown until real data arrives."
   552→        actionLabel="Connect / Sync ERP"
   553→        onAction={onOpenSettings}
   554→      />
   555→    );
   556→  }
   557→
   558→  return (
   559→    <motion.div
   560→      className="space-y-4"
   561→      initial="hidden"
   562→      animate="show"
   563→      variants={{
   564→        hidden: {},
   565→        show: { transition: { staggerChildren: 0.05 } },
   566→      }}
   567→    >
   568→      {/* Subject detail drawer (also covers what-if planning) */}
   569→      <SubjectDrawer
   570→        subject={data.subjects.find((s) => s.code === selectedCode) ?? null}
   571→        target={target}
   572→        globalTarget={target}
   573→        open={selectedCode !== null}
   574→        onOpenChange={(o) => {
   575→          if (!o) setSelectedCode(null);
   576→        }}
   577→      />
   578→
   579→      {/* Row 0 — next deadline countdown */}
   580→      <NextUpStrip items={data.reminders.items} />
   581→
   582→      {/* Row 1 — overall + stats */}
   583→      <motion.div
   584→        className="grid gap-4 md:grid-cols-2 xl:grid-cols-5"
   585→        variants={{ hidden: { opacity: 0, y: 10 }, show: { opacity: 1, y: 0, transition: { duration: 0.25, ease: "easeOut" } } }}
   586→      >
   587→        <Card className="erp-card-accent rounded-xl border-border/60 transition-shadow hover:shadow-md md:col-span-2 xl:col-span-2">
   588→          <CardContent className="flex flex-col items-center gap-4 p-6 sm:flex-row sm:gap-6">
   589→            <ProgressRing percentage={overall.percentage} target={target} size={150} animated />
   590→            <div className="min-w-0 flex-1 space-y-2.5 text-center sm:text-left">
   591→              <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   592→                <h2 className="text-lg font-semibold tracking-tight">Overall attendance</h2>
   593→                {data.semester !== "all" && (
   594→                  <Badge
   595→                    variant="outline"
   596→                    className="gap-1 border-primary/40 text-primary"
   597→                    title="A semester partition is active — only this semester's ERP subjects are counted"
   598→                  >
   599→                    <GraduationCap className="size-3" aria-hidden="true" />
   600→                    Sem {data.semester}
   601→                  </Badge>
   602→                )}
   603→                <StatusChip status={overall.status} target={target} />
   604→                <Button
   605→                  variant="ghost"
   606→                  size="sm"
   607→                  className="ml-auto h-7 gap-1 rounded-full px-2.5 text-xs text-muted-foreground hover:text-foreground"
   608→                  onClick={() => setDigestOpen(true)}
   609→                  aria-label="Open the weekly attendance report"
   610→                >
   611→                  <CalendarRange className="size-3.5" aria-hidden="true" />
   612→                  This week
   613→                </Button>
   614→                {weekDelta !== null && (
   615→                  <span
   616→                    className={cn(
   617→                      "rounded-full px-2 py-0.5 text-[11px] font-semibold tabular-nums",
   618→                      weekDelta.delta >= 0
   619→                        ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   620→                        : "bg-red-500/10 text-red-600 dark:text-red-400"
   621→                    )}
   622→                    title={
   623→                      weekDelta.fullWeek
   624→                        ? "Change over the last 7 days"
   625→                        : `Change over the ${weekDelta.spanDays < 1 ? "first day" : `last ${Math.max(1, Math.round(weekDelta.spanDays))} day(s)`} of tracked history`
   626→                    }
   627→                  >
   628→                    {weekDelta.delta >= 0 ? "▲" : "▼"} {Math.abs(weekDelta.delta).toFixed(2)} pts{" "}
   629→                    {weekDelta.fullWeek
   630→                      ? "/ 7d"
   631→                      : weekDelta.spanDays < 1
   632→                        ? "· day 1"
   633→                        : `/ ${Math.max(1, Math.round(weekDelta.spanDays))}d`}
   634→                  </span>
   635→                )}
   636→              </div>
   637→              <p className="text-sm text-muted-foreground tabular-nums">
   638→                <span className="font-semibold text-foreground">{overall.present}</span> present of{" "}
   639→                <span className="font-semibold text-foreground">{overall.total}</span> classes
   640→                <span className="text-muted-foreground/70"> ({overall.absent} missed)</span>
   641→              </p>
   642→              <div className="space-y-1 text-sm">
   643→                {overall.classesToTarget > 0 ? (

   651→                    <CheckCircle2 className="size-3.5 shrink-0" aria-hidden="true" />
   652→                    You are at or above the {target}% target
   653→                  </p>
   654→                )}
   655→                {overall.canSkip > 0 && (
   656→                  <p className="flex items-center justify-center gap-1.5 text-muted-foreground sm:justify-start">
   657→                    <CalendarClock className="size-3.5 shrink-0" aria-hidden="true" />
   658→                    You can skip <b className="tabular-nums text-foreground">{overall.canSkip}</b>{" "}
   659→                    {overall.canSkip === 1 ? "class" : "classes"} and stay above {target}%
   660→                  </p>
   661→                )}
   662→              </div>
   663→            </div>
   664→          </CardContent>
   665→        </Card>
   666→
   667→        <StatCard icon={Layers} label="Subjects" value={subjects.length} tone="emerald" />
   668→        <StatCard
   669→          icon={AlertTriangle}
   670→          label="Below target"
   671→          value={data.warningSubjects}
   672→          tone={data.warningSubjects > 0 ? "amber" : "default"}
   673→        />
   674→        <StatCard icon={BellRing} label="Active reminders" value={data.reminders.items.length} />
   675→      </motion.div>
   676→
   677→      {/* Weekly digest report dialog */}
   678→      <WeeklyDigestDialog open={digestOpen} onOpenChange={setDigestOpen} demoMode={mode === "demo"} />
   679→
   680→      {/* Row 2 — subject list */}
   681→      <Card className="rounded-xl border-border/60 transition-shadow hover:shadow-md">
   682→        <CardHeader className="flex flex-col gap-2 space-y-0 pb-3 sm:flex-row sm:items-center sm:justify-between">
   683→          <div className="min-w-0">
   684→            <CardTitle className="text-base tracking-tight">Subject attendance</CardTitle>
   685→            <CardDescription className="text-xs">Tap a subject for details and what-if planning</CardDescription>
   686→          </div>
   687→          <div className="flex items-center gap-1.5">
   688→            <Select value={sort} onValueChange={(v) => setSort(v as SubjectSort)}>
   689→              <SelectTrigger
   690→                className="h-8 w-full min-w-0 gap-1 text-xs sm:w-[190px]"
   691→                aria-label="Sort subjects"
   692→              >
   693→                <ArrowDownUp className="size-3.5 shrink-0 text-muted-foreground" aria-hidden="true" />
   694→                <SelectValue />
   695→              </SelectTrigger>
   696→              <SelectContent>
   697→                {SUBJECT_SORTS.map((o) => (
   698→                  <SelectItem key={o.key} value={o.key} className="text-xs">
   699→                    {o.label}
   700→                  </SelectItem>
   701→                ))}
   702→              </SelectContent>
   703→            </Select>
   704→            <Button variant="ghost" size="sm" className="h-8 shrink-0 gap-1 text-xs" onClick={onOpenAttendance}>
   705→              View all <ChevronRight className="size-3.5" aria-hidden="true" />
   706→            </Button>
   707→          </div>
   708→        </CardHeader>
   709→        <CardContent>
   710→          {subjects.length === 0 ? (

   712→              {data.semester !== "all"
   713→                ? `No subjects synchronized for Semester ${data.semester}`
   714→                : "No subjects synchronized yet"}
   715→            </p>

   717→            <div className="max-h-96 space-y-2 overflow-y-auto pr-1">
   718→              {sorted.map((s) => (
   719→                <SubjectRow
   720→                  key={s.id}
   721→                  s={s}
   722→                  onOpen={() => setSelectedCode(s.code)}
   723→                  shimmer={data.erp.sync.status === "syncing"}
   724→                />
   725→              ))}
   726→            </div>
   727→          )}
   728→        </CardContent>
   729→      </Card>
   730→
   731→      {/* Row 2.5 — today's schedule / weekly timetable */}
   732→      {subjects.length > 0 && <TimetableCard subjects={subjects} mode={data.mode} />}
   733→
   734→      {/* Row 3 — reminders preview + ERP status */}
   735→      <div className="grid gap-4 lg:grid-cols-2">
   736→        <RemindersPreview
   737→          items={data.reminders.items}
   738→          todayCount={data.reminders.todayCount}
   739→          weekCount={data.reminders.weekCount}
   740→          alaPending={data.reminders.alaPending}
   741→          alaTotal={data.reminders.alaTotal}
   742→          onOpenReminders={onOpenReminders}
   743→        />
   744→        <ErpStatusCard
   745→          connected={data.erp.connected}
   746→          demoMode={data.erp.demoMode ?? false}
   747→          username={data.erp.username}
   748→          sync={data.erp.sync}
   749→        />
   750→      </div>
   751→
   752→      {/* Row 4 — changes feed + notifications */}
   753→      <div className="grid gap-4 lg:grid-cols-2">
   754→        <Card className="rounded-xl border-border/60">
   755→          <CardHeader className="pb-3">
   756→            <CardTitle className="text-base tracking-tight">Recent ERP changes</CardTitle>
   757→            <CardDescription className="text-xs">
   758→              Detected automatically by the sync engine
   759→              {data.changes.some((c) => c.subjectCode && data.subjects.some((s) => s.code === c.subjectCode)) &&
   760→                " — tap a code to open the subject"}
   761→            </CardDescription>
   762→          </CardHeader>
   763→          <CardContent>
   764→            {data.changes.length === 0 ? (
   765→              <p className="py-6 text-center text-sm text-muted-foreground">No changes recorded yet</p>
   766→            ) : (
   767→              <>
   768→                <ul className="max-h-96 space-y-1.5 overflow-y-auto pr-1">
   769→                  {(showAllChanges ? data.changes : data.changes.slice(0, 8)).map((c) => (
   770→                    <ChangeRow
   771→                      key={c.id}
   772→                      c={c}
   773→                      /* Only rows whose subject still exists are clickable — a code that
   774→                         no longer matches a live subject would open nothing (honest affordance). */
   775→                      onOpen={
   776→                        c.subjectCode && data.subjects.some((s) => s.code === c.subjectCode)
   777→                          ? setSelectedCode
   778→                          : undefined
   779→                      }
   780→                    />
   781→                  ))}
   782→                </ul>
   783→                {data.changes.length > 8 && (
   784→                  <Button
   785→                    variant="ghost"
   786→                    size="sm"
   787→                    className="mt-2 h-7 w-full gap-1 text-xs text-muted-foreground hover:text-foreground"
   788→                    onClick={() => setShowAllChanges((v) => !v)}
   789→                    aria-expanded={showAllChanges}
   790→                  >
   791→                    {showAllChanges ? (
   792→                      <>Show less</>
   793→                    ) : (
   794→                      <>
   795→                        Show all {data.changes.length} events <ChevronRight className="size-3.5" aria-hidden="true" />
   796→                      </>
   797→                    )}
   798→                  </Button>
   799→                )}
   800→              </>
   801→            )}
   802→          </CardContent>
   803→        </Card>
   804→
   805→        <Card className="rounded-xl border-border/60">
   806→          <CardHeader className="pb-3">
   807→            <CardTitle className="text-base tracking-tight">Recent notifications</CardTitle>
   808→            <CardDescription className="text-xs">{data.unreadCount} unread</CardDescription>
   809→          </CardHeader>
   810→          <CardContent>
   811→            {data.notifications.length === 0 ? (
   812→              <p className="py-6 text-center text-sm text-muted-foreground">Nothing here yet</p>
   813→            ) : (
   814→              <>
   815→                <ul className="max-h-96 space-y-1 overflow-y-auto pr-1">
   816→                  {(showAllNotifs ? data.notifications : data.notifications.slice(0, 5)).map((n) => (
   817→                    <NotificationRow key={n.id} n={n} />
   818→                  ))}
   819→                </ul>
   820→                {data.notifications.length > 5 && (
   821→                  <Button
   822→                    variant="ghost"
   823→                    size="sm"
   824→                    className="mt-2 h-7 w-full gap-1 text-xs text-muted-foreground hover:text-foreground"
   825→                    onClick={() => setShowAllNotifs((v) => !v)}
   826→                    aria-expanded={showAllNotifs}
   827→                  >
   828→                    {showAllNotifs ? (
   829→                      <>Show less</>
   830→                    ) : (
   831→                      <>
   832→                        Show all {data.notifications.length} <ChevronRight className="size-3.5" aria-hidden="true" />
   833→                      </>
   834→                    )}
   835→                  </Button>
   836→                )}
   837→              </>
   838→            )}
   839→          </CardContent>
   840→        </Card>
   841→      </div>
   842→    </motion.div>
   843→  );
