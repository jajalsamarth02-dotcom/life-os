   104→export function CommandPalette({

   116→  const queryClient = useQueryClient();
   117→  const { resolvedTheme, setTheme } = useTheme();
   118→
   119→  /* r102/r103: the palette's live search string, fed by CommandInput's
   120→     onValueChange (NOT useCommandState — that hook reads the cmdk store

   125→  const [paletteSearch, setPaletteSearch] = React.useState("");

   127→    if (!open) setPaletteSearch("");
   128→  }, [open]);
   129→
   130→  const run = (fn: () => void) => {

   143→        queryClient.setQueryData<SubjectFocusHandoff>(qk.subjectFocus, {
   144→          code,
   145→          nonce: Date.now(),
   146→        });
   147→      }, 120);
   148→    });
   149→  };
   150→
   151→  /* Retry-story deep-link: write the handoff AFTER the dashboard tab is
   152→     active (run() already delays 60ms; the extra tick lets the tab's card
   153→     mount first). code=null = generic "open retry history". */
   154→  const openRetryStory = (code: string | null) => {

   156→      onNavigate("dashboard");
   157→      window.setTimeout(() => {
   158→        queryClient.setQueryData<RetryFocusHandoff>(qk.retryFocus, {
   159→          code,
   160→          nonce: Date.now(),
   161→        });
   162→      }, 120);
   163→    });
   164→  };
   165→
   166→  const newReminder = () => {
   167→    run(() => {
   168→      onNavigate("reminders");
   169→      window.setTimeout(() => {
   170→        window.dispatchEvent(new CustomEvent("erp:new-reminder"));
   171→      }, 120);
   172→    });
   173→  };
   174→
   175→  const toggleTheme = () => {
   176→    run(() => setTheme(resolvedTheme === "dark" ? "light" : "dark"));
   177→  };
   178→
   179→  const [textSize, setTextSizeState] = useTextSize();
   180→  const cycleText = () => {
   181→    const nextSize = cycleTextSize(textSize);
   182→    run(() => {
   183→      setTextSizeState(nextSize);
   184→      toast.success(
   185→        `Text size: ${TEXT_SIZES.find((s) => s.key === nextSize)?.label ?? nextSize}`,
   186→        { description: "Adjust any time in Settings → Text size." }

   188→    });
   189→  };

   232→      .filter((a) => !!a.externalId && prio.has(a.externalId))
   233→      .map((a) => ({
   234→        a,
   235→        priority: prio.get(a.externalId as string) ?? 0,
   236→        note: note.get(a.externalId as string) ?? "",
   237→      }))
   238→      .sort(
   239→        (x, y) =>
   240→          y.priority - x.priority ||
   241→          new Date(x.a.dueAt).getTime() - new Date(y.a.dueAt).getTime()

   243→  }, [assignmentsQ.data, alaNotesQ.data]);
   244→  /* r103: search-aware cap — a flagged ALA below the idle 3-row cap stays
   245→     reachable by name while typing (same rankedCap contract everywhere). */
   246→  const flaggedItems = React.useMemo(
   247→    () =>
   248→      rankedCap(
   249→        flaggedAll,
   250→        ({ a, note }) => `${a.title} ${a.subjectCode ?? ""} ${note} flagged priority local triage`,
   251→        3,
   252→        paletteSearch
   253→      ),
   254→    [flaggedAll, paletteSearch]
   255→  );
   256→  const nowMs = React.useMemo(() => Date.now(), [assignmentsQ.data]);

   259→    const sorted = [...active].sort(
   260→      (a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime()
   261→    );
   262→    const dueMs = (a: { dueAt: string }) => new Date(a.dueAt).getTime();
   263→    return {
   264→      sortedDeadlines: sorted,
   265→      activeDeadlineCount: active.length,
   266→      overdueCount: active.filter((a) => dueMs(a) < nowMs).length,
   267→      soonCount: active.filter((a) => {
   268→        const d = dueMs(a);
   269→        return d >= nowMs && d - nowMs <= 7 * 86_400_000;
   270→      }).length,
   271→    };
   272→  }, [assignmentsQ.data, nowMs]);
   273→  /* r103: search-aware cap — the headline fix this round. 41 active

   275→     week 6's ALA…) found NOTHING because cmdk never saw the row. While
   276→     typing, all of them are scored and the best 8 matches render. */
   277→  const deadlineItems = React.useMemo(
   278→    () =>
   279→      rankedCap(
   280→        sortedDeadlines,
   281→        (a) => `${a.title} ${a.subjectCode ?? ""} ala assignment deadline`,
   282→        8,
   283→        paletteSearch
   284→      ),
   285→    [sortedDeadlines, paletteSearch]
   286→  );
   287→  const openDeadline = (title: string) => {
   288→    run(() => {
   289→      onNavigate("reminders");
   290→      window.setTimeout(() => {
   291→        window.dispatchEvent(new CustomEvent("erp:focus-ala", { detail: title }));
   292→      }, 150);
   293→    });
   294→  };
   295→
   296→  /* r104 — "Your reminders" group: the r102 badge's second half finally has

   301→     Reminders tab with the Personal scope pre-applied (erp:filter-reminders
   302→     → setFilter — persisted like a manual toggle). */
   303→  const personalRemindersQ = useQuery({
   304→    queryKey: qk.reminders("personal", false),
   305→    queryFn: () => fetchReminders("personal", false),
   306→    staleTime: 30_000,
   307→  });
   308→  const allReminderRows = React.useMemo(
   309→    () => dueFirstReminderRows(personalRemindersQ.data ?? [], nowMs),
   310→    [personalRemindersQ.data, nowMs]
   311→  );
   312→  const reminderItems = React.useMemo(
   313→    () =>
   314→      rankedCap(
   315→        allReminderRows,
   316→        (r) => `${r.title} ${r.subjectCode ?? ""} personal reminder due reminder`,
   317→        8,
   318→        paletteSearch
   319→      ),
   320→    [allReminderRows, paletteSearch]
   321→  );
   322→  const dueReminderCount = React.useMemo(
   323→    () => duePersonalCount(personalRemindersQ.data ?? [], nowMs),
   324→    [personalRemindersQ.data, nowMs]
   325→  );
   326→  const openReminders = () => {
   327→    run(() => {
   328→      onNavigate("reminders");
   329→      window.setTimeout(() => {
   330→        window.dispatchEvent(new CustomEvent("erp:filter-reminders", { detail: "personal" }));
   331→      }, 150);
   332→    });
   333→  };
   334→
   335→  /* ALA quick-filter — a fresh triage entry: the AlaSection listener clears
   336→     every other axis (search/subject/day) and applies the requested status. */
   337→  const filterAla = (status: string) => {
   338→    run(() => {

   340→      window.setTimeout(() => {
   341→        window.dispatchEvent(new CustomEvent("erp:filter-ala", { detail: status }));

   433→  const NAV = [
   434→    { tab: "dashboard" as const, label: "Dashboard", icon: LayoutDashboard, kbd: "1" },

   438→  ];
   439→
   440→  return (

   454→            </p>
   455→          </div>
   456→        </CommandEmpty>
   457→
   458→        {subjects && subjects.length > 0 && (
   459→          <>
   460→            <CommandGroup heading="Subjects">
   461→              {subjects.map((s) => (
   462→                <CommandItem
   463→                  key={s.code}
   464→                  value={`${s.code} ${s.name} ${s.percentage.toFixed(1)}% target`}
   465→                  onSelect={() => openSubject(s.code)}

   467→                  <span
   468→                    className={cn("size-2 shrink-0 rounded-full", STATUS_DOT[s.status] ?? "bg-muted-foreground")}
   469→                    aria-hidden="true"
   470→                  />
   471→                  <span className="font-mono text-xs font-semibold">{s.code}</span>
   472→                  <span className="truncate text-sm text-muted-foreground">{s.name}</span>
   473→                  <span className="ml-auto flex items-center gap-1.5 whitespace-nowrap text-xs tabular-nums text-muted-foreground">
   474→                    <span className={cn("font-semibold", s.status === "danger" && "text-red-600 dark:text-red-400", s.status === "warning" && "text-amber-600 dark:text-amber-400", s.status === "safe" && "text-emerald-600 dark:text-emerald-400")}>
   475→                      {s.percentage.toFixed(1)}%
   476→                    </span>
   477→                    <span aria-hidden="true">·</span>
   478→                    <Target className="size-3" aria-hidden="true" />
   479→                    {s.targetPct}%
   480→                    {s.hasCustomTarget && (
   481→                      <span className="rounded-full border border-amber-500/40 bg-amber-500/10 px-1.5 text-[10px] font-semibold text-amber-700 dark:text-amber-400">
   482→                        custom
   483→                      </span>
   484→                    )}
   485→                  </span>
   486→                </CommandItem>
   487→              ))}
   488→            </CommandGroup>
   489→            <CommandSeparator />
   490→          </>
   491→        )}
   492→
   493→        {classEntries.length > 0 && (
   494→          <>
   495→            <CommandGroup
   496→              heading={`Add a class to your calendar${
   497→                classTotal > classEntries.length
   498→                  ? ` · next ${classEntries.length} of ${classTotal}`
   499→                  : ""
   500→              }`}
   501→            >
   502→              {classEntries.map((slot) => {
   503→                const uid = `${slot.dayOfWeek}-${slot.startMin}-${slot.endMin}-${slot.subjectCode}`;
   504→                const saved = savedClassUid === uid;
   505→                const name = subjectName.get(slot.subjectCode) ?? null;
   506→                return (
   507→                  <CommandItem
   508→                    key={uid}
   509→                    value={classRowValue(slot, name)}
   510→                    onSelect={() => downloadClass(slot)}
   511→                  >
   512→                    {saved ? (
   513→                      <Check className="size-4 shrink-0 text-emerald-500" aria-hidden="true" />
   514→                    ) : (
   515→                      <CalendarPlus className="size-4 shrink-0 text-violet-500" aria-hidden="true" />
   516→                    )}
   517→                    <span className="font-mono text-xs font-semibold">{slot.subjectCode}</span>
   518→                    <span className="truncate text-sm text-muted-foreground">
   519→                      {name ?? "Class"}
   520→                    </span>
