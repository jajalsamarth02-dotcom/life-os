    65→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    66→  const copy = [...list];
    67→  switch (sort) {
    68→    case "lowest":
    69→      return copy.sort((a, b) => a.percentage - b.percentage);
    70→    case "highest":
    71→      return copy.sort((a, b) => b.percentage - a.percentage);
    72→    case "name":
    73→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    74→    case "recent":
    75→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    76→  }
    77→}

    79→/** Live countdown label for the next upcoming deadline. */
    80→function countdownLabel(dueAt: string, now: number): string {
    81→  const diff = new Date(dueAt).getTime() - now;
    82→  const abs = Math.abs(diff);
    83→  const mins = Math.round(abs / 60_000);
    84→  const hours = Math.round(abs / 3_600_000);
    85→  const days = Math.round(abs / 86_400_000);
    86→  let span: string;
    87→  if (mins < 60) span = `${mins} min`;
    88→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    89→  else span = `${days} day${days === 1 ? "" : "s"}`;
    90→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    91→}

    93→function NextUpStrip({ items }: { items: Reminder[] }) {
    94→  const now = useNow();
    95→  const upcoming = items
    96→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
    97→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
    98→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
    99→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   100→
   101→  if (upcoming.length === 0) return null;
   102→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   103→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   104→
   105→  return (
   106→    <div
   107→      className={cn(
   108→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   109→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   110→      )}
   111→    >
   112→      <span
   113→        className={cn(
   114→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   115→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   116→        )}
   117→      >
   118→        <Hourglass className="size-4" aria-hidden="true" />
   119→      </span>
   120→      <div className="min-w-0 flex-1">
   121→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   122→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   123→        </p>
   124→        <p className="truncate text-sm font-medium">{shown.title}</p>
   125→      </div>
   126→      <div className="shrink-0 text-right">
   127→        <p
   128→          className={cn(
   129→            "text-sm font-bold tabular-nums",
   130→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"

   132→        >
   133→          {countdownLabel(shown.dueAt, now)}

   435→export function DashboardTab({
   436→  onOpenAttendance,
   437→  onOpenSettings,
   438→  onOpenReminders,
   439→}: {
   440→  onOpenAttendance: () => void;
   441→  onOpenSettings: () => void;
   442→  onOpenReminders: () => void;
   443→}) {
   444→  const [sort, setSort] = useState<SubjectSort>("lowest");
   445→  /* Store the CODE (not the object) and derive the view from fresh query data,
   446→     so the drawer reflects updated targets/percentages after refetches. */
   447→  const [selectedCode, setSelectedCode] = useState<string | null>(null);
   448→  const [digestOpen, setDigestOpen] = useState(false);
   449→  const { data, isPending, isError, error } = useQuery({
   450→    queryKey: qk.dashboard,
   451→    queryFn: fetchDashboard,
   452→    refetchInterval: 20_000,
   453→  });
   454→  const history = useQuery({
   455→    queryKey: qk.history,
   456→    queryFn: fetchHistory,
   457→    refetchInterval: 60_000,
   458→  });
   459→
   460→  const weekDelta = useMemo(() => {
   461→    const pts = history.data?.overall ?? [];
   462→    if (pts.length < 2) return null;
   463→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   464→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   465→    const last = pts[pts.length - 1];
   466→    const d = last.percentage - first.percentage;
   467→    if (Math.abs(d) < 0.005) return null;
   468→    // Be honest about the window: when tracked history is younger than the
   469→    // 7-day span (e.g. live tracking just started), label the real span.
   470→    const spanDays = Math.max(
   471→      0,
   472→      (new Date(last.t).getTime() - new Date(first.t).getTime()) / 86_400_000
   473→    );
   474→    return { delta: d, spanDays, fullWeek: spanDays >= 6.5 };
   475→  }, [history.data]);
   476→
   477→  if (isPending) {
   478→    return (
   479→      <div className="space-y-4">
   480→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   481→          <CardSkeletonLight className="md:col-span-2 xl:col-span-2" />
   482→          <CardSkeletonLight />
   483→          <CardSkeletonLight />
   484→          <CardSkeletonLight />
   485→        </div>
   486→        <CardSkeletonLight />
   487→        <div className="grid gap-4 lg:grid-cols-2">
   488→          <CardSkeletonLight />
   489→          <CardSkeletonLight />
   490→        </div>
   491→      </div>
   492→    );
   493→  }
   494→
   495→  if (isError || !data) {
   496→    return (
   497→      <Card className="rounded-xl border-red-500/30 bg-red-500/5">
   498→        <CardContent className="flex items-center gap-3 p-6 text-sm text-red-600 dark:text-red-400">
   499→          <XCircle className="size-5 shrink-0" aria-hidden="true" />
   500→          Could not load the dashboard: {error?.message ?? "unknown error"}
   501→        </CardContent>
   502→      </Card>
   503→    );
   504→  }
   505→
   506→  const { overall, subjects, target, mode } = data;
   507→  const sorted = sortSubjects(subjects, sort);
   508→
   509→  if (mode === "none") {
   510→    return (
   511→      <EmptyHero
   512→        title="No ERP data available yet."
   513→        description="Connect your ERP account or run a first synchronization to fetch your attendance and reminders. Nothing is shown until real data arrives."
   514→        actionLabel="Connect / Sync ERP"
   515→        onAction={onOpenSettings}
   516→      />
   517→    );
   518→  }
   519→
   520→  return (
   521→    <motion.div
   522→      className="space-y-4"
   523→      initial="hidden"
   524→      animate="show"
   525→      variants={{
   526→        hidden: {},
   527→        show: { transition: { staggerChildren: 0.05 } },
   528→      }}
   529→    >
   530→      {/* Subject detail drawer (also covers what-if planning) */}
   531→      <SubjectDrawer
   532→        subject={data.subjects.find((s) => s.code === selectedCode) ?? null}
   533→        target={target}
   534→        globalTarget={target}
   535→        open={selectedCode !== null}
   536→        onOpenChange={(o) => {
   537→          if (!o) setSelectedCode(null);
   538→        }}
   539→      />
   540→
   541→      {/* Row 0 — next deadline countdown */}
   542→      <NextUpStrip items={data.reminders.items} />
   543→
   544→      {/* Row 1 — overall + stats */}
   545→      <motion.div
   546→        className="grid gap-4 md:grid-cols-2 xl:grid-cols-5"
   547→        variants={{ hidden: { opacity: 0, y: 10 }, show: { opacity: 1, y: 0, transition: { duration: 0.25, ease: "easeOut" } } }}
   548→      >
   549→        <Card className="erp-card-accent rounded-xl border-border/60 transition-shadow hover:shadow-md md:col-span-2 xl:col-span-2">
   550→          <CardContent className="flex flex-col items-center gap-4 p-6 sm:flex-row sm:gap-6">
   551→            <ProgressRing percentage={overall.percentage} target={target} size={150} animated />
   552→            <div className="min-w-0 flex-1 space-y-2.5 text-center sm:text-left">
   553→              <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   554→                <h2 className="text-lg font-semibold tracking-tight">Overall attendance</h2>
   555→                {data.semester !== "all" && (
   556→                  <Badge
   557→                    variant="outline"
   558→                    className="gap-1 border-primary/40 text-primary"
   559→                    title="A semester partition is active — only this semester's ERP subjects are counted"
   560→                  >
   561→                    <GraduationCap className="size-3" aria-hidden="true" />
   562→                    Sem {data.semester}
   563→                  </Badge>
   564→                )}
   565→                <StatusChip status={overall.status} target={target} />
   566→                <Button
   567→                  variant="ghost"
   568→                  size="sm"
   569→                  className="ml-auto h-7 gap-1 rounded-full px-2.5 text-xs text-muted-foreground hover:text-foreground"
   570→                  onClick={() => setDigestOpen(true)}
   571→                  aria-label="Open the weekly attendance report"
   572→                >
   573→                  <CalendarRange className="size-3.5" aria-hidden="true" />
   574→                  This week
   575→                </Button>
   576→                {weekDelta !== null && (
   577→                  <span
   578→                    className={cn(
   579→                      "rounded-full px-2 py-0.5 text-[11px] font-semibold tabular-nums",
   580→                      weekDelta.delta >= 0
   581→                        ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   582→                        : "bg-red-500/10 text-red-600 dark:text-red-400"
   583→                    )}
   584→                    title={
   585→                      weekDelta.fullWeek
   586→                        ? "Change over the last 7 days"
   587→                        : `Change over the ${weekDelta.spanDays < 1 ? "first day" : `last ${Math.max(1, Math.round(weekDelta.spanDays))} day(s)`} of tracked history`
   588→                    }
   589→                  >
   590→                    {weekDelta.delta >= 0 ? "▲" : "▼"} {Math.abs(weekDelta.delta).toFixed(2)} pts{" "}
   591→                    {weekDelta.fullWeek
   592→                      ? "/ 7d"
   593→                      : weekDelta.spanDays < 1
   594→                        ? "· day 1"
   595→                        : `/ ${Math.max(1, Math.round(weekDelta.spanDays))}d`}
   596→                  </span>
   597→                )}
   598→              </div>
   599→              <p className="text-sm text-muted-foreground tabular-nums">
   600→                <span className="font-semibold text-foreground">{overall.present}</span> present of{" "}
   601→                <span className="font-semibold text-foreground">{overall.total}</span> classes
   602→                <span className="text-muted-foreground/70"> ({overall.absent} missed)</span>
   603→              </p>
   604→              <div className="space-y-1 text-sm">
   605→                {overall.classesToTarget > 0 ? (
   606→                  <p className="flex items-center justify-center gap-1.5 text-amber-600 dark:text-amber-400 sm:justify-start">
   607→                    <ArrowRight className="size-3.5 shrink-0" aria-hidden="true" />
   608→                    Attend the next <b className="tabular-nums">{overall.classesToTarget}</b>{" "}
   609→                    {overall.classesToTarget === 1 ? "class" : "classes"} to reach {target}%
   610→                  </p>
   611→                ) : (
   612→                  <p className="flex items-center justify-center gap-1.5 text-emerald-600 dark:text-emerald-400 sm:justify-start">
   613→                    <CheckCircle2 className="size-3.5 shrink-0" aria-hidden="true" />
   614→                    You are at or above the {target}% target
   615→                  </p>
   616→                )}
   617→                {overall.canSkip > 0 && (
   618→                  <p className="flex items-center justify-center gap-1.5 text-muted-foreground sm:justify-start">
   619→                    <CalendarClock className="size-3.5 shrink-0" aria-hidden="true" />
   620→                    You can skip <b className="tabular-nums text-foreground">{overall.canSkip}</b>{" "}
   621→                    {overall.canSkip === 1 ? "class" : "classes"} and stay above {target}%
   622→                  </p>
   623→                )}
   624→              </div>
   625→            </div>
   626→          </CardContent>
   627→        </Card>
   628→
   629→        <StatCard icon={Layers} label="Subjects" value={subjects.length} tone="emerald" />
   630→        <StatCard
   631→          icon={AlertTriangle}
   632→          label="Below target"
   633→          value={data.warningSubjects}
   634→          tone={data.warningSubjects > 0 ? "amber" : "default"}
   635→        />
   636→        <StatCard icon={BellRing} label="Active reminders" value={data.reminders.items.length} />
   637→      </motion.div>
   638→
   639→      {/* Weekly digest report dialog */}
   640→      <WeeklyDigestDialog open={digestOpen} onOpenChange={setDigestOpen} demoMode={mode === "demo"} />
   641→
   642→      {/* Row 2 — subject list */}
   643→      <Card className="rounded-xl border-border/60 transition-shadow hover:shadow-md">
   644→        <CardHeader className="flex flex-col gap-2 space-y-0 pb-3 sm:flex-row sm:items-center sm:justify-between">
   645→          <div className="min-w-0">
   646→            <CardTitle className="text-base tracking-tight">Subject attendance</CardTitle>
   647→            <CardDescription className="text-xs">Tap a subject for details and what-if planning</CardDescription>
   648→          </div>
   649→          <div className="flex items-center gap-1.5">
   650→            <Select value={sort} onValueChange={(v) => setSort(v as SubjectSort)}>
   651→              <SelectTrigger
   652→                className="h-8 w-full min-w-0 gap-1 text-xs sm:w-[190px]"
   653→                aria-label="Sort subjects"
   654→              >
   655→                <ArrowDownUp className="size-3.5 shrink-0 text-muted-foreground" aria-hidden="true" />
   656→                <SelectValue />
   657→              </SelectTrigger>
   658→              <SelectContent>
   659→                {SUBJECT_SORTS.map((o) => (
   660→                  <SelectItem key={o.key} value={o.key} className="text-xs">
   661→                    {o.label}
   662→                  </SelectItem>
   663→                ))}
   664→              </SelectContent>
   665→            </Select>
   666→            <Button variant="ghost" size="sm" className="h-8 shrink-0 gap-1 text-xs" onClick={onOpenAttendance}>
   667→              View all <ChevronRight className="size-3.5" aria-hidden="true" />
   668→            </Button>
   669→          </div>
   670→        </CardHeader>
   671→        <CardContent>
   672→          {subjects.length === 0 ? (
   673→            <p className="py-6 text-center text-sm text-muted-foreground">
   674→              {data.semester !== "all"
   675→                ? `No subjects synchronized for Semester ${data.semester}`
   676→                : "No subjects synchronized yet"}
   677→            </p>
   678→          ) : (
   679→            <div className="max-h-96 space-y-2 overflow-y-auto pr-1">
   680→              {sorted.map((s) => (
   681→                <SubjectRow
   682→                  key={s.id}
   683→                  s={s}
   684→                  onOpen={() => setSelectedCode(s.code)}
   685→                  shimmer={data.erp.sync.status === "syncing"}
   686→                />
   687→              ))}
   688→            </div>
   689→          )}
   690→        </CardContent>
   691→      </Card>
   692→
   693→      {/* Row 2.5 — today's schedule / weekly timetable */}
   694→      {subjects.length > 0 && <TimetableCard subjects={subjects} mode={data.mode} />}
   695→
   696→      {/* Row 3 — reminders preview + ERP status */}
   697→      <div className="grid gap-4 lg:grid-cols-2">
   698→        <RemindersPreview
   699→          items={data.reminders.items}
   700→          todayCount={data.reminders.todayCount}
   701→          weekCount={data.reminders.weekCount}
   702→          alaPending={data.reminders.alaPending}
   703→          alaTotal={data.reminders.alaTotal}
   704→          onOpenReminders={onOpenReminders}
   705→        />
   706→        <ErpStatusCard
   707→          connected={data.erp.connected}
   708→          demoMode={data.erp.demoMode ?? false}
   709→          username={data.erp.username}
   710→          sync={data.erp.sync}
   711→        />
   712→      </div>
   713→
   714→      {/* Row 4 — changes feed + notifications */}
   715→      <div className="grid gap-4 lg:grid-cols-2">
   716→        <Card className="rounded-xl border-border/60">
   717→          <CardHeader className="pb-3">
   718→            <CardTitle className="text-base tracking-tight">Recent ERP changes</CardTitle>
   719→            <CardDescription className="text-xs">Detected automatically by the sync engine</CardDescription>
   720→          </CardHeader>
   721→          <CardContent>
   722→            {data.changes.length === 0 ? (
   723→              <p className="py-6 text-center text-sm text-muted-foreground">No changes recorded yet</p>
   724→            ) : (
   725→              <ul className="max-h-96 space-y-1.5 overflow-y-auto pr-1">
   726→                {data.changes.slice(0, 8).map((c) => (
   727→                  <ChangeRow key={c.id} c={c} />
   728→                ))}
   729→              </ul>
   730→            )}
   731→          </CardContent>
   732→        </Card>
   733→
   734→        <Card className="rounded-xl border-border/60">
   735→          <CardHeader className="pb-3">
   736→            <CardTitle className="text-base tracking-tight">Recent notifications</CardTitle>
   737→            <CardDescription className="text-xs">{data.unreadCount} unread</CardDescription>
   738→          </CardHeader>
   739→          <CardContent>
   740→            {data.notifications.length === 0 ? (
   741→              <p className="py-6 text-center text-sm text-muted-foreground">Nothing here yet</p>
   742→            ) : (
   743→              <ul className="max-h-96 space-y-1 overflow-y-auto pr-1">
   744→                {data.notifications.slice(0, 5).map((n) => (
   745→                  <NotificationRow key={n.id} n={n} />
   746→                ))}
   747→              </ul>
   748→            )}
   749→          </CardContent>
   750→        </Card>
   751→      </div>
   752→    </motion.div>
   753→  );
