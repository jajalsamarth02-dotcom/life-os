    70→const SUBJECT_SORTS = [
    71→  { key: "lowest", label: "Lowest attendance first" },
    72→  { key: "highest", label: "Highest attendance first" },
    73→  { key: "name", label: "Name A–Z" },

    77→type SubjectSort = (typeof SUBJECT_SORTS)[number]["key"];
    78→
    79→/* No-data subjects (total === 0) have no attendance yet — a 0.00% is not a
    80→   real value, so they never compete with actual percentages: they sink to
    81→   the end of both directions with a name tie-break. */
    82→function noDataLast(a: SubjectView, b: SubjectView): number {
    83→  const an = a.total === 0;
    84→  const bn = b.total === 0;
    85→  if (an !== bn) return an ? 1 : -1;
    86→  return 0;
    87→}

    89→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    90→  const copy = [...list];
    91→  switch (sort) {
    92→    case "lowest":
    93→      return copy.sort(
    94→        (a, b) => noDataLast(a, b) || a.percentage - b.percentage || a.name.localeCompare(b.name)
    95→      );
    96→    case "highest":
    97→      return copy.sort(
    98→        (a, b) => noDataLast(a, b) || b.percentage - a.percentage || a.name.localeCompare(b.name)
    99→      );
   100→    case "name":
   101→      return copy.sort((a, b) => a.name.localeCompare(b.name));
   102→    case "recent":
   103→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
   104→  }
   105→}

   107→/** Live countdown label for the next upcoming deadline. */
   108→function countdownLabel(dueAt: string, now: number): string {
   109→  const diff = new Date(dueAt).getTime() - now;
   110→  const abs = Math.abs(diff);
   111→  const mins = Math.round(abs / 60_000);
   112→  const hours = Math.round(abs / 3_600_000);
   113→  const days = Math.round(abs / 86_400_000);
   114→  let span: string;
   115→  if (mins < 60) span = `${mins} min`;
   116→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
   117→  else span = `${days} day${days === 1 ? "" : "s"}`;
   118→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
   119→}

   121→/** Relative urgency pill for an ALA deadline: red overdue/≤48h, amber ≤7d, muted later. */
   122→function deadlinePill(dueAt: string, now: number): { label: string; className: string } {
   123→  const due = new Date(dueAt).getTime();
   124→  const diff = due - now;
   125→  if (diff < 0)
   126→    return {
   127→      label: countdownLabel(dueAt, now),
   128→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   129→    };
   130→  if (diff <= 2 * 86_400_000)
   131→    return {
   132→      label: countdownLabel(dueAt, now),
   133→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   134→    };
   135→  if (diff <= 7 * 86_400_000)
   136→    return {
   137→      label: countdownLabel(dueAt, now),
   138→      className: "bg-amber-500/10 text-amber-700 dark:text-amber-400",
   139→    };
   140→  return { label: countdownLabel(dueAt, now), className: "bg-muted text-muted-foreground" };
   141→}

   449→/** Live "in 12 min"-style countdown for a future instant (null when absent/past). */
   450→function inCountdown(iso: string | null, now: number): string | null {
   451→  if (!iso) return null;
   452→  const diff = new Date(iso).getTime() - now;
   453→  if (diff <= 0) return null;
   454→  const mins = Math.max(1, Math.round(diff / 60_000));
   455→  if (mins < 60) return `in ${mins} min`;
   456→  const hrs = Math.floor(mins / 60);
   457→  const rem = mins % 60;
   458→  return rem > 0 ? `in ${hrs}h ${rem}m` : `in ${hrs}h`;
   459→}
