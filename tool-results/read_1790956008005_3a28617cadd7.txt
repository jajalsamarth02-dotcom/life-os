    63→const SUBJECT_SORTS = [
    64→  { key: "lowest", label: "Lowest attendance first" },

    72→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    73→  const copy = [...list];
    74→  switch (sort) {
    75→    case "lowest":
    76→      return copy.sort((a, b) => a.percentage - b.percentage);
    77→    case "highest":
    78→      return copy.sort((a, b) => b.percentage - a.percentage);
    79→    case "name":
    80→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    81→    case "recent":
    82→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    83→  }
    84→}

    86→/** Live countdown label for the next upcoming deadline. */
    87→function countdownLabel(dueAt: string, now: number): string {
    88→  const diff = new Date(dueAt).getTime() - now;
    89→  const abs = Math.abs(diff);
    90→  const mins = Math.round(abs / 60_000);
    91→  const hours = Math.round(abs / 3_600_000);
    92→  const days = Math.round(abs / 86_400_000);
    93→  let span: string;
    94→  if (mins < 60) span = `${mins} min`;
    95→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    96→  else span = `${days} day${days === 1 ? "" : "s"}`;
    97→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    98→}

   100→/** Relative urgency pill for an ALA deadline: red overdue/≤48h, amber ≤7d, muted later. */
   101→function deadlinePill(dueAt: string, now: number): { label: string; className: string } {
   102→  const due = new Date(dueAt).getTime();
   103→  const diff = due - now;
   104→  if (diff < 0)
   105→    return {
   106→      label: countdownLabel(dueAt, now),
   107→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   108→    };
   109→  if (diff <= 2 * 86_400_000)
   110→    return {
   111→      label: countdownLabel(dueAt, now),
   112→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   113→    };
   114→  if (diff <= 7 * 86_400_000)
   115→    return {
   116→      label: countdownLabel(dueAt, now),
   117→      className: "bg-amber-500/10 text-amber-700 dark:text-amber-400",
   118→    };
   119→  return { label: countdownLabel(dueAt, now), className: "bg-muted text-muted-foreground" };
   120→}

   226→function NextUpStrip({ items }: { items: Reminder[] }) {
   227→  const now = useNow();
   228→  const upcoming = items
   229→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
   230→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
   231→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
   232→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   233→
   234→  if (upcoming.length === 0) return null;
   235→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   236→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   237→
   238→  return (
   239→    <div
   240→      className={cn(
   241→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   242→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   243→      )}
   244→    >
   245→      <span
   246→        className={cn(
   247→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   248→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   249→        )}
   250→      >
   251→        <Hourglass className="size-4" aria-hidden="true" />
   252→      </span>
   253→      <div className="min-w-0 flex-1">
   254→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   255→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}

   276→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"

   418→/** Live "in 12 min"-style countdown for a future instant (null when absent/past). */
   419→function inCountdown(iso: string | null, now: number): string | null {
   420→  if (!iso) return null;
   421→  const diff = new Date(iso).getTime() - now;
   422→  if (diff <= 0) return null;
   423→  const mins = Math.max(1, Math.round(diff / 60_000));
   424→  if (mins < 60) return `in ${mins} min`;
   425→  const hrs = Math.floor(mins / 60);
   426→  const rem = mins % 60;
   427→  return rem > 0 ? `in ${hrs}h ${rem}m` : `in ${hrs}h`;
   428→}

   627→      <span className={cn("mt-0.5 flex size-6 shrink-0 items-center justify-center rounded-md", cls)}>

   662→  onOpenSettings: () => void;
   663→  onOpenReminders: () => void;
   664→}) {
   665→  const [sort, setSort] = useState<SubjectSort>("lowest");
   666→  /* Store the CODE (not the object) and derive the view from fresh query data,
   667→     so the drawer reflects updated targets/percentages after refetches. */
   668→  const [selectedCode, setSelectedCode] = useState<string | null>(null);
   669→  const [digestOpen, setDigestOpen] = useState(false);
   670→  const [showAllChanges, setShowAllChanges] = useState(false);
   671→  const [showAllNotifs, setShowAllNotifs] = useState(false);
   672→  const { data, isPending, isError, error } = useQuery({
   673→    queryKey: qk.dashboard,
   674→    queryFn: fetchDashboard,
   675→    refetchInterval: 20_000,
   676→  });
   677→  const history = useQuery({
   678→    queryKey: qk.history,
   679→    queryFn: fetchHistory,
   680→    refetchInterval: 60_000,
   681→  });
   682→
   683→  const weekDelta = useMemo(() => {
   684→    const pts = history.data?.overall ?? [];
   685→    if (pts.length < 2) return null;
   686→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   687→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   688→    const last = pts[pts.length - 1];
   689→    const d = last.percentage - first.percentage;
   690→    if (Math.abs(d) < 0.005) return null;
   691→    // Be honest about the window: when tracked history is younger than the
   692→    // 7-day span (e.g. live tracking just started), label the real span.
   693→    const spanDays = Math.max(
   694→      0,
   695→      (new Date(last.t).getTime() - new Date(first.t).getTime()) / 86_400_000
   696→    );
   697→    return { delta: d, spanDays, fullWeek: spanDays >= 6.5 };
   698→  }, [history.data]);
   699→
   700→  const [copied, setCopied] = useState(false);
   701→  const copyTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   702→  const summaryText = useMemo(() => {
   703→    if (!data) return "";
   704→    const lines: string[] = [];
   705→    lines.push(
   706→      `Attendance summary — ${new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata", dateStyle: "medium", timeStyle: "short" })} IST`

   709→      `Overall: ${data.overall.percentage.toFixed(2)}% (${data.overall.present}/${data.overall.total} classes, ${data.overall.absent} missed) — target ${data.target}% [${data.overall.status}]`
   710→    );

   712→      lines.push(
   713→        `${s.code} ${s.name}: ${s.percentage.toFixed(2)}% (${s.present}/${s.total}) — target ${s.targetPct}% [${s.status}]${s.hasCustomTarget ? " (custom)" : ""}`
   714→      );
   715→    }
   716→    return lines.join("\n");
   717→  }, [data]);
   718→
   719→  const copySummary = async () => {

   740→    if (!summaryText) return;
   741→    try {
   742→      await navigator.share({ title: "Attendance summary", text: summaryText });
   743→      setShared(true);
   744→      if (shareTimer.current) clearTimeout(shareTimer.current);
   745→      shareTimer.current = setTimeout(() => setShared(false), 2000);

   747→      // share sheet dismissed (AbortError) or unavailable — stay inert

   749→  };

   751→  if (isPending) {

   753→      <div className="space-y-4">
   754→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   755→          <CardSkeletonLight className="md:col-span-2 xl:col-span-2" />
   756→          <CardSkeletonLight />
   757→          <CardSkeletonLight />
   758→          <CardSkeletonLight />

   763→          <CardSkeletonLight />
   764→        </div>
   765→      </div>

   767→  }
   768→
   769→  if (isError || !data) {
