    47→const DAY_NAMES = ["Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday"];

    50→const CHIP_COLORS = ["#10b981", "#f59e0b", "#84cc16", "#f97316", "#14b8a6", "#a8a29e", "#ef4444"];
    51→
    52→function colorForCode(code: string, subjects: SubjectView[]): string {
    53→  const idx = subjects.findIndex((s) => s.code === code);
    54→  return CHIP_COLORS[(idx < 0 ? 0 : idx) % CHIP_COLORS.length];
    55→}
    56→
    57→function hhmm(min: number): string {
    58→  return `${String(Math.floor(min / 60)).padStart(2, "0")}:${String(min % 60).padStart(2, "0")}`;
    59→}
    60→
    61→/** Live status label for a slot relative to now. */
    62→function slotStatus(
    63→  slot: TimetableSlotRow,
    64→  nowDay: number,
    65→  nowMin: number
    66→): { key: "ongoing" | "upcoming" | "done" | "other"; label?: string } {
    67→  if (slot.dayOfWeek !== nowDay) return { key: "other" };
    68→  if (nowMin >= slot.startMin && nowMin < slot.endMin) {
    69→    return { key: "ongoing", label: `ends in ${slot.endMin - nowMin} min` };
    70→  }
    71→  if (nowMin < slot.startMin) {
    72→    return { key: "upcoming", label: `starts in ${slot.startMin - nowMin} min` };
    73→  }
    74→  return { key: "done" };
    75→}

    79→function SlotChip({
    80→  slot,
    81→  color,
    82→  nowDay,
    83→  nowMin,
    84→  compact = false,
    85→}: {
    86→  slot: TimetableSlotRow;
    87→  color: string;
    88→  nowDay: number;
    89→  nowMin: number;
    90→  compact?: boolean;
    91→}) {
    92→  const st = slotStatus(slot, nowDay, nowMin);
    93→  const ongoing = st.key === "ongoing";
    94→  return (
    95→    <div
    96→      className={cn(
    97→        "group rounded-lg border px-2 py-1.5 text-left transition-colors",
    98→        ongoing
    99→          ? "border-emerald-500/50 bg-emerald-500/10"
   100→          : st.key === "done"
   101→            ? "border-border/40 bg-muted/30 opacity-55"
   102→            : "border-border/60 bg-card hover:bg-muted/40"
   103→      )}
   104→    >
   105→      <div className="flex items-center gap-1.5">
   106→        <span className="size-2 shrink-0 rounded-full" style={{ backgroundColor: color }} aria-hidden="true" />
   107→        <span className="min-w-0 flex-1 truncate font-mono text-[11px] font-semibold">{slot.subjectCode}</span>
   108→        {ongoing && <CircleDot className="size-3 shrink-0 animate-pulse text-emerald-500" aria-hidden="true" />}
   109→      </div>
   110→      <p className="mt-0.5 text-[11px] tabular-nums text-muted-foreground">
   111→        {hhmm(slot.startMin)}–{hhmm(slot.endMin)}
   112→      </p>
   113→      {!compact && st.label && (
   114→        <p className={cn("mt-0.5 text-[10px] font-medium", ongoing ? "text-emerald-600 dark:text-emerald-400" : "text-muted-foreground")}>
   115→          {st.label}
   116→        </p>
   117→      )}
   118→    </div>
   119→  );
   120→}

   360→            </div>
   361→          )}
   362→        </div>
   363→      </DialogContent>
   364→    </Dialog>

   366→}

   370→export function TimetableCard({

   378→  const nowDay = new Date(now).getDay();
   379→  const nowMin = new Date(now).getHours() * 60 + new Date(now).getMinutes();
   380→  const [weekOpen, setWeekOpen] = useState(false);
   381→
   382→  const { data, isPending, isError } = useQuery({
   383→    queryKey: qk.timetable,
   384→    queryFn: fetchTimetable,
   385→    refetchInterval: 60_000,
   386→    retry: 2,
   387→  });

   389→  const slots = data ?? [];

   391→    () => slots.filter((s) => s.dayOfWeek === nowDay),

   402→            <CalendarDays className="size-4 text-emerald-500" aria-hidden="true" />
   403→            Today&apos;s schedule

   413→          size="icon"

   416→          onClick={() => {
   417→            const filename = downloadTimetableIcs(slots, subjects);
   418→            toast.success(`${filename} downloaded`, {
   419→              description: "Classes repeat weekly — import into your calendar app.",
   420→            });
   421→          }}
   422→          aria-label="Export the weekly timetable as a calendar (.ics) file"
   423→        >
   424→          <Download className="size-4" aria-hidden="true" />
   425→        </Button>

   428→        {isPending ? (

   430→            {Array.from({ length: 3 }).map((_, i) => (

   435→          <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed border-red-500/40 py-8 text-center">
   436→            <CalendarDays className="size-6 text-red-500/60" aria-hidden="true" />
   437→            <p className="text-sm font-medium">Could not load the timetable</p>
   438→            <p className="text-xs text-muted-foreground">It will retry automatically in a moment.</p>
   439→          </div>
   440→        ) : todaySlots.length === 0 ? (
   441→          <div className="flex flex-col items-center gap-1.5 rounded-xl border border-dashed border-border/70 py-8 text-center">
   442→            <CalendarDays className="size-6 text-muted-foreground/60" aria-hidden="true" />
   443→            <p className="text-sm font-medium">No classes today</p>
   444→            <p className="max-w-xs text-xs text-muted-foreground">
   445→              {mode === "live" && slots.length === 0
   446→                ? "Your weekly schedule is empty — the ERP doesn’t publish one, so add your real classes via Manage."

   448→            </p>
   449→          </div>
   450→        ) : (
   451→          <div className="space-y-2">
   452→            {ongoing && (
   453→              <div className="mb-2.5 flex items-center gap-2.5 rounded-xl border border-emerald-500/40 bg-emerald-500/[0.07] px-3.5 py-2.5">

   455→                  <Play className="size-4" aria-hidden="true" />
   456→                </span>

   459→                  <p className="truncate text-sm font-semibold">
   460→                    {ongoing.subjectCode} · {hhmm(ongoing.startMin)}–{hhmm(ongoing.endMin)}
   461→                  </p>
