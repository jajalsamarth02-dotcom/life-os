    98→    98→/** Compute a snooze target from "now" for the given preset. */
    99→    99→function snoozeTarget(preset: "h1" | "h3" | "tomorrow" | "week"): Date {
   100→   100→  const d = new Date();
   101→   101→  switch (preset) {
   102→   102→    case "h1":
   103→   103→      return new Date(d.getTime() + 60 * 60_000);
   104→   104→    case "h3":
   105→   105→      return new Date(d.getTime() + 3 * 60 * 60_000);
   106→   106→    case "tomorrow":
   107→   107→      d.setDate(d.getDate() + 1);
   108→   108→      d.setHours(9, 0, 0, 0);
   109→   109→      return d;
   110→   110→    case "week":
   111→   111→      d.setDate(d.getDate() + 7);
   112→   112→      d.setHours(9, 0, 0, 0);
   113→   113→      return d;
   114→   114→  }
   115→   115→}

   558→   558→/** Compact opt-in banner for system due-reminders (only until a decision). */
   559→   559→function BrowserRemindersBanner() {
   560→   560→  const [state, setState] = useState<{ permission: string; enabled: boolean; dismissed: boolean }>({
   561→   561→    permission: "default",
   562→   562→    enabled: false,
   563→   563→    dismissed: true,
   564→   564→  });
   565→   565→  const [busy, setBusy] = useState(false);
   566→   566→
   567→   567→  useEffect(() => {
   568→   568→    const read = () => {
   569→   569→      let dismissed = true;
   570→   570→      try {
   571→   571→        dismissed = window.localStorage.getItem("erp.browserReminders.bannerDismissed") === "1";
   572→   572→      } catch {
   573→   573→        /* non-fatal */
   574→   574→      }
   575→   575→      setState({
   576→   576→        permission: browserRemindersPermission(),
   577→   577→        enabled: browserRemindersEnabled(),
   578→   578→        dismissed,
   579→   579→      });
   580→   580→    };
   581→   581→    read();
   582→   582→    window.addEventListener(BROWSER_REMINDERS_EVENT, read);
   583→   583→    return () => window.removeEventListener(BROWSER_REMINDERS_EVENT, read);
   584→   584→  }, []);
   585→   585→
   586→   586→  if (state.dismissed || state.permission !== "default") return null;
   587→   587→
   588→   588→  const dismiss = () => {
   589→   589→    try {
   590→   590→      window.localStorage.setItem("erp.browserReminders.bannerDismissed", "1");
   591→   591→    } catch {
   592→   592→      /* non-fatal */
   593→   593→    }
   594→   594→    setState((s) => ({ ...s, dismissed: true }));
   595→   595→  };
   596→   596→
   597→   597→  const enable = async () => {
   598→   598→    setBusy(true);
   599→   599→    const result = await requestNotificationPermission();
   600→   600→    setBusy(false);
   601→   601→    if (result === "granted") {
   602→   602→      setBrowserRemindersEnabled(true);
   603→   603→      toast.success("Browser reminders on");
   604→   604→    } else if (result === "denied") {
   605→   605→      toast.error("Permission denied", { description: "Allow notifications for this site in your browser settings." });
   606→   606→    }
   607→   607→    dismiss();
   608→   608→  };
   609→   609→
   610→   610→  return (
   611→   611→    <div className="flex flex-wrap items-center gap-2.5 rounded-xl border border-primary/25 bg-primary/[0.05] px-4 py-3">
   612→   612→      <CalendarClock className="size-4 shrink-0 text-primary" aria-hidden="true" />
   613→   613→      <p className="min-w-0 flex-1 text-xs sm:text-sm">
   614→   614→        <span className="font-medium">Never miss a deadline.</span>{" "}
   615→   615→        <span className="text-muted-foreground">Get a system alert when a reminder becomes due.</span>
   616→   616→      </p>
   617→   617→      <div className="flex items-center gap-1.5">
   618→   618→        <Button size="sm" className="h-7 gap-1 px-2.5 text-xs" onClick={enable} disabled={busy}
   619→   619→          aria-label="Enable browser due-reminder notifications">
   620→   620→          {busy ? <Loader2 className="size-3.5 animate-spin" aria-hidden="true" /> : <BellRing className="size-3.5" aria-hidden="true" />}
   621→   621→          Enable
   622→   622→        </Button>
   623→   623→        <Button variant="ghost" size="sm" className="h-7 px-2 text-xs text-muted-foreground" onClick={dismiss}
   624→   624→          aria-label="Dismiss browser reminders banner">
   625→   625→          Not now
   626→   626→        </Button>
   627→   627→      </div>
   628→   628→    </div>
   629→   629→  );
   630→   630→}
