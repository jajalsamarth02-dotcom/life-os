    62→type PlanMode = "attend" | "skip";
    63→
    64→function WhatIfPlanner({ s, target }: { s: SubjectView; target: number }) {
    65→  const [mode, setMode] = useState<PlanMode>("attend");
    66→  const [n, setN] = useState(0);
    67→
    68→  const toTarget = classesToReachTarget(s.present, s.total, target);
    69→  const maxSkip = classesCanSkip(s.present, s.total, target);
    70→  // NOTE: this component is remounted with key={s.id} when the subject changes,
    71→  // so planner state resets naturally without effects.
    72→
    73→  const projected = useMemo(() => {
    74→    if (n === 0) return s.percentage;
    75→    if (mode === "attend") return ((s.present + n) / (s.total + n)) * 100;
    76→    return (s.present / (s.total + n)) * 100;
    77→  }, [mode, n, s.present, s.total, s.percentage]);
    78→
    79→  const delta = projected - s.percentage;
    80→  const projectedStatus: AttendanceStatus =
    81→    projected >= target ? "safe" : projected >= target - 5 ? "warning" : "danger";
    82→  const pc = statusColors(projectedStatus);
    83→
    84→  const setToTarget = () => {
    85→    setMode("attend");
    86→    setN(Math.max(1, toTarget));
    87→  };
    88→  const setToMaxSkip = () => {
    89→    setMode("skip");
    90→    setN(Math.max(1, maxSkip));
    91→  };
    92→
    93→  return (
    94→    <div className="rounded-xl border border-border/70 bg-muted/30 p-4">
    95→      <div className="flex flex-wrap items-center justify-between gap-2">
    96→        <p className="flex items-center gap-1.5 text-sm font-semibold tracking-tight">
    97→          <Zap className="size-4 text-amber-500" aria-hidden="true" />
    98→          What-if planner
    99→        </p>
   100→        {/* mode switch */}
   101→        <div
   102→          role="tablist"
   103→          aria-label="Planning mode"
   104→          className="flex rounded-lg border border-border/70 bg-background p-0.5"
   105→        >
   106→          {(
   107→            [
   108→              { key: "attend", label: "Attend", icon: Footprints },
   109→              { key: "skip", label: "Skip", icon: TrendingDown },
   110→            ] as const
   111→          ).map(({ key, label, icon: Icon }) => (
   112→            <button
   113→              key={key}
   114→              type="button"
   115→              role="tab"
   116→              aria-selected={mode === key}
   117→              onClick={() => {
   118→                setMode(key);
   119→                setN(0);
   120→              }}
   121→              className={cn(
   122→                "flex items-center gap-1 rounded-md px-2.5 py-1 text-xs font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   123→                mode === key
   124→                  ? key === "attend"
   125→                    ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400"
   126→                    : "bg-amber-500/15 text-amber-700 dark:text-amber-400"
   127→                  : "text-muted-foreground hover:bg-muted/60"
   128→              )}
   129→            >
   130→              <Icon className="size-3.5" aria-hidden="true" />
   131→              {label}
   132→            </button>
   133→          ))}
   134→        </div>
   135→      </div>
   136→
   137→      <p className="mt-2.5 text-xs text-muted-foreground">
   138→        {mode === "attend" ? (
   139→          <>
   140→            Simulate <b className="tabular-nums text-foreground">attending</b> the next classes of {s.code}. Each
   141→            attended class adds to both present and total.
   142→          </>
   143→        ) : (
   144→          <>
   145→            Simulate <b className="tabular-nums text-foreground">skipping</b> the next classes of {s.code}. Skipped
   146→            classes only increase the total — a fast way to see how fast attendance falls.
   147→          </>
   148→        )}
   149→      </p>
   150→
   151→      <div className="mt-3 flex items-center gap-3">
   152→        <Slider
   153→          value={[n]}
   154→          min={0}
   155→          max={40}
   156→          step={1}
   157→          aria-label={`Number of classes to ${mode} in the simulation`}
   158→          onValueChange={(v) => setN(v[0] ?? 0)}
   159→          className="flex-1"
   160→        />
   161→        <span className="w-14 shrink-0 rounded-lg border border-border/70 bg-background px-2 py-1 text-center text-sm font-bold tabular-nums">
   162→          {n}
   163→        </span>
   164→        <Button
   165→          variant="ghost"
   166→          size="icon"
   167→          className="size-8 shrink-0 text-muted-foreground"
   168→          aria-label="Reset simulation"
   169→          onClick={() => setN(0)}
   170→        >
   171→          <RotateCcw className="size-3.5" aria-hidden="true" />
   172→        </Button>
   173→      </div>
   174→      <div className="mt-1 flex justify-between text-[10px] text-muted-foreground/70 tabular-nums">
   175→        <span>0</span>
   176→        <span>10</span>
   177→        <span>20</span>
   178→        <span>30</span>
   179→        <span>40</span>
   180→      </div>
   181→
   182→      {/* Projection result */}
   183→      <div className={cn("mt-3 flex flex-wrap items-center justify-between gap-3 rounded-lg border px-3.5 py-3", pc.border, pc.bg)}>
   184→        <div>
   185→          <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   186→            Projected attendance
   187→          </p>
   188→          <div className="mt-0.5 flex items-baseline gap-2">
   189→            <span className={cn("text-2xl font-bold tabular-nums", pc.text)}>
   190→              {n === 0 ? s.percentage.toFixed(2) : projected.toFixed(2)}
   191→              <span className="text-sm font-semibold">%</span>
   192→            </span>
   193→            {n > 0 && (
   194→              <span
   195→                className={cn(
   196→                  "rounded-full px-1.5 py-0.5 text-[11px] font-semibold tabular-nums",
   197→                  delta >= 0 ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400" : "bg-red-500/15 text-red-700 dark:text-red-400"
   198→                )}
   199→              >
   200→                {delta >= 0 ? "+" : ""}
   201→                {delta.toFixed(2)} pts
   202→              </span>
   203→            )}
   204→          </div>
   205→        </div>
   206→        <div className="text-right">
   207→          {projectedStatus === "safe" ? (
   208→            <p className="flex items-center gap-1 text-xs font-medium text-emerald-700 dark:text-emerald-400">
   209→              <CheckCircle2 className="size-3.5" aria-hidden="true" /> Above {target}% target
   210→            </p>
   211→          ) : (
   212→            <p className="flex items-center gap-1 text-xs font-medium text-amber-700 dark:text-amber-400">
   213→              <Target className="size-3.5" aria-hidden="true" /> Below {target}% target
   214→            </p>
   215→          )}
   216→          <p className="mt-0.5 text-[11px] text-muted-foreground tabular-nums">
   217→            {mode === "attend"
   218→              ? `${s.present + n} present of ${s.total + n}`
   219→              : `${s.present} present of ${s.total + n}`}
   220→          </p>
   221→        </div>
   222→      </div>
   223→
   224→      {/* Shortcuts */}
   225→      <div className="mt-2.5 flex flex-wrap gap-1.5">
   226→        {s.status !== "safe" && toTarget > 0 && (
   227→          <Button variant="outline" size="sm" className="h-7 gap-1 border-emerald-500/40 px-2.5 text-[11px] text-emerald-700 hover:bg-emerald-500/10 hover:text-emerald-800 dark:text-emerald-400" onClick={setToTarget}>
   228→            <TrendingUp className="size-3" aria-hidden="true" />
   229→            Reach {target}% → attend {toTarget}
   230→          </Button>
   231→        )}
   232→        {maxSkip > 0 && (
   233→          <Button variant="outline" size="sm" className="h-7 gap-1 px-2.5 text-[11px]" onClick={setToMaxSkip}>
   234→            <TrendingDown className="size-3" aria-hidden="true" />
   235→            Max skips ({maxSkip}) — still {target}%
   236→          </Button>
   237→        )}
   238→      </div>
   239→    </div>
   240→  );
   241→}

   246→
   247→function SubjectTargetEditor({ s, globalTarget }: { s: SubjectView; globalTarget: number }) {
   248→  const qc = useQueryClient();
   249→  const [justSet, setJustSet] = useState<number | null>(null);
   250→
   251→  const mutation = useMutation({
   252→    mutationFn: (v: number | null) => putSubjectTarget(s.code, v),
   253→    onSuccess: (res) => {
   254→      qc.invalidateQueries({ queryKey: qk.attendance });
   255→      qc.invalidateQueries({ queryKey: qk.dashboard });
   256→      qc.invalidateQueries({ queryKey: qk.settings });
   257→      setJustSet(res.targetPct);
   258→      window.setTimeout(() => setJustSet(null), 1600);
   259→      toast.success(
   260→        res.hasCustomTarget
   261→          ? `Target for ${s.code} set to ${res.targetPct}%`
   262→          : `${s.code} reset to the global target (${res.targetPct}%)`
   263→      );
   264→    },
   265→    onError: (e) => toast.error(e instanceof Error ? e.message : "Could not save the target"),
   266→  });
   267→
   268→  const isActive = (p: number) => (s.hasCustomTarget ? s.targetPct === p : p === globalTarget);
   269→
   270→  return (
   271→    <div className="rounded-xl border border-border/70 bg-muted/30 p-4">
   272→      <div className="flex flex-wrap items-center justify-between gap-2">
   273→        <p className="flex items-center gap-1.5 text-sm font-semibold tracking-tight">
   274→          <Target className="size-4 text-emerald-600 dark:text-emerald-400" aria-hidden="true" />
   275→          Target for this subject
   276→        </p>
   277→        {s.hasCustomTarget ? (
   278→          <CustomTargetChip target={s.targetPct} />
   279→        ) : (
   280→          <Badge variant="outline" className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   281→            global {globalTarget}%
   282→          </Badge>
   283→        )}
   284→      </div>
   285→      <p className="mt-2 text-xs text-muted-foreground">
   286→        Different subjects can have different requirements (e.g. labs at 70%, theory at 75%). This changes the
   287→        status colors, warnings and what-if math for <b className="text-foreground">{s.code}</b> only.
   288→      </p>
   289→
   290→      <div className="mt-3 flex flex-wrap items-center gap-1.5" role="group" aria-label={`Attendance target for ${s.code}`}>
   291→        {TARGET_PRESETS.map((p) => (
   292→          <button
   293→            key={p}
   294→            type="button"
   295→            disabled={mutation.isPending}
   296→            onClick={() => {
   297→              if (!isActive(p)) mutation.mutate(p);
   298→            }}
   299→            aria-pressed={isActive(p)}
   300→            className={cn(
   301→              "h-8 min-w-11 rounded-lg border px-2 text-xs font-semibold tabular-nums transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring disabled:opacity-60",
   302→              isActive(p)
   303→                ? "border-emerald-500/50 bg-emerald-500/15 text-emerald-700 dark:text-emerald-400"
   304→                : "border-border/70 bg-background text-muted-foreground hover:border-emerald-500/40 hover:text-foreground"
   305→            )}
   306→          >
   307→            {p}%
   308→          </button>
   309→        ))}
   310→        <Button
   311→          variant="outline"
   312→          size="sm"
   313→          disabled={mutation.isPending || !s.hasCustomTarget}
   314→          className="h-8 gap-1 border-dashed text-xs text-muted-foreground"
   315→          onClick={() => mutation.mutate(null)}
   316→          aria-label={`Reset ${s.code} target to the global ${globalTarget}%`}
   317→        >
   318→          {mutation.isPending ? (
   319→            <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   320→          ) : (
   321→            <RotateCcw className="size-3.5" aria-hidden="true" />
   322→          )}
   323→          Default ({globalTarget}%)
   324→        </Button>
   325→        {justSet !== null && (
   326→          <span className="flex items-center gap-1 text-[11px] font-medium text-emerald-600 dark:text-emerald-400" role="status">
   327→            <CheckCircle2 className="size-3.5" aria-hidden="true" /> saved — {justSet}%
   328→          </span>
   329→        )}
   330→      </div>
   331→    </div>
   332→  );
   333→}

   369→function MiniAreaChart({ points }: { points: number[] }) {
   370→  const w = 320;
   371→  const h = 64;
   372→  const path = useMemo(() => {
   373→    const min = Math.min(...points);
   374→    const max = Math.max(...points);
   375→    const span = max - min || 1;
   376→    const stepX = w / (points.length - 1);
   377→    const coords = points.map((p, i) => {
   378→      const x = i * stepX;
   379→      const y = h - 6 - ((p - min) / span) * (h - 12);
   380→      return [x, y] as const;
   381→    });
   382→    const line = coords.reduce((acc, [x, y], i) => (i === 0 ? `M ${x},${y}` : `${acc} L ${x},${y}`), "");
   383→    const area = `${line} L ${w},${h} L 0,${h} Z`;
   384→    return { line, area };
   385→  }, [points]);
   386→
   387→  return (
   388→    <svg viewBox={`0 0 ${w} ${h}`} className="h-16 w-full" role="img" aria-label="Subject attendance trend">
   389→      <defs>
   390→        <linearGradient id="drawerTrendFill" x1="0" y1="0" x2="0" y2="1">
   391→          <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
   392→          <stop offset="100%" stopColor="#10b981" stopOpacity={0.02} />
   393→        </linearGradient>
   394→      </defs>
   395→      <path d={path.area} fill="url(#drawerTrendFill)" />
   396→      <path d={path.line} fill="none" stroke="#10b981" strokeWidth={2} strokeLinecap="round" strokeLinejoin="round" />
   397→    </svg>
   398→  );
   399→}

   405→function SubjectTimetable({ slots, subjectCode, subjectName }: { slots: TimetableSlotRow[]; subjectCode: string; subjectName: string }) {
   406→  const now = useNow();
   407→  const today = new Date(now).getDay();
   408→  const qc = useQueryClient();
   409→  const [adding, setAdding] = useState(false);
   410→  const [day, setDay] = useState<string>(String(today));
   411→  const [start, setStart] = useState("");
   412→  const [end, setEnd] = useState("");
   413→  const [conflicts, setConflicts] = useState<SlotConflict[] | null>(null);
   414→
   415→  const addMutation = useMutation({
   416→    mutationFn: (force: boolean) =>
   417→      createSlot({ dayOfWeek: Number(day), start, end, subjectCode, force }),
   418→    onSuccess: (_res, force) => {
   419→      qc.invalidateQueries({ queryKey: qk.timetable });
   420→      qc.invalidateQueries({ queryKey: qk.dashboard });
   421→      toast.success(
   422→        force
   423→          ? `${DAY_SHORT[Number(day)]} ${start}–${end} added for ${subjectCode} (overlap accepted)`
   424→          : `${DAY_SHORT[Number(day)]} ${start}–${end} added for ${subjectCode}`
   425→      );
   426→      setConflicts(null);
   427→      setAdding(false);
   428→      setStart("");
   429→      setEnd("");
   430→    },
   431→    onError: (e) => {
   432→      if (e instanceof ApiError && e.status === 409 && Array.isArray(e.body.conflicts)) {
   433→        setConflicts(e.body.conflicts as SlotConflict[]);
   434→        return;
   435→      }
   436→      setConflicts(null);
   437→      toast.error(e instanceof Error ? e.message : "Could not add the class");
   438→    },
   439→  });
   440→
   441→  const valid = /^\d{2}:\d{2}$/.test(start) && /^\d{2}:\d{2}$/.test(end) && start < end;
   442→  const resetConflicts = () => setConflicts(null);
   443→
   444→  const byDay = useMemo(() => {
   445→    const m = new Map<number, TimetableSlotRow[]>();
   446→    for (const s of slots) {
   447→      const arr = m.get(s.dayOfWeek) ?? [];
   448→      arr.push(s);
   449→      m.set(s.dayOfWeek, arr);
   450→    }
   451→    for (const arr of m.values()) arr.sort((a, b) => a.startMin - b.startMin);
   452→    return m;
   453→  }, [slots]);
   454→
   455→  const orderedDays = useMemo(() => {
   456→    // Today first, then the rest of the week in order
   457→    const days = [1, 2, 3, 4, 5, 6, 0].filter((d) => byDay.has(d));
   458→    return days.sort((a, b) => ((a - today + 7) % 7) - ((b - today + 7) % 7));
   459→  }, [byDay, today]);
   460→
   461→  return (
   462→    <div className="rounded-xl border border-border/70 p-4">
   463→      <div className="flex items-center gap-1.5">
   464→        <p className="flex items-center gap-1.5 text-sm font-semibold tracking-tight">
   465→          <CalendarDays className="size-4 text-emerald-500" aria-hidden="true" />
   466→          Weekly classes
   467→        </p>
   468→        <Badge variant="secondary" className="ml-auto tabular-nums">{slots.length}/week</Badge>
   469→        <Button
   470→          variant="ghost"
   471→          size="sm"
   472→          className="h-7 gap-1 px-2 text-[11px] text-muted-foreground"
   473→          onClick={() => setAdding((v) => !v)}
   474→          aria-expanded={adding}
   475→          aria-label={adding ? "Cancel adding a class" : `Add a weekly class for ${subjectName}`}
   476→        >
   477→          {adding ? <X className="size-3.5" aria-hidden="true" /> : <CalendarPlus className="size-3.5" aria-hidden="true" />}
   478→          {adding ? "Cancel" : "Add"}
   479→        </Button>
   480→      </div>
   481→
   482→      {adding && (
   483→        <div className="mt-3 rounded-lg border border-border/60 bg-background p-3">
   484→          <div className="grid grid-cols-3 gap-2">
   485→            <div className="space-y-1">
   486→              <label htmlFor="slot-day" className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   487→                Day
   488→              </label>
   489→              <Select value={day} onValueChange={(v) => { setDay(v); resetConflicts(); }}>
   490→                <SelectTrigger id="slot-day" className="h-9 text-xs" aria-label="Day of the week">
   491→                  <SelectValue />
   492→                </SelectTrigger>
   493→                <SelectContent>
   494→                  {[1, 2, 3, 4, 5, 6, 0].map((d) => (
   495→                    <SelectItem key={d} value={String(d)} className="text-xs">
   496→                      {DAY_SHORT[d]}
   497→                      {d === today ? " · today" : ""}
   498→                    </SelectItem>
   499→                  ))}
   500→                </SelectContent>
   501→              </Select>
   502→            </div>
   503→            <div className="space-y-1">
   504→              <label htmlFor="slot-start" className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   505→                Start
   506→              </label>
   507→              <Input
   508→                id="slot-start"
   509→                type="time"
   510→                value={start}
   511→                onChange={(e) => { setStart(e.target.value); resetConflicts(); }}
   512→                className="h-9 text-xs"
   513→              />
   514→            </div>
   515→            <div className="space-y-1">
   516→              <label htmlFor="slot-end" className="text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">
   517→                End
   518→              </label>
   519→              <Input
   520→                id="slot-end"
   521→                type="time"
   522→                value={end}
   523→                onChange={(e) => { setEnd(e.target.value); resetConflicts(); }}
   524→                className="h-9 text-xs"
   525→              />
   526→            </div>
   527→          </div>
   528→          {conflicts && conflicts.length > 0 && (
   529→            <div role="alert" className="mt-2.5 rounded-lg border border-amber-500/50 bg-amber-500/[0.08] p-2.5">
   530→              <p className="flex items-center gap-1.5 text-[11px] font-semibold text-amber-700 dark:text-amber-400">
   531→                <TriangleAlert className="size-3.5 shrink-0" aria-hidden="true" />

   533→              </p>
   534→              <ul className="mt-1.5 space-y-1">
   535→                {conflicts.map((c) => (
   536→                  <li key={c.id} className="flex items-center gap-1.5 text-[11px] text-amber-800 dark:text-amber-300/90">
   537→                    <span className="size-1.5 shrink-0 rounded-full bg-amber-500" aria-hidden="true" />
   538→                    <span className="font-mono font-semibold">{c.subjectCode}</span>
   539→                    <span className="tabular-nums">{c.start}–{c.end}</span>
   540→                    {c.source === "demo" && <span className="text-amber-700/60 dark:text-amber-400/50">· demo</span>}
   541→                  </li>
   542→                ))}
   543→              </ul>
   544→              <div className="mt-2 flex gap-1.5">
   545→                <Button
   546→                  size="sm"
   547→                  variant="outline"
   548→                  className="h-7 gap-1 border-amber-500/50 px-2.5 text-[11px] text-amber-700 hover:bg-amber-500/15 dark:text-amber-400"
   549→                  disabled={addMutation.isPending}
   550→                  onClick={() => addMutation.mutate(true)}
   551→                >
   552→                  {addMutation.isPending ? <Loader2 className="size-3 animate-spin" aria-hidden="true" /> : <CalendarPlus className="size-3" aria-hidden="true" />}
   553→                  Add anyway
   554→                </Button>
   555→                <Button
   556→                  size="sm"
   557→                  variant="ghost"
   558→                  className="h-7 px-2.5 text-[11px] text-muted-foreground"

   560→                >
   561→                  Pick another time
   562→                </Button>
   563→              </div>
   564→            </div>
   565→          )}
   566→          <div className="mt-2.5 flex items-center justify-between gap-2">
   567→            <p className="text-[11px] text-muted-foreground">
   568→              {valid ? "Looks good — saved locally, never written to the ERP." : "Pick a day and a start/end time (end must be after start)."}
   569→            </p>
   570→            <Button
   571→              size="sm"
   572→              className="h-8 gap-1 text-xs"
   573→              disabled={!valid || addMutation.isPending}
   574→              onClick={() => addMutation.mutate(false)}
   575→            >
   576→              {addMutation.isPending && <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />}
   577→              Add class
   578→            </Button>
   579→          </div>
   580→        </div>
   581→      )}
   582→
   583→      {slots.length === 0 ? (
   584→        <p className="mt-2 text-xs text-muted-foreground">
   585→          No timetable slots recorded for this subject{adding ? "." : " — use the Add button above to record one."}
   586→        </p>
   587→      ) : (
   588→        <ul className="mt-3 space-y-1.5">
   589→          {orderedDays.map((d) => (
   590→            <li key={d} className="flex items-center gap-2.5">
   591→              <span
   592→                className={cn(
   593→                  "w-10 shrink-0 rounded-md px-1.5 py-0.5 text-center text-[11px] font-semibold",
   594→                  d === today ? "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400" : "bg-muted text-muted-foreground"
   595→                )}
   596→              >
   597→                {DAY_SHORT[d]}
   598→              </span>
   599→              <div className="flex flex-wrap gap-1.5">
   600→                {byDay.get(d)!.map((slot) => (
   601→                  <span
   602→                    key={slot.id}
   603→                    className="rounded-md border border-border/60 bg-background px-2 py-0.5 text-[11px] tabular-nums text-foreground/80"
   604→                  >
   605→                    {hhmm(slot.startMin)}–{hhmm(slot.endMin)}
   606→                  </span>
   607→                ))}
   608→              </div>
   609→            </li>
   610→          ))}
   611→        </ul>
   612→      )}
   613→    </div>
   614→  );
   615→}
