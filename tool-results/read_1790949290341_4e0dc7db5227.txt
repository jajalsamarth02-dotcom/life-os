    60→const SUBJECT_SORTS = [
    61→  { key: "lowest", label: "Lowest attendance first" },

    69→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    70→  const copy = [...list];
    71→  switch (sort) {
    72→    case "lowest":
    73→      return copy.sort((a, b) => a.percentage - b.percentage);
    74→    case "highest":
    75→      return copy.sort((a, b) => b.percentage - a.percentage);
    76→    case "name":
    77→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    78→    case "recent":
    79→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    80→  }
    81→}

    83→/** Live countdown label for the next upcoming deadline. */
    84→function countdownLabel(dueAt: string, now: number): string {
    85→  const diff = new Date(dueAt).getTime() - now;
    86→  const abs = Math.abs(diff);
    87→  const mins = Math.round(abs / 60_000);
    88→  const hours = Math.round(abs / 3_600_000);
    89→  const days = Math.round(abs / 86_400_000);
    90→  let span: string;
    91→  if (mins < 60) span = `${mins} min`;
    92→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    93→  else span = `${days} day${days === 1 ? "" : "s"}`;
    94→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    95→}

    97→/** Relative urgency pill for an ALA deadline: red overdue/≤48h, amber ≤7d, muted later. */
    98→function deadlinePill(dueAt: string, now: number): { label: string; className: string } {
    99→  const due = new Date(dueAt).getTime();
   100→  const diff = due - now;
   101→  if (diff < 0)
   102→    return {
   103→      label: countdownLabel(dueAt, now),
   104→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   105→    };
   106→  if (diff <= 2 * 86_400_000)
   107→    return {
   108→      label: countdownLabel(dueAt, now),
   109→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   110→    };
   111→  if (diff <= 7 * 86_400_000)
   112→    return {
   113→      label: countdownLabel(dueAt, now),
   114→      className: "bg-amber-500/10 text-amber-700 dark:text-amber-400",
   115→    };
   116→  return { label: countdownLabel(dueAt, now), className: "bg-muted text-muted-foreground" };
   117→}

   223→function NextUpStrip({ items }: { items: Reminder[] }) {
   224→  const now = useNow();
   225→  const upcoming = items
   226→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
   227→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
   228→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
   229→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   230→
   231→  if (upcoming.length === 0) return null;
   232→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   233→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   234→
   235→  return (
   236→    <div
   237→      className={cn(
   238→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   239→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   240→      )}
   241→    >
   242→      <span
   243→        className={cn(
   244→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   245→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   246→        )}
   247→      >
   248→        <Hourglass className="size-4" aria-hidden="true" />
   249→      </span>
   250→      <div className="min-w-0 flex-1">
   251→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   252→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   253→        </p>

   257→              className="shrink-0 rounded-md px-1.5 py-0.5 font-mono text-[10px] font-semibold"

   273→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"

   279→      </div>

   415→/** Live "in 12 min"-style countdown for a future instant (null when absent/past). */
   416→function inCountdown(iso: string | null, now: number): string | null {
   417→  if (!iso) return null;
   418→  const diff = new Date(iso).getTime() - now;
   419→  if (diff <= 0) return null;
   420→  const mins = Math.max(1, Math.round(diff / 60_000));
   421→  if (mins < 60) return `in ${mins} min`;
   422→  const hrs = Math.floor(mins / 60);
   423→  const rem = mins % 60;
   424→  return rem > 0 ? `in ${hrs}h ${rem}m` : `in ${hrs}h`;
   425→}

   600→function NotificationRow({
   601→  n,
   602→}: {
   603→  n: { id: string; title: string; message: string; severity: string; createdAt: string; read: boolean };
   604→}) {
   605→  const now = useNow();
   606→  const Icon =
   607→    n.severity === "critical"
   608→      ? XCircle
   609→      : n.severity === "warning"
   610→        ? AlertTriangle
   611→        : n.severity === "success"
   612→          ? CheckCircle2
   613→          : Info;
   614→  const cls =
   615→    n.severity === "critical"
   616→      ? "text-red-600 dark:text-red-400 bg-red-500/10"
   617→      : n.severity === "warning"
   618→        ? "text-amber-600 dark:text-amber-400 bg-amber-500/10"
   619→        : n.severity === "success"
   620→          ? "text-emerald-600 dark:text-emerald-400 bg-emerald-500/10"
   621→          : "text-muted-foreground bg-muted";
   622→  return (
   623→    <li className="flex items-start gap-2.5 rounded-lg p-2 transition-colors hover:bg-muted/50">
   624→      <span className={cn("mt-0.5 flex size-6 shrink-0 items-center justify-center rounded-md", cls)}>
   625→        <Icon className="size-3.5" aria-hidden="true" />
   626→      </span>
   627→      <div className="min-w-0 flex-1">
   628→        <p className="text-sm font-medium leading-snug">{n.title}</p>
   629→        <p className="line-clamp-2 text-xs text-muted-foreground">{compactErrorText(n.message, 240)}</p>
   630→        <p className="mt-0.5 text-[11px] text-muted-foreground/80">{ago(n.createdAt, now)}</p>
   631→      </div>
   632→      {!n.read && <span className="mt-1.5 size-1.5 shrink-0 rounded-full bg-primary" aria-label="unread" />}
   633→    </li>
   634→  );
   635→}

   662→  const [sort, setSort] = useState<SubjectSort>("lowest");
   663→  /* Store the CODE (not the object) and derive the view from fresh query data,
   664→     so the drawer reflects updated targets/percentages after refetches. */
   665→  const [selectedCode, setSelectedCode] = useState<string | null>(null);
   666→  const [digestOpen, setDigestOpen] = useState(false);
   667→  const [showAllChanges, setShowAllChanges] = useState(false);
   668→  const [showAllNotifs, setShowAllNotifs] = useState(false);
   669→  const { data, isPending, isError, error } = useQuery({
   670→    queryKey: qk.dashboard,
   671→    queryFn: fetchDashboard,
   672→    refetchInterval: 20_000,
   673→  });
   674→  const history = useQuery({
   675→    queryKey: qk.history,
   676→    queryFn: fetchHistory,
   677→    refetchInterval: 60_000,
   678→  });
   679→
   680→  const weekDelta = useMemo(() => {
   681→    const pts = history.data?.overall ?? [];
   682→    if (pts.length < 2) return null;
   683→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   684→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   685→    const last = pts[pts.length - 1];
   686→    const d = last.percentage - first.percentage;
   687→    if (Math.abs(d) < 0.005) return null;
   688→    // Be honest about the window: when tracked history is younger than the
   689→    // 7-day span (e.g. live tracking just started), label the real span.
   690→    const spanDays = Math.max(
   691→      0,
   692→      (new Date(last.t).getTime() - new Date(first.t).getTime()) / 86_400_000
   693→    );
   694→    return { delta: d, spanDays, fullWeek: spanDays >= 6.5 };
