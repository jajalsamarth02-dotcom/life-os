    65→    65→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    66→    66→  const copy = [...list];
    67→    67→  switch (sort) {
    68→    68→    case "lowest":
    69→    69→      return copy.sort((a, b) => a.percentage - b.percentage);
    70→    70→    case "highest":
    71→    71→      return copy.sort((a, b) => b.percentage - a.percentage);
    72→    72→    case "name":
    73→    73→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    74→    74→    case "recent":
    75→    75→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    76→    76→  }
    77→    77→}

    79→    79→/** Live countdown label for the next upcoming deadline. */
    80→    80→function countdownLabel(dueAt: string, now: number): string {
    81→    81→  const diff = new Date(dueAt).getTime() - now;
    82→    82→  const abs = Math.abs(diff);
    83→    83→  const mins = Math.round(abs / 60_000);
    84→    84→  const hours = Math.round(abs / 3_600_000);
    85→    85→  const days = Math.round(abs / 86_400_000);
    86→    86→  let span: string;
    87→    87→  if (mins < 60) span = `${mins} min`;
    88→    88→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    89→    89→  else span = `${days} day${days === 1 ? "" : "s"}`;
    90→    90→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    91→    91→}

    93→    93→function NextUpStrip({ items }: { items: Reminder[] }) {
    94→    94→  const now = useNow();
    95→    95→  const upcoming = items
    96→    96→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
    97→    97→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
    98→    98→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
    99→    99→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   100→   100→
   101→   101→  if (upcoming.length === 0) return null;
   102→   102→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   103→   103→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   104→   104→
   105→   105→  return (
   106→   106→    <div
   107→   107→      className={cn(
   108→   108→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   109→   109→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   110→   110→      )}
   111→   111→    >
   112→   112→      <span
   113→   113→        className={cn(
   114→   114→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   115→   115→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   116→   116→        )}
   117→   117→      >
   118→   118→        <Hourglass className="size-4" aria-hidden="true" />
   119→   119→      </span>

   121→   121→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   122→   122→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   123→   123→        </p>
   124→   124→        <p className="truncate text-sm font-medium">{shown.title}</p>
   125→   125→      </div>
   126→   126→      <div className="shrink-0 text-right">
   127→   127→        <p
   128→   128→          className={cn(
   129→   129→            "text-sm font-bold tabular-nums",
   130→   130→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"
   131→   131→          )}

   143→   143→function DueBadge({ iso, now }: { iso: string; now: number }) {
   144→   144→  const d = new Date(iso);

   146→   146→    return <span className="text-xs font-medium text-red-600 dark:text-red-400">{fmtDT(iso)} · overdue</span>;
   147→   147→  }
   148→   148→  return <span className="text-xs text-muted-foreground">{fmtDT(iso)}</span>;
   149→   149→}

   504→   504→  }

   506→   506→  const { overall, subjects, target, mode } = data;
