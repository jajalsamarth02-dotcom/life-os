    94→function loadSavedSort(): SubjectSort {
    95→  if (typeof window === "undefined") return "lowest";
    96→  try {
    97→    const saved = window.localStorage.getItem(SUBJECT_SORT_KEY);
    98→    return SUBJECT_SORTS.some((o) => o.key === saved) ? (saved as SubjectSort) : "lowest";
    99→  } catch {
   100→    return "lowest";
   101→  }
   102→}
   103→
   104→/* No-data subjects (total === 0) have no attendance yet — a 0.00% is not a
   105→   real value, so they never compete with actual percentages: they sink to
   106→   the end of both directions with a name tie-break. */
   107→function noDataLast(a: SubjectView, b: SubjectView): number {
   108→  const an = a.total === 0;
   109→  const bn = b.total === 0;
   110→  if (an !== bn) return an ? 1 : -1;
   111→  return 0;
   112→}
   113→
   114→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
   115→  const copy = [...list];
   116→  switch (sort) {
   117→    case "lowest":
   118→      return copy.sort(
   119→        (a, b) => noDataLast(a, b) || a.percentage - b.percentage || a.name.localeCompare(b.name)
   120→      );
   121→    case "highest":
   122→      return copy.sort(
   123→        (a, b) => noDataLast(a, b) || b.percentage - a.percentage || a.name.localeCompare(b.name)
   124→      );
   125→    case "name":
   126→      return copy.sort((a, b) => a.name.localeCompare(b.name));
   127→    case "recent":
   128→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
   129→  }
   130→}

   132→/** Live countdown label for the next upcoming deadline. */
   133→function hhmmFromMin(min: number): string {
   134→  return `${String(Math.floor(min / 60)).padStart(2, "0")}:${String(min % 60).padStart(2, "0")}`;
   135→}
   136→
   137→function countdownLabel(dueAt: string, now: number): string {
   138→  const diff = new Date(dueAt).getTime() - now;
   139→  const abs = Math.abs(diff);
   140→  const mins = Math.round(abs / 60_000);
   141→  const hours = Math.round(abs / 3_600_000);
   142→  const days = Math.round(abs / 86_400_000);
   143→  let span: string;
   144→  if (mins < 60) span = `${mins} min`;
   145→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
   146→  else span = `${days} day${days === 1 ? "" : "s"}`;
   147→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
   148→}

   150→/** Relative urgency pill for an ALA deadline: red overdue/≤48h, amber ≤7d, muted later. */
   151→function deadlinePill(dueAt: string, now: number): { label: string; className: string } {
   152→  const due = new Date(dueAt).getTime();
   153→  const diff = due - now;
   154→  if (diff < 0)
   155→    return {
   156→      label: countdownLabel(dueAt, now),
   157→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   158→    };
   159→  if (diff <= 2 * 86_400_000)
   160→    return {
   161→      label: countdownLabel(dueAt, now),
   162→      className: "bg-red-500/10 text-red-600 dark:text-red-400",
   163→    };
   164→  if (diff <= 7 * 86_400_000)
   165→    return {
   166→      label: countdownLabel(dueAt, now),
   167→      className: "bg-amber-500/10 text-amber-700 dark:text-amber-400",
   168→    };
   169→  return { label: countdownLabel(dueAt, now), className: "bg-muted text-muted-foreground" };
   170→}

   439→function NextUpStrip({
   440→  items,
   441→  flagPriorities,
   442→}: {
   443→  items: Reminder[];
   444→  /** externalId → priority (1–3) from the local AlaNote overlay. */
   445→  flagPriorities: Map<string, number>;
   446→}) {
   447→  const now = useNow();
   448→  /* Next class TODAY from the synced routine — shares the timetable query
   449→     cache with the week plan + timetable cards, so no extra request. The
   450→     batch-pick math (same-batch collapse + parallel picks) matches the
   451→     week plan exactly, so the chip and the plan can never disagree. */
   452→  const { data: ttRows } = useQuery({ queryKey: qk.timetable, queryFn: fetchTimetable });
   453→  const nowDay = new Date(now).getDay();
   454→  const nowMin = new Date(now).getHours() * 60 + new Date(now).getMinutes();
   455→  const classInfo = useMemo(() => {
   456→    const attending = applyBatchPicks(ttRows?.slots ?? [], ttRows?.picks ?? {}).attending;
   457→    const today = attending.filter((s) => s.dayOfWeek === nowDay);
   458→    const left = today
   459→      .filter((s) => s.endMin > nowMin)
   460→      .sort((a, b) => a.startMin - b.startMin);
   461→    const next = left[0] ?? null;
   462→    return {
   463→      next,
   464→      ongoing: next !== null && next.startMin <= nowMin,
   465→      allDone: next === null && today.length > 0,
   466→    };
   467→  }, [ttRows, nowDay, nowMin]);
   468→  const upcoming = items
   469→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
   470→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
   471→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
   472→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
   473→
   474→  if (upcoming.length === 0) return null;
   475→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   476→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   477→
   478→  return (
   479→    <div
   480→      className={cn(
   481→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   482→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   483→      )}
   484→    >
   485→      <span
   486→        className={cn(
   487→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   488→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   489→        )}
   490→      >
   491→        <Hourglass className="size-4" aria-hidden="true" />
   492→      </span>
   493→      <div className="min-w-0 flex-1">
   494→        <p className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   495→          <span className="relative flex size-1.5 shrink-0" aria-hidden="true">

   718→/** Live "in 12 min"-style countdown for a future instant (null when absent/past). */
   719→function inCountdown(iso: string | null, now: number): string | null {
   720→  if (!iso) return null;
   721→  const diff = new Date(iso).getTime() - now;
   722→  if (diff <= 0) return null;
   723→  const mins = Math.max(1, Math.round(diff / 60_000));
   724→  if (mins < 60) return `in ${mins} min`;
   725→  const hrs = Math.floor(mins / 60);
   726→  const rem = mins % 60;
   727→  return rem > 0 ? `in ${hrs}h ${rem}m` : `in ${hrs}h`;
   728→}
