    54→const SUBJECT_SORTS = [
    55→  { key: "lowest", label: "Lowest attendance first" },

    63→function sortSubjects(list: SubjectView[], sort: SubjectSort): SubjectView[] {
    64→  const copy = [...list];
    65→  switch (sort) {
    66→    case "lowest":
    67→      return copy.sort((a, b) => a.percentage - b.percentage);
    68→    case "highest":
    69→      return copy.sort((a, b) => b.percentage - a.percentage);
    70→    case "name":
    71→      return copy.sort((a, b) => a.name.localeCompare(b.name));
    72→    case "recent":
    73→      return copy.sort((a, b) => new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime());
    74→  }
    75→}

    77→/** Live countdown label for the next upcoming deadline. */
    78→function countdownLabel(dueAt: string, now: number): string {
    79→  const diff = new Date(dueAt).getTime() - now;
    80→  const abs = Math.abs(diff);
    81→  const mins = Math.round(abs / 60_000);
    82→  const hours = Math.round(abs / 3_600_000);
    83→  const days = Math.round(abs / 86_400_000);
    84→  let span: string;
    85→  if (mins < 60) span = `${mins} min`;
    86→  else if (hours < 36) span = `${hours} hr${hours === 1 ? "" : "s"}`;
    87→  else span = `${days} day${days === 1 ? "" : "s"}`;
    88→  return diff >= 0 ? `in ${span}` : `${span} overdue`;
    89→}

    91→function NextUpStrip({ items }: { items: Reminder[] }) {
    92→  const now = useNow();
    93→  const upcoming = items
    94→    .filter((r) => r.status === "ACTIVE" && r.type !== "ATTENDANCE_WARNING")
    95→    .sort((a, b) => new Date(a.dueAt).getTime() - new Date(b.dueAt).getTime());
    96→  const overdue = upcoming.filter((r) => new Date(r.dueAt).getTime() < now);
    97→  const next = upcoming.find((r) => new Date(r.dueAt).getTime() >= now);
    98→
    99→  if (upcoming.length === 0) return null;
   100→  const shown = overdue.length > 0 ? overdue[0] : (next ?? upcoming[0]);
   101→  const isOverdue = new Date(shown.dueAt).getTime() < now;
   102→
   103→  return (
   104→    <div
   105→      className={cn(
   106→        "flex items-center gap-3 rounded-xl border px-4 py-3",
   107→        isOverdue ? "border-red-500/30 bg-red-500/[0.05]" : "border-primary/25 bg-primary/[0.04]"
   108→      )}
   109→    >
   110→      <span
   111→        className={cn(
   112→          "flex size-9 shrink-0 items-center justify-center rounded-lg",
   113→          isOverdue ? "bg-red-500/10 text-red-600 dark:text-red-400" : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   114→        )}
   115→      >
   116→        <Hourglass className="size-4" aria-hidden="true" />
   117→      </span>
   118→      <div className="min-w-0 flex-1">
   119→        <p className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   120→          Next up {overdue.length > 0 && <span className="text-red-500">· {overdue.length} overdue</span>}
   121→        </p>
   122→        <p className="truncate text-sm font-medium">{shown.title}</p>
   123→      </div>
   124→      <div className="shrink-0 text-right">
   125→        <p
   126→          className={cn(
   127→            "text-sm font-bold tabular-nums",
   128→            isOverdue ? "text-red-600 dark:text-red-400" : "text-emerald-600 dark:text-emerald-400"
   129→          )}
   130→        >
   131→          {countdownLabel(shown.dueAt, now)}
   132→        </p>
   133→        <p className="text-[11px] text-muted-foreground">{fmtDT(shown.dueAt)}</p>
   134→      </div>
   135→    </div>
   136→  );
   137→}

   145→  }
   146→  return <span className="text-xs text-muted-foreground">{fmtDT(iso)}</span>;

   166→function RemindersPreview({

   410→export function DashboardTab({
   411→  onOpenAttendance,
   412→  onOpenSettings,
   413→}: {
   414→  onOpenAttendance: () => void;
   415→  onOpenSettings: () => void;
   416→}) {
   417→  const [sort, setSort] = useState<SubjectSort>("lowest");
   418→  /* Store the CODE (not the object) and derive the view from fresh query data,
   419→     so the drawer reflects updated targets/percentages after refetches. */
   420→  const [selectedCode, setSelectedCode] = useState<string | null>(null);
   421→  const [digestOpen, setDigestOpen] = useState(false);
   422→  const { data, isPending, isError, error } = useQuery({
   423→    queryKey: qk.dashboard,
   424→    queryFn: fetchDashboard,
   425→    refetchInterval: 20_000,
   426→  });
   427→  const history = useQuery({
   428→    queryKey: qk.history,
   429→    queryFn: fetchHistory,
   430→    refetchInterval: 60_000,
   431→  });
   432→
   433→  const weekDelta = useMemo(() => {
   434→    const pts = history.data?.overall ?? [];
   435→    if (pts.length < 2) return null;
   436→    const cutoff = Date.now() - 7 * 24 * 3600 * 1000;
   437→    const first = pts.find((p) => new Date(p.t).getTime() >= cutoff) ?? pts[0];
   438→    const last = pts[pts.length - 1];
   439→    const d = last.percentage - first.percentage;
   440→    return Math.abs(d) < 0.005 ? null : d;
   441→  }, [history.data]);
   442→
   443→  if (isPending) {
   444→    return (
   445→      <div className="space-y-4">
   446→        <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-5">
   447→          <CardSkeletonLight className="md:col-span-2 xl:col-span-2" />
   448→          <CardSkeletonLight />
   449→          <CardSkeletonLight />
   450→          <CardSkeletonLight />
   451→        </div>
   452→        <CardSkeletonLight />
   453→        <div className="grid gap-4 lg:grid-cols-2">
   454→          <CardSkeletonLight />
   455→          <CardSkeletonLight />
   456→        </div>
   457→      </div>
   458→    );
   459→  }
   460→
   461→  if (isError || !data) {
   462→    return (
   463→      <Card className="rounded-xl border-red-500/30 bg-red-500/5">
   464→        <CardContent className="flex items-center gap-3 p-6 text-sm text-red-600 dark:text-red-400">
   465→          <XCircle className="size-5 shrink-0" aria-hidden="true" />
   466→          Could not load the dashboard: {error?.message ?? "unknown error"}
   467→        </CardContent>
   468→      </Card>
   469→    );
   470→  }
   471→
   472→  const { overall, subjects, target, mode } = data;
   473→  const sorted = sortSubjects(subjects, sort);
   474→
   475→  if (mode === "none") {
   476→    return (
   477→      <EmptyHero
   478→        title="No ERP data available yet."
   479→        description="Connect your ERP account or run a first synchronization to fetch your attendance and reminders. Nothing is shown until real data arrives."
   480→        actionLabel="Connect / Sync ERP"
   481→        onAction={onOpenSettings}
   482→      />
   483→    );
   484→  }
   485→
   486→  return (
   487→    <motion.div
   488→      className="space-y-4"
   489→      initial="hidden"
   490→      animate="show"
   491→      variants={{
   492→        hidden: {},
   493→        show: { transition: { staggerChildren: 0.05 } },
   494→      }}
   495→    >
   496→      {/* Subject detail drawer (also covers what-if planning) */}
   497→      <SubjectDrawer
   498→        subject={data.subjects.find((s) => s.code === selectedCode) ?? null}
   499→        target={target}
   500→        globalTarget={target}
   501→        open={selectedCode !== null}
   502→        onOpenChange={(o) => {
   503→          if (!o) setSelectedCode(null);
   504→        }}
   505→      />
   506→
   507→      {/* Row 0 — next deadline countdown */}
   508→      <NextUpStrip items={data.reminders.items} />
   509→
   510→      {/* Row 1 — overall + stats */}
   511→      <motion.div
   512→        className="grid gap-4 md:grid-cols-2 xl:grid-cols-5"
   513→        variants={{ hidden: { opacity: 0, y: 10 }, show: { opacity: 1, y: 0, transition: { duration: 0.25, ease: "easeOut" } } }}
