    73→    73→/** Tiny CGPA sparkline from the REGULAR exams' printed career CGPA line. */
    74→    74→function CgpaSparkline({ points }: { points: CgpaPoint[] }) {
    75→    75→  if (points.length < 2) return null;
    76→    76→  const W = 96;
    77→    77→  const H = 26;
    78→    78→  const PAD = 3;
    79→    79→  const vals = points.map((p) => p.cgpa);
    80→    80→  const min = Math.min(...vals);
    81→    81→  const max = Math.max(...vals);
    82→    82→  const span = Math.max(max - min, 0.8); // never amplify a near-flat line
    83→    83→  const xs = points.map((_, i) => PAD + (i * (W - 2 * PAD)) / (points.length - 1));
    84→    84→  const ys = vals.map((v) => H - PAD - ((v - min) / span) * (H - 2 * PAD));
    85→    85→  const line = xs.map((x, i) => `${i ? "L" : "M"}${x.toFixed(1)},${ys[i].toFixed(1)}`).join(" ");
    86→    86→  const area = `${line} L${xs[xs.length - 1].toFixed(1)},${H - PAD} L${xs[0].toFixed(1)},${H - PAD} Z`;
    87→    87→  const last = points.length - 1;
    88→    88→  return (
    89→    89→    <svg
    90→    90→      width={W}
    91→    91→      height={H}
    92→    92→      viewBox={`0 0 ${W} ${H}`}
    93→    93→      className="mt-1.5"
    94→    94→      role="img"
    95→    95→      aria-label={`CGPA progression: ${vals.map((v) => v.toFixed(2)).join(" → ")}`}
    96→    96→    >
    97→    97→      <defs>
    98→    98→        <linearGradient id="cgpa-spark-fill" x1="0" y1="0" x2="0" y2="1">
    99→    99→          <stop offset="0%" stopColor="rgb(139 92 246)" stopOpacity="0.28" />
   100→   100→          <stop offset="100%" stopColor="rgb(139 92 246)" stopOpacity="0" />
   101→   101→        </linearGradient>
   102→   102→      </defs>
   103→   103→      <path d={area} fill="url(#cgpa-spark-fill)" />
   104→   104→      <path d={line} fill="none" stroke="rgb(139 92 246)" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   105→   105→      {xs.map((x, i) => (
   106→   106→        <circle
   107→   107→          key={i}
   108→   108→          cx={x}
   109→   109→          cy={ys[i]}
   110→   110→          r={i === last ? 2.2 : 1.5}
   111→   111→          fill="rgb(139 92 246)"
   112→   112→          opacity={i === last ? 1 : 0.65}
   113→   113→        />
   114→   114→      ))}
   115→   115→      <text x={xs[last]} y={ys[last] - 4} textAnchor="end" fontSize="6.5" fill="currentColor" opacity="0.6">
   116→   116→        S{points[last].semester ?? "?"}
   117→   117→      </text>
   118→   118→    </svg>
   119→   119→  );
   120→   120→}

   122→   122→function DeltaChip({ delta }: { delta: number }) {
   123→   123→  const up = delta > 0.0005;

   333→   333→
   334→   334→  const onRefresh = async () => {

   340→   340→      else toast.message("No published results on the ERP yet");

   343→   343→    } finally {
   344→   344→      setRefreshing(false);
   345→   345→    }
   346→   346→  };

   349→   349→    summary && summary.creditsEarned != null && summary.creditsOffered

   360→   360→    summary && sgpaValid && creditsValid

   370→   370→            aria-hidden="true"

   372→   372→            <GraduationCap className="size-4" />
   373→   373→          </span>
   374→   374→          <div className="min-w-0 flex-1">
   375→   375→            <CardTitle className="text-base tracking-tight">Academic record</CardTitle>
   376→   376→            <CardDescription className="text-xs">
   377→   377→              Published exam results from the ERP&apos;s Final Results page — grades, SGPA/CGPA and pending subjects.
   378→   378→            </CardDescription>
   379→   379→          </div>
   380→   380→          <div className="flex shrink-0 items-center gap-1.5 print:hidden">
   381→   381→            {data?.fetchedAt && (
   382→   382→              <span className="hidden text-[10px] text-muted-foreground sm:inline" title="Last fetched from the ERP">

   391→   391→            <Button

   394→   394→              className="h-7 gap-1 px-2 text-xs"
   395→   395→              onClick={onRefresh}
