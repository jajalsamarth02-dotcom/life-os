    80→/** Tiny CGPA sparkline from the REGULAR exams' printed career CGPA line. */
    81→function CgpaSparkline({ points }: { points: CgpaPoint[] }) {
    82→  if (points.length < 2) return null;
    83→  const W = 96;
    84→  const H = 26;
    85→  const PAD = 3;
    86→  const vals = points.map((p) => p.cgpa);
    87→  const min = Math.min(...vals);
    88→  const max = Math.max(...vals);
    89→  const span = Math.max(max - min, 0.8); // never amplify a near-flat line
    90→  const xs = points.map((_, i) => PAD + (i * (W - 2 * PAD)) / (points.length - 1));
    91→  const ys = vals.map((v) => H - PAD - ((v - min) / span) * (H - 2 * PAD));
    92→  const line = xs.map((x, i) => `${i ? "L" : "M"}${x.toFixed(1)},${ys[i].toFixed(1)}`).join(" ");
    93→  const area = `${line} L${xs[xs.length - 1].toFixed(1)},${H - PAD} L${xs[0].toFixed(1)},${H - PAD} Z`;
    94→  const last = points.length - 1;
    95→  return (
    96→    <svg
    97→      width={W}
    98→      height={H}
    99→      viewBox={`0 0 ${W} ${H}`}
   100→      className="mt-1.5"
   101→      role="img"
   102→      aria-label={`CGPA progression: ${vals.map((v) => v.toFixed(2)).join(" → ")}`}
   103→    >
   104→      <defs>
   105→        <linearGradient id="cgpa-spark-fill" x1="0" y1="0" x2="0" y2="1">
   106→          <stop offset="0%" stopColor="rgb(139 92 246)" stopOpacity="0.28" />
   107→          <stop offset="100%" stopColor="rgb(139 92 246)" stopOpacity="0" />
   108→        </linearGradient>
   109→      </defs>
   110→      <path d={area} fill="url(#cgpa-spark-fill)" />
   111→      <path d={line} fill="none" stroke="rgb(139 92 246)" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   112→      {xs.map((x, i) => (
   113→        <circle
   114→          key={i}
   115→          cx={x}
   116→          cy={ys[i]}
   117→          r={i === last ? 2.2 : 1.5}
   118→          fill="rgb(139 92 246)"
   119→          opacity={i === last ? 1 : 0.65}
   120→          className="transition-[r] hover:r-3"
   121→        >
   122→          <title>{`S${points[i].semester ?? "?"} · CGPA ${vals[i].toFixed(2)}`}</title>
   123→        </circle>
   124→      ))}
   125→      <text x={xs[last]} y={ys[last] - 4} textAnchor="end" fontSize="6.5" fill="currentColor" opacity="0.6">
   126→        S{points[last].semester ?? "?"}
   127→      </text>
   128→    </svg>
   129→  );
   130→}

   132→function DeltaChip({ delta }: { delta: number }) {
   133→  const up = delta > 0.0005;
   134→  const down = delta < -0.0005;
   135→  return (
   136→    <span
   137→      className={cn(
   138→        "inline-flex items-center gap-0.5 rounded-full border px-1.5 py-0.5 font-mono text-[10px] tabular-nums",
   139→        up
   140→          ? "border-emerald-500/40 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300"
   141→          : down
   142→            ? "border-rose-500/40 bg-rose-500/10 text-rose-700 dark:text-rose-300"
   143→            : "border-border bg-muted text-muted-foreground"
   144→      )}
   145→    >
   146→      {up && <TrendingUp className="size-3" aria-hidden="true" />}
   147→      {down && <TrendingDown className="size-3" aria-hidden="true" />}
   148→      {up ? "+" : ""}
   149→      {delta.toFixed(2)}
   150→    </span>
   151→  );

   175→function ExamRow({
   176→  exam,
   177→  defaultOpen = false,
   178→  focusNonce = 0,
   179→  closeNonce = 0,
   180→  accented = false,
   181→  careerDelta = null,
   182→}: {
   183→  exam: ErpResultExam;
   184→  defaultOpen?: boolean;
   185→  /** Bump to programmatically open + scroll to + flash this row. */
   186→  focusNonce?: number;
   187→  /** Bump to programmatically COLLAPSE this row — deep-links spotlight one row by closing its siblings. */
   188→  closeNonce?: number;
   189→  /** This row was the deep-link target — subtle violet left edge while open. */
   190→  accented?: boolean;
   191→  /** Career-CGPA change this REGULAR exam printed vs the previous one. */
   192→  careerDelta?: number | null;
   193→}) {
   194→  const [open, setOpen] = useState(defaultOpen);
   195→  const [flash, setFlash] = useState(false);
   196→  const rootRef = useRef<HTMLDivElement>(null);
   197→  const kindLabel = exam.examKind === "REMEDIAL" ? "Remedial" : exam.examKind === "REGULAR" ? "Regular" : "Exam";
   198→  useEffect(() => {
   199→    if (focusNonce > 0) {
   200→      setOpen(true);
   201→      setFlash(true);
   202→      rootRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   203→      const t = setTimeout(() => setFlash(false), 1800);
   204→      return () => clearTimeout(t);
   205→    }
   206→  }, [focusNonce]);
   207→  // Spotlight: when a sibling row is deep-linked, this one quietly collapses.
   208→  // Manual toggles are untouched — closeNonce only changes on deep-links.
   209→  useEffect(() => {
   210→    if (closeNonce > 0) setOpen((v) => (v ? false : v));
   211→  }, [closeNonce]);
   212→  return (
   213→    <div
   214→      ref={rootRef}
   215→      className={cn(
   216→        "overflow-hidden rounded-lg border transition-all duration-300",
   217→        exam.examKind === "REMEDIAL" ? "border-amber-500/25" : "border-border/70",
   218→        open && "bg-muted/30",

   223→      <button
   224→        type="button"

   233→          {kindLabel} · {exam.examSession ?? exam.examDateLabel ?? exam.label}
   234→        </span>
   235→        <span className="shrink-0 font-mono text-[11px] tabular-nums text-muted-foreground">
   236→          SGPA {exam.sgpa?.toFixed(2) ?? "—"}
   237→        </span>
   238→        <VerdictChip result={exam.result} />
   239→        <ChevronDown
   240→          className={cn("size-3.5 shrink-0 text-muted-foreground transition-transform", open && "rotate-180")}
   241→          aria-hidden="true"
   242→        />
   243→      </button>
   244→      {open && (
   245→        <div className="animate-in fade-in-0 slide-in-from-top-1 space-y-2 border-t px-3 py-2 duration-200">
   246→          <div className="flex flex-wrap items-center gap-x-3 gap-y-1 text-[11px] text-muted-foreground">
   247→            <span>{exam.examDateLabel ? `Held ${exam.examDateLabel}` : null}</span>

   255→              {exam.earnedCredit ?? "—"}/{exam.offeredCredit ?? "—"} credits this exam
   256→            </span>

   258→              <span>
   259→                career {exam.totalEarnedCredit}/{exam.totalOfferedCredit} · CGPA {exam.cgpa?.toFixed(2) ?? "—"}

   433→export function ResultsCard({ className }: { className?: string }) {
   434→  const queryClient = useQueryClient();
   435→  const rootRef = useRef<HTMLDivElement>(null);
   436→  const [refreshing, setRefreshing] = useState(false);

   439→  const [whatIfCredits, setWhatIfCredits] = useState("");
   440→  // Deep-link state: bumping focusNonce opens + scrolls to + flashes the
   441→  // targeted exam row (used by the remedial SGPA markers).
   442→  const [focusTarget, setFocusTarget] = useState<string | null>(null);
   443→  const [focusNonce, setFocusNonce] = useState(0);
   444→  const [storiesOpen, setStoriesOpen] = useState(false);
   445→  const query = useQuery({ queryKey: qk.results, queryFn: () => fetchResults() });
   446→  const data = query.data;
   447→  const summary = data ? deriveResultsSummary(data.exams) : null;
   448→  const deltas = data ? careerDeltas(data.exams) : null;
   449→  const stories = data ? subjectStories(data.exams) : [];
   450→  const openStoryCount = stories.filter((st) => !st.cleared).length;
   451→
   452→  const onFocusExam = (typeId: string) => {
   453→    setFocusTarget(typeId);
   454→    setFocusNonce((n) => n + 1);
   455→  };
   456→
   457→  const onRefresh = async () => {
   458→    setRefreshing(true);
   459→    try {
   460→      const fresh = await fetchResults(true);
