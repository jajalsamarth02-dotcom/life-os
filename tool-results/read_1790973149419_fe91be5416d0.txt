   116→function ErpConnectionCard() {
   117→  const qc = useQueryClient();
   118→  const { data: erp, isPending } = useQuery({
   119→    queryKey: qk.erpStatus,
   120→    queryFn: fetchErpStatus,
   121→    refetchInterval: 20_000,
   122→  });
   123→
   124→  const [username, setUsername] = useState("");
   125→  const [password, setPassword] = useState("");
   126→  const [connectError, setConnectError] = useState<ErpConnectFailure | null>(null);
   127→  const [confirmDisconnect, setConfirmDisconnect] = useState(false);
   128→
   129→  const invalidateAll = () => qc.invalidateQueries();
   130→
   131→  const connect = useMutation({
   132→    mutationFn: () => connectErp(username.trim(), password),
   133→    onSuccess: (res) => {
   134→      setConnectError(null);
   135→      setPassword("");
   136→      invalidateAll();
   137→      toast.success(`Connected as ${res.username}`, {
   138→        description: `Initial sync fetched ${res.snapshotSubjects} subject(s).`,
   139→      });
   140→    },
   141→    onError: (e) => {
   142→      const body: ErpConnectFailure =
   143→        e instanceof ApiError
   144→          ? { ok: false, errorCode: e.errorCode ?? "UNKNOWN", error: e.message, details: e.details }
   145→          : { ok: false, errorCode: "NETWORK", error: e.message };
   146→      setConnectError(body);
   147→    },
   148→  });

   150→  const disconnect = useMutation({
   151→    mutationFn: disconnectErp,
   152→    onSuccess: () => {
   153→      invalidateAll();
   154→      toast.success("ERP account disconnected");
   155→    },
   156→    onError: (e) => toast.error("Disconnect failed", { description: e.message }),
   157→  });
   158→
   159→  if (isPending || !erp) {
   160→    return (
   161→      <Card className="rounded-xl border-border/60">
   162→        <CardHeader className="pb-3">
   163→          <Skeleton className="h-5 w-40" />
   164→          <Skeleton className="h-3 w-64" />
   165→        </CardHeader>
   166→        <CardContent className="space-y-3">
   167→          <Skeleton className="h-10 w-full" />
   168→          <Skeleton className="h-10 w-full" />
   169→          <Skeleton className="h-10 w-28" />
   170→        </CardContent>
   171→      </Card>
   172→    );
   173→  }
   174→
   175→  return (
   176→    <Card className="rounded-xl border-border/60">
   177→      <CardHeader className="pb-3">
   178→        <CardTitle className="flex items-center gap-2 text-base tracking-tight">
   179→          <KeyRound className="size-4 text-primary" aria-hidden="true" />
   180→          ERP connection
   181→        </CardTitle>
   182→        <CardDescription>
   183→          Connect with your own student credentials from erp.gmiu.edu.in.
   184→        </CardDescription>
   185→      </CardHeader>
   186→      <CardContent className="space-y-4">
   187→        {erp.connected ? (
   188→          <div className="space-y-3">
   189→            <div className="flex flex-wrap items-center gap-3 rounded-lg border border-emerald-500/30 bg-emerald-500/10 p-4">
   190→              <CheckCircle2 className="size-6 shrink-0 text-emerald-600 dark:text-emerald-400" aria-hidden="true" />
   191→              <div className="min-w-0 flex-1">
   192→                <p className="text-sm font-semibold text-emerald-700 dark:text-emerald-400">
   193→                  Connected as {erp.username}
   194→                </p>
   195→                <p className="text-xs text-muted-foreground">
   196→                  Status: {erp.connectionStatus ?? "ok"} · Last verified: {fmtDT(erp.lastVerifiedAt)}
   197→                </p>
   198→              </div>
   199→              {erp.liveAvailable !== "unknown" && (
   200→                <Badge variant="outline" className="shrink-0">
   201→                  live sync: {erp.liveAvailable}
   202→                </Badge>
   203→              )}
   204→            </div>
   205→
   206→            <AlertDialog open={confirmDisconnect} onOpenChange={setConfirmDisconnect}>
   207→              <AlertDialogTrigger asChild>
   208→                <Button variant="outline" className="gap-1.5 text-red-600 hover:bg-red-500/10 hover:text-red-700 dark:text-red-400" aria-label="Disconnect ERP account">
   209→                  <LogOut className="size-4" aria-hidden="true" />
   210→                  Disconnect
   211→                </Button>

   229→                </AlertDialogFooter>

   232→          </div>

   235→            className="space-y-3"
   236→            onSubmit={(e) => {

   249→                onChange={(e) => setUsername(e.target.value)}

   251→                required
   252→              />

   254→            <div className="space-y-1.5">
   255→              <Label htmlFor="erp-password">ERP password</Label>

   268→              <Alert variant="destructive" className="border-red-500/40">
   269→                <AlertCircle className="size-4" />
   270→                <AlertTitle>
   271→                  {connectError.error}
   272→                  {connectError.errorCode && (

   276→                  )}
   277→                </AlertTitle>

   279→                  <AlertDescription className="mt-1 break-all font-mono text-[11px] leading-relaxed">

   285→
   286→            <Button type="submit" disabled={connect.isPending || !username.trim() || !password} className="gap-1.5">
   287→              {connect.isPending ? (
   288→                <>
   289→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   290→                  Connecting…
   291→                </>
   292→              ) : (
   293→                <>
   294→                  <Lock className="size-4" aria-hidden="true" />

   321→function SyncSettingsForm({ initial }: { initial: AppSettings }) {
   322→  const qc = useQueryClient();
   323→  const [target, setTarget] = useState<number>(initial.targetPercentage);
   324→  const [interval, setIntervalMin] = useState<number>(initial.syncIntervalMinutes);
   325→  const [autoSync, setAutoSync] = useState<boolean>(initial.autoSyncEnabled);
   326→
   327→  const save = useMutation({
   328→    mutationFn: () =>
   329→      putSettings({ targetPercentage: target, syncIntervalMinutes: interval, autoSyncEnabled: autoSync }),
   330→    onSuccess: () => {
   331→      qc.invalidateQueries();
   332→      toast.success("Settings saved", {
   333→        description: `Target ${target}% · sync every ${interval} min${autoSync ? " · auto-sync on" : " · auto-sync off"}`,
   334→      });
   335→    },
   336→    onError: (e) => toast.error("Could not save settings", { description: e.message }),
   337→  });
   338→
   339→  return (
   340→    <CardContent className="space-y-5">
   341→        <div className="space-y-2">
   342→          <div className="flex items-center justify-between">
   343→            <Label htmlFor="target-slider">Attendance target</Label>
   344→            <span className="rounded-md bg-muted px-2 py-0.5 text-sm font-semibold tabular-nums">{target}%</span>
   345→          </div>
   346→          <Slider
   347→            id="target-slider"
   348→            value={[target]}
   349→            min={50}
   350→            max={95}
   351→            step={1}
   352→            aria-label="Attendance target percentage"
   353→            onValueChange={(v) => setTarget(v[0] ?? 75)}

   355→          <p className="text-[11px] text-muted-foreground">Between 50% and 95%. Warnings are recalculated immediately.</p>

   358→        <div className="space-y-2">
   359→          <div className="flex items-center justify-between">

   469→function NotificationPrefsCard() {
   470→  const qc = useQueryClient();
   471→  const { data: settings } = useQuery({ queryKey: qk.settings, queryFn: fetchSettings });
   472→  const [draft, setDraft] = useState<Record<string, boolean> | null>(null);
   473→
   474→  const prefs = draft ?? settings?.notifyPrefs ?? {};
   475→  const enabledCount = NOTIFY_META.filter((m) => prefs[m.key] !== false).length;
   476→
   477→  const save = useMutation({
   478→    mutationFn: (next: Record<string, boolean>) => putSettings({ notifyPrefs: next }),
   479→    onSuccess: (_s, next) => {
   480→      setDraft(null);
   481→      qc.invalidateQueries({ queryKey: qk.settings });
   482→      const off = Object.entries(next).filter(([, v]) => !v).length;
   483→      toast.success(
   484→        off > 0 ? `Notifications updated — ${off} type${off === 1 ? "" : "s"} muted` : "All notifications enabled"
   485→      );
   486→    },
   487→    onError: (e) => toast.error("Could not save notification preferences", { description: e.message }),
   488→  });
   489→
   490→  const setPref = (key: string, value: boolean) => {
   491→    setDraft({ ...prefs, [key]: value });
   492→  };
   493→
   494→  const dirty =
   495→    draft !== null &&
   496→    NOTIFY_META.some((m) => (draft[m.key] ?? true) !== (settings?.notifyPrefs?.[m.key] ?? true));
   497→
   498→  return (
   499→    <Card className="rounded-xl border-border/60">
   500→      <CardHeader className="flex flex-row items-start justify-between space-y-0 pb-3">
   501→        <div>
   502→          <CardTitle className="flex items-center gap-2 text-base tracking-tight">
   503→            <Bell className="size-4 text-primary" aria-hidden="true" />
   504→            Notifications
   505→          </CardTitle>
   506→          <CardDescription>Choose which events create a notification</CardDescription>
   507→        </div>
   508→        <Badge variant="secondary" className="text-[11px] tabular-nums">
   509→          {enabledCount}/{NOTIFY_META.length} on
   510→        </Badge>
   511→      </CardHeader>
   512→      <CardContent className="space-y-3">
   513→        {!settings ? (
   514→          <div className="space-y-2">
   515→            {Array.from({ length: 3 }).map((_, i) => (
   516→              <Skeleton key={i} className="h-11 w-full rounded-lg" />
   517→            ))}
   518→          </div>
   519→        ) : (
   520→          <div className="grid gap-2 sm:grid-cols-2">
   521→            {NOTIFY_META.map((m) => {
   522→              const on = prefs[m.key] !== false;
   523→              const Icon = on ? m.icon : BellOff;
   524→              return (
   525→                <div
   526→                  key={m.key}
   527→                  className={cn(
   528→                    "flex items-center gap-3 rounded-lg border p-3 transition-colors",
   529→                    on ? "border-border/50 bg-card" : "border-border/40 bg-muted/40"
   530→                  )}
   531→                >
   532→                  <span
   533→                    className={cn(
   534→                      "flex size-8 shrink-0 items-center justify-center rounded-lg",
   535→                      on ? "bg-muted" : "bg-muted/60 opacity-50"
   536→                    )}
   537→                    aria-hidden="true"
   538→                  >
   539→                    <Icon className={cn("size-4", on ? m.className : "text-muted-foreground")} />
   540→                  </span>
   541→                  <div className="min-w-0 flex-1">
   542→                    <Label htmlFor={`notify-${m.key}`} className={cn("text-sm", !on && "text-muted-foreground")}>
   543→                      {m.label}
   544→                    </Label>
   545→                    <p className="truncate text-[11px] text-muted-foreground">{m.description}</p>
   546→                  </div>
   547→                  <Switch
   548→                    id={`notify-${m.key}`}
   549→                    checked={on}
   550→                    onCheckedChange={(v) => setPref(m.key, v)}
   551→                    disabled={save.isPending}
   552→                    aria-label={`${on ? "Disable" : "Enable"} ${m.label} notifications`}
   553→                  />
   554→                </div>
   555→              );
   556→            })}
   557→          </div>
   558→        )}
   559→        <div className="flex items-center justify-between gap-3">
   560→          <p className="text-[11px] text-muted-foreground">
   561→            Applies to future events — existing notifications are kept.
   562→          </p>
   563→          <Button
   564→            size="sm"
   565→            className="h-8 gap-1.5"
   566→            disabled={!dirty || save.isPending}
   567→            onClick={() => save.mutate(draft ?? {})}
   568→            aria-label="Save notification preferences"
   569→          >
   570→            {save.isPending ? <Loader2 className="size-3.5 animate-spin" aria-hidden="true" /> : <Save className="size-3.5" aria-hidden="true" />}
   571→            Save
   572→          </Button>
   573→        </div>
   574→      </CardContent>
