    98→/** Compute a snooze target from "now" for the given preset. */
    99→function snoozeTarget(preset: "h1" | "h3" | "tomorrow" | "week"): Date {
   100→  const d = new Date();
   101→  switch (preset) {
   102→    case "h1":
   103→      return new Date(d.getTime() + 60 * 60_000);
   104→    case "h3":
   105→      return new Date(d.getTime() + 3 * 60 * 60_000);
   106→    case "tomorrow":
   107→      d.setDate(d.getDate() + 1);
   108→      d.setHours(9, 0, 0, 0);
   109→      return d;
   110→    case "week":
   111→      d.setDate(d.getDate() + 7);
   112→      d.setHours(9, 0, 0, 0);
   113→      return d;
   114→  }
   115→}

   117→function ReminderRow({ r, onDeleted }: { r: Reminder; onDeleted: (r: Reminder) => void }) {

   124→  const overdue = r.status === "ACTIVE" && dueMs < 0;
   125→  const dueSoon = r.status === "ACTIVE" && dueMs >= 0 && dueMs <= 24 * 3600 * 1000;
   126→  const canExport = !done && !isWarn;
   127→

   129→    mutationFn: (status: string) => updateReminderStatus(r.id, status),

   133→    },
   134→    onError: (e) => {
   135→      qc.invalidateQueries();

   137→    },
   138→  });
   139→
   140→  const del = useMutation({
   141→    mutationFn: () => deleteReminder(r.id),
   142→    onSuccess: () => {
   143→      qc.invalidateQueries();

   196→          done ? "border-emerald-500 bg-emerald-500 text-white" : "border-muted-foreground/40 hover:border-emerald-500"

   479→const GROUP_ORDER: DueGroup[] = ["overdue", "today", "tomorrow", "week", "later", "done"];
   480→
   481→function RemindersList({ items }: { items: Reminder[] }) {
   482→  const now = useNow();
   483→  const [doneOpen, setDoneOpen] = useState(false);
   484→
   485→  const grouped = useMemo(() => {
   486→    const map = new Map<DueGroup, Reminder[]>();
   487→    for (const g of GROUP_ORDER) map.set(g, []);
   488→    for (const r of items) {
   489→      map.get(dueGroup(r.dueAt, r.status, now))?.push(r);
   490→    }
   491→    return map;
   492→  }, [items, now]);
   493→
   494→  if (items.length === 0) {
   495→    return (
   496→      <div className="flex flex-col items-center gap-2 rounded-xl border border-dashed border-border/80 bg-card/50 px-6 py-12 text-center">
   497→        <CalendarClock className="size-8 text-muted-foreground/50" aria-hidden="true" />
   498→        <p className="text-sm font-medium">No reminders here</p>
   499→        <p className="max-w-sm text-xs text-muted-foreground">
   500→          Active reminders from the ERP (exams, submissions, attendance warnings) and anything you add yourself will show up in this list.
   501→        </p>
   502→      </div>
   503→    );
   504→  }
   505→
   506→  const doneCount = grouped.get("done")?.length ?? 0;
   507→
   508→  return (
   509→    <div className="space-y-5">
   510→      {GROUP_ORDER.filter((g) => g !== "done").map((g) => {
   511→        const list = grouped.get(g) ?? [];
   512→        if (list.length === 0) return null;
   513→        const meta = GROUP_META[g];
   514→        return (
   515→          <section key={g} aria-label={meta.label}>
   516→            <h3 className={cn("mb-2 flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide", meta.className)}>
   517→              {g === "overdue" && <AlertTriangle className="size-3.5" aria-hidden="true" />}
   518→              {meta.label}
   519→              <span className="font-normal text-muted-foreground tabular-nums">({list.length})</span>
   520→            </h3>
   521→            <ul className="space-y-2">
   522→              {list.map((r) => (
   523→                <ReminderRow key={r.id} r={r} onDeleted={() => undefined} />
   524→              ))}
   525→            </ul>
   526→          </section>
   527→        );
   528→      })}
   529→
   530→      {doneCount > 0 && (
   531→        <Collapsible open={doneOpen} onOpenChange={setDoneOpen}>
   532→          <CollapsibleTrigger asChild>
   533→            <button
   534→              type="button"
   535→              className="flex w-full items-center gap-1.5 rounded-lg px-1 py-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground transition-colors hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   536→              aria-expanded={doneOpen}
   537→            >
   538→              <ChevronDown className={cn("size-3.5 transition-transform", doneOpen && "rotate-180")} aria-hidden="true" />
   539→              Done

   558→/** Compact opt-in banner for system due-reminders (only until a decision). */
   559→function BrowserRemindersBanner() {
   560→  const [state, setState] = useState<{ permission: string; enabled: boolean; dismissed: boolean }>({
   561→    permission: "default",
   562→    enabled: false,
   563→    dismissed: true,
   564→  });
   565→  const [busy, setBusy] = useState(false);
   566→
   567→  useEffect(() => {
   568→    const read = () => {
   569→      let dismissed = true;
   570→      try {
   571→        dismissed = window.localStorage.getItem("erp.browserReminders.bannerDismissed") === "1";
   572→      } catch {
   573→        /* non-fatal */
   574→      }
   575→      setState({
   576→        permission: browserRemindersPermission(),
   577→        enabled: browserRemindersEnabled(),
   578→        dismissed,
   579→      });
   580→    };
   581→    read();
   582→    window.addEventListener(BROWSER_REMINDERS_EVENT, read);
   583→    return () => window.removeEventListener(BROWSER_REMINDERS_EVENT, read);
   584→  }, []);
   585→
   586→  if (state.dismissed || state.permission !== "default") return null;
   587→
   588→  const dismiss = () => {
   589→    try {
   590→      window.localStorage.setItem("erp.browserReminders.bannerDismissed", "1");
   591→    } catch {
   592→      /* non-fatal */
   593→    }
   594→    setState((s) => ({ ...s, dismissed: true }));
   595→  };
   596→
   597→  const enable = async () => {
   598→    setBusy(true);
   599→    const result = await requestNotificationPermission();
   600→    setBusy(false);
   601→    if (result === "granted") {
   602→      setBrowserRemindersEnabled(true);
   603→      toast.success("Browser reminders on");
   604→    } else if (result === "denied") {
   605→      toast.error("Permission denied", { description: "Allow notifications for this site in your browser settings." });
   606→    }
   607→    dismiss();
   608→  };
   609→
   610→  return (
   611→    <div className="flex flex-wrap items-center gap-2.5 rounded-xl border border-primary/25 bg-primary/[0.05] px-4 py-3">
   612→      <CalendarClock className="size-4 shrink-0 text-primary" aria-hidden="true" />
   613→      <p className="min-w-0 flex-1 text-xs sm:text-sm">
   614→        <span className="font-medium">Never miss a deadline.</span>{" "}
   615→        <span className="text-muted-foreground">Get a system alert when a reminder becomes due.</span>
   616→      </p>
   617→      <div className="flex items-center gap-1.5">
   618→        <Button size="sm" className="h-7 gap-1 px-2.5 text-xs" onClick={enable} disabled={busy}
   619→          aria-label="Enable browser due-reminder notifications">
   620→          {busy ? <Loader2 className="size-3.5 animate-spin" aria-hidden="true" /> : <BellRing className="size-3.5" aria-hidden="true" />}
   621→          Enable
   622→        </Button>
   623→        <Button variant="ghost" size="sm" className="h-7 px-2 text-xs text-muted-foreground" onClick={dismiss}
   624→          aria-label="Dismiss browser reminders banner">
   625→          Not now
   626→        </Button>
   627→      </div>
   628→    </div>
   629→  );
   630→}
